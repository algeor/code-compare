+# Pre-delete Job that drains FaultHandler CRs before Helm removes the Kopf
+# controller that owns their finalizers.
+#
+# Without this hook, helm uninstall removes the Kopf controller pod and the
+# fl-handlers ArgoCD Application concurrently. If the controller pod goes first,
+# no process is left to clear kopf.zalando.org/KopfFinalizerMarker from each
+# FaultHandler CR, leaving the CRs stuck and the namespace permanently in
+# Terminating state.
+#
+# ArgoCD's cascade-delete only removes resources it provisioned via its own
+# Application — FaultHandlers created outside ArgoCD are not in its inventory
+# and will not be touched. The hook therefore deletes all FaultHandlers directly,
+# which triggers Kopf (still running at this point) to process the finalizer and
+# remove each CR cleanly.
+#
+# This hook runs at weight 0 (after RBAC at -1) and:
+#   1. Deletes the fl-handlers Application so ArgoCD stops re-syncing FaultHandlers.
+#   2. Deletes all FaultHandler CRs in the namespace explicitly.
+#   3. Polls until no FaultHandler CRs remain (Kopf clears the finalizer on each).
+# Only then does Helm proceed to delete the controllers.
+{{- if .Values.handlerRegistry.enabled }}
+apiVersion: batch/v1
+kind: Job
+metadata:
+  name: {{ include "fl-control-plane.fullname" . }}-argocd-wait-job
+  namespace: {{ .Release.Namespace }}
+  labels:
+    {{- include "fl-control-plane.labels" . | nindent 4 }}
+    app.kubernetes.io/component: argocd-wait-hook
+  annotations:
+    "helm.sh/hook": pre-delete
+    "helm.sh/hook-weight": "0"
+    "helm.sh/hook-delete-policy": before-hook-creation,hook-succeeded,hook-failed
+spec:
+  backoffLimit: 0
+  activeDeadlineSeconds: {{ .Values.handlerRegistry.waitJobActiveDeadlineSeconds | default 300 }}
+  template:
+    spec:
+      serviceAccountName: {{ include "fl-control-plane.fullname" . }}-argocd-wait
+      restartPolicy: Never
+      containers:
+        - name: argocd-wait
+          image: {{ .Values.image.alpineK8s }}
+          env:
+            - name: TIMEOUT
+              value: "{{ sub (.Values.handlerRegistry.waitJobActiveDeadlineSeconds | default 300) 60 }}"
+          command:
+            - /bin/sh
+            - -c
+            - |
+              set -eu
+              NS="{{ .Release.Namespace }}"
+
+              echo "==> Deleting fl-handlers ArgoCD Application..."
+              if kubectl get application fl-handlers -n "$NS" >/dev/null 2>&1; then
+                kubectl delete application fl-handlers -n "$NS"
+                echo "    Application deleted — ArgoCD will no longer re-sync FaultHandlers."
+              else
+                echo "    fl-handlers Application not found — skipping delete."
+              fi
+
+              echo "==> Deleting all FaultHandler CRs (triggers Kopf finalizer processing)..."
+              kubectl delete faulthandlers --all -n "$NS" --wait=false 2>/dev/null && \
+                echo "    FaultHandler deletion requests issued." || \
+                echo "    No FaultHandlers to delete."
+
+              echo "==> Waiting for all FaultHandler CRs to be removed..."
+              INTERVAL=5
+              elapsed=0
+              while true; do
+                count=$(kubectl get faulthandlers -n "$NS" -o name 2>/dev/null | wc -l | tr -d ' ')
+                if [ "$count" = "0" ]; then
+                  echo "    All FaultHandlers removed."
+                  break
+                fi
+                if [ "$elapsed" -ge "$TIMEOUT" ]; then
+                  echo "ERROR: Timed out waiting for FaultHandlers to be removed. Remaining:"
+                  kubectl get faulthandlers -n "$NS" --no-headers
+                  exit 1
+                fi
+                echo "    $count FaultHandler(s) still present — waiting ${INTERVAL}s (${elapsed}s elapsed)..."
+                sleep "$INTERVAL"
+                elapsed=$((elapsed + INTERVAL))
+              done
+
+              echo "==> Pre-delete complete. Helm may now remove the controllers."
+{{- end }}
