+def _target_from_url(raw_url: str) -> XmakeConsoleLogTarget | None:
+    parsed = urlparse(raw_url).geturl()
+    parsed_url = urlparse(parsed)
+    if parsed_url.scheme != "https" or not _is_xmake_host(parsed_url):
+        return None
+
+    raw_segments = [segment for segment in parsed_url.path.split("/") if segment]
+    decoded_segments = [unquote(segment) for segment in raw_segments]
+    for job_index in range(len(decoded_segments) - 2, -1, -1):
+        if decoded_segments[job_index] != "job":
+            continue
+        build_index = job_index + 2
+        if build_index >= len(decoded_segments) or not decoded_segments[build_index].isdigit():
+            continue
+        job_name = decoded_segments[job_index + 1]
+        build_number = decoded_segments[build_index]
+        if not _is_supported_console_suffix(decoded_segments[build_index + 1 :]):
+            continue
+        console_text_url = _console_text_url(parsed_url, raw_segments[: build_index + 1])
+        artifact_name = _artifact_name(job_name, build_number)
+        return XmakeConsoleLogTarget(
+            job_name=job_name,
+            build_number=build_number,
+            console_text_url=console_text_url,

+def _add_unique_target(
+    targets_by_key: dict[str, XmakeConsoleLogTarget],
+    target: XmakeConsoleLogTarget,
+) -> None:
+    """Insert a target by artifact key, suffixing only later URL collisions.
+
+    The first target for a sanitized artifact key keeps the bare key. When a
+    later target resolves to the same key but points at a different Jenkins
+    console URL, only that later target is rewritten to a hash-suffixed key.
+    """
+    existing = targets_by_key.get(target.artifact_key)
+    if existing is None or existing.console_text_url == target.console_text_url:
+        targets_by_key.setdefault(target.artifact_key, target)
+        return
+
+    digest = hashlib.sha256(target.console_text_url.encode("utf-8")).hexdigest()[:12]
+    disambiguated_key = target.artifact_key.removesuffix("_console_log.txt")
+    targets_by_key[f"{disambiguated_key}_{digest}_console_log.txt"] = target.model_copy(
+        update={"artifact_key": f"{disambiguated_key}_{digest}_console_log.txt"}
+    )
