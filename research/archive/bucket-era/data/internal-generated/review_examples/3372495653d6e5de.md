+  echo "==> Resolved commit: ${COMMIT_SHA:0:12}"
+  if git rev-parse --verify "refs/tags/${TAG_NAME}" &>/dev/null; then
+    # Verify the existing tag points to the same commit we intend to release.
+    # A mismatch means a prior failed run or manual tag creation used a different SHA.
+    EXISTING_TAG_SHA="$(git rev-list -n 1 "refs/tags/${TAG_NAME}")"
+    if [[ "$EXISTING_TAG_SHA" != "$COMMIT_SHA" ]]; then
+      echo "ERROR: Tag '${TAG_NAME}' already exists but points to ${EXISTING_TAG_SHA:0:12}, not ${COMMIT_SHA:0:12}." >&2
+      echo "       Delete it with: git tag -d ${TAG_NAME} && git push origin :${TAG_NAME}" >&2
+      exit 1
+    fi
+    echo "==> Tag '${TAG_NAME}' already exists at the correct commit — reusing it"
+  else
+    echo "==> Creating tag  : ${TAG_NAME}"
+    git tag "$TAG_NAME" "$COMMIT_SHA"
+    git push origin "$TAG_NAME"
+    echo "    Tag '${TAG_NAME}' pushed."
+  fi
+fi
+
+# ---------------------------------------------------------------------------
+# Create release branch
+# ---------------------------------------------------------------------------
+if git rev-parse --verify "refs/heads/${RELEASE_BRANCH}" &>/dev/null; then
+  echo "ERROR: Local branch '${RELEASE_BRANCH}' already exists." >&2
+  echo "       Delete it with: git branch -D ${RELEASE_BRANCH}" >&2
+  exit 1
+fi
+
+git checkout -b "$RELEASE_BRANCH" "$COMMIT_SHA"
+echo "==> Created branch '${RELEASE_BRANCH}' at ${COMMIT_SHA:0:12}"
+
+# ---------------------------------------------------------------------------
+# Modify .pipeline/config.yml — set this branch as productive and enable Promote
+# ---------------------------------------------------------------------------
+CONFIG_FILE="${REPO_DIR}/.pipeline/config.yml"
+
+# Tell Piper this branch is the productive branch so all stages including
+# Promote actually run (Piper skips Promote on non-productive branches).
+# Escape sed replacement metacharacters (& \ |) so a version like "1.0.&rc1"
+# doesn't expand & to the matched text and silently corrupt the YAML.
+_SAFE_BRANCH="$(printf '%s' "$RELEASE_BRANCH" | sed 's/[\\&|]/\\&/g')"
+sed -i.bak "s|  productiveBranch: 'main'|  productiveBranch: '${_SAFE_BRANCH}'|" "$CONFIG_FILE"
+rm -f "${CONFIG_FILE}.bak"
+if ! grep -qF "productiveBranch: '${RELEASE_BRANCH}'" "$CONFIG_FILE"; then
+  echo "ERROR: Failed to set productiveBranch in .pipeline/config.yml — check YAML formatting." >&2
+  exit 1
+fi
+echo "==> Set productiveBranch to '${RELEASE_BRANCH}' in .pipeline/config.yml"
+
+# Enable Promote
+if grep -q "sapCallStagingService: true" "$CONFIG_FILE"; then
+  echo "==> Promote already enabled in .pipeline/config.yml (no change needed)"
+else
+  sed -i.bak 's/    sapCallStagingService: false/    sapCallStagingService: true/' "$CONFIG_FILE"
+  rm -f "${CONFIG_FILE}.bak"
+  if ! grep -q "sapCallStagingService: true" "$CONFIG_FILE"; then
+    echo "ERROR: Failed to enable sapCallStagingService in .pipeline/config.yml — check YAML formatting." >&2
+    exit 1
+  fi
+  echo "==> Enabled sapCallStagingService in .pipeline/config.yml"
+fi
+
+# Disable kubernetesDeploy in Acceptance — the release branch does not need a
+# test deploy; skipping it prevents helm failures from blocking the Promote stage.
+python3 - "$CONFIG_FILE" << 'PYEOF'
+import sys
+path = sys.argv[1]
+with open(path) as f:
+    lines = f.readlines()
+out = []
+in_acceptance = False
+already_patched = False
+for i, line in enumerate(lines):
+    out.append(line)
+    # Detect entry into the Acceptance: stage block (two-space indent = top-level stage key)
+    if line.rstrip() == '  Acceptance:':
+        in_acceptance = True
+        already_patched = False
+        continue
+    if in_acceptance:
+        stripped = line.strip()
+        # If we already have kubernetesDeploy here, nothing to do
+        if stripped.startswith('kubernetesDeploy:'):
+            already_patched = True
+            in_acceptance = False
+            continue
+        # Leaving the Acceptance block (next top-level stage or steps key)
+        if line.startswith('  ') and not line.startswith('    ') and stripped and not stripped.startswith('#'):
+            if not already_patched:
+                # Insert before this line
+                out.insert(-1, '    kubernetesDeploy: false\n')
+            in_acceptance = False
+
+# Handle case where Acceptance is the last block
+if in_acceptance and not already_patched:
+    out.append('    kubernetesDeploy: false\n')
+
+with open(path, 'w') as f:
+    f.writelines(out)
+PYEOF
+if ! grep -A5 '  Acceptance:' "$CONFIG_FILE" | grep -q 'kubernetesDeploy: false'; then
+  echo "ERROR: Failed to disable kubernetesDeploy in Acceptance stage — check config.yml structure." >&2
+  exit 1
+fi
+echo "==> Disabled kubernetesDeploy in Acceptance stage"
+
+# ---------------------------------------------------------------------------
+# Commit and push
+# ---------------------------------------------------------------------------
+git add .pipeline/config.yml
+git commit -m "[INTERNAL] Enable promote stage for release ${VERSION}"
+git push origin "$RELEASE_BRANCH"
+_BRANCH_PUSHED=true
+echo "==> Branch '${RELEASE_BRANCH}' pushed"
+
+# ---------------------------------------------------------------------------
+# Trigger the workflow via workflow_dispatch
+# (push triggers evaluate against the workflow on the default branch, not
+# the new branch, so the first push to rel-* won't auto-trigger Piper)
+# ---------------------------------------------------------------------------
+echo ""
+echo "==> Triggering piper.yaml workflow on '${RELEASE_BRANCH}'..."
+_TRIGGER_TIME=$(date -u +"%Y-%m-%dT%H:%M:%SZ")
+
+gh workflow run "$WORKFLOW_FILE" \
+  --repo "$GH_REPO" \
+  --ref "$RELEASE_BRANCH"
+
+# ---------------------------------------------------------------------------
+# Wait for the run to appear
+# ---------------------------------------------------------------------------
+echo "==> Waiting for run to appear..."
+_RUN_ID=""
+for _i in $(seq 1 40); do
+  _RUN_ID=$(gh run list \
+    --repo "$GH_REPO" \
+    --workflow "$WORKFLOW_FILE" \
+    --branch "$RELEASE_BRANCH" \
+    --limit 5 \
+    --json databaseId,status,createdAt \
+    2>/dev/null \
+    | jq -r --arg since "$_TRIGGER_TIME" \
+        '[.[] | select(.createdAt >= $since)] | .[0].databaseId // empty' \
+    2>/dev/null || true)
+  if [[ -n "$_RUN_ID" && "$_RUN_ID" != "null" ]]; then
+    break
+  fi
+  printf "."
+  sleep 5
+done
+echo ""
+
+if [[ -z "$_RUN_ID" || "$_RUN_ID" == "null" ]]; then
+  # The workflow was dispatched but no run appeared in the poll window.
+  # Preserve the remote branch so the in-flight run can complete and be
+  # investigated manually — do not let _cleanup delete it.
+  echo "ERROR: No Piper run appeared for '${RELEASE_BRANCH}' within the poll window." >&2
+  echo "       The workflow was dispatched — check manually:" >&2
+  echo "       gh run list --repo ${GH_REPO} --workflow ${WORKFLOW_FILE} --branch ${RELEASE_BRANCH}" >&2
+  trap - EXIT
+  git checkout "$ORIGINAL_BRANCH" 2>/dev/null || git checkout main 2>/dev/null || true
+  if [[ "$_STASH_APPLIED" == "true" ]]; then
+    echo "==> Restoring stashed changes..."
+    git stash pop 2>/dev/null || true
+  fi
+  exit 1
+fi
+
+echo "==> Run ID: ${_RUN_ID}"
+echo "    Watching progress (Ctrl-C stops watching; the run continues in CI)..."
+echo ""
+
+# ---------------------------------------------------------------------------
+# Watch run to completion — retry gh run watch on network errors; only stop
+# when the run reaches a terminal conclusion (success/failure/cancelled/etc.).
+# Max wait: 360 iterations × 10s = 1 hour before giving up.
+# ---------------------------------------------------------------------------
+RUN_FAILED=false
+_WATCH_MAX_ITERATIONS=360
+_WATCH_ITER=0
+while true; do
+  (( _WATCH_ITER++ )) || true
+  if [[ $_WATCH_ITER -gt $_WATCH_MAX_ITERATIONS ]]; then
+    echo "ERROR: Timed out waiting for run ${_RUN_ID} after 1 hour." >&2
+    RUN_FAILED=true
+    break
+  fi
+
+  gh run watch "$_RUN_ID" \
+    --repo "$GH_REPO" \
+    --exit-status && break || true
+
+  CONCLUSION=$(gh run view "$_RUN_ID" \
+    --repo "$GH_REPO" \
+    --json conclusion \
+    --jq '.conclusion' 2>/dev/null || true)
+
+  if [[ "$CONCLUSION" == "success" ]]; then
+    break
+  elif [[ "$CONCLUSION" == "failure" || "$CONCLUSION" == "cancelled" || \
+          "$CONCLUSION" == "timed_out" || "$CONCLUSION" == "skipped" || \
+          "$CONCLUSION" == "startup_failure" || "$CONCLUSION" == "action_required" ]]; then
+    RUN_FAILED=true
+    break
+  elif [[ -n "$CONCLUSION" && "$CONCLUSION" != "null" && \
+          "$CONCLUSION" != "in_progress" && "$CONCLUSION" != "queued" && \
+          "$CONCLUSION" != "waiting" && "$CONCLUSION" != "requested" && \
+          "$CONCLUSION" != "pending" ]]; then
+    # Unknown terminal conclusion — treat as failure rather than loop forever
+    echo "==> Unknown run conclusion '${CONCLUSION}' — treating as failure." >&2
+    RUN_FAILED=true
+    break
+  fi
+  # No terminal conclusion yet — network blip; wait and re-attach
+  echo "==> gh run watch disconnected (iter ${_WATCH_ITER}/${_WATCH_MAX_ITERATIONS}), reconnecting in 10s..."
+  sleep 10
+done
+
+# ---------------------------------------------------------------------------
+# Parse promoted images from full run logs
+# ---------------------------------------------------------------------------
+echo ""
+echo "==> Fetching run logs to parse promoted images..."
+_ALL_LOGS=$(gh run view "$_RUN_ID" \
+  --repo "$GH_REPO" \
+  --log 2>/dev/null || true)
+
+IMAGES_LINE=$(echo "$_ALL_LOGS" | grep "Promoted Docker images are" | tail -1 || true)
+
+# Warn if multiple "Promoted Docker images are" lines appear — that is anomalous
+# and means tail -1 silently discarded earlier lines.
+_IMAGES_LINE_COUNT=$(echo "$_ALL_LOGS" | grep -c "Promoted Docker images are" || true)
+if [[ "$_IMAGES_LINE_COUNT" -gt 1 ]]; then
+  echo "  WARNING: Found ${_IMAGES_LINE_COUNT} 'Promoted Docker images are' log lines — using last one." >&2
+fi
+
+# ---------------------------------------------------------------------------
+# Print results
+# ---------------------------------------------------------------------------
+echo ""
+echo "============================================================"
+echo "  Release ${VERSION}"
+echo "============================================================"
+
+RELEASE_TAG=""
+if [[ -n "$IMAGES_LINE" ]]; then
+  # entry format from Piper logs: "registry.host/image-name:tag"
+  # The bracket content is parsed and the registry host prefix stripped to
+  # construct the Artifactory sub-path.
+  BRACKET_CONTENT=$(echo "$IMAGES_LINE" | python3 -c "
+import sys, re
+line = sys.stdin.read()
+m = re.search(r\"\[([^\]]+)\]\", line)
+if m:
+    print(m.group(1))
+")
+  if [[ -z "$BRACKET_CONTENT" ]]; then
+    echo "  WARNING: Found 'Promoted Docker images are' but could not parse image list." >&2
+    echo "  Raw line: ${IMAGES_LINE}" >&2
+  else
+    echo ""
+    while IFS= read -r entry; do
+      entry="${entry#"${entry%%[![:space:]]*}"}"
+      entry="${entry%"${entry##*[![:space:]]}"}"
+      [[ -z "$entry" ]] && continue
+      IMAGE_REF="${entry#*/}"
+      IMAGE_NAME="${IMAGE_REF%%:*}"
+      IMAGE_TAG_VAL="${IMAGE_REF#*:}"
+      RELEASE_TAG="${IMAGE_TAG_VAL}"
+      echo "  ${ARTIFACTORY_BASE}/${IMAGE_NAME}:${IMAGE_TAG_VAL}"
+    done <<< "$(echo "$BRACKET_CONTENT" | tr ' ' '\n')"
+    echo ""
+    echo "  JFrog UI:"
+    echo "  https://int.repositories.cloud.sap/ui/repos/tree/General/proxy-deploy-releases-hyperspace-docker"
+  fi
+else
+  echo ""
+  echo "  WARNING: Could not parse promoted image URLs from the run logs." >&2
+  echo "  Full logs: gh run view ${_RUN_ID} --repo ${GH_REPO} --log" >&2
+fi
+
+echo "============================================================"
+echo ""
+
+# ---------------------------------------------------------------------------
+# On failure: preserve the branch for debugging; restore stash and original branch
+# ---------------------------------------------------------------------------
+if [[ "$RUN_FAILED" == "true" ]]; then
+  echo "ERROR: Piper run failed — preserving '${RELEASE_BRANCH}' for debugging." >&2
+  echo "       Full logs: gh run view ${_RUN_ID} --repo ${GH_REPO} --log" >&2
+  trap - EXIT
+  git checkout "$ORIGINAL_BRANCH" 2>/dev/null || git checkout main 2>/dev/null || true
+  if [[ "$_STASH_APPLIED" == "true" ]]; then
+    echo "==> Restoring stashed changes..."
+    git stash pop 2>/dev/null || true
+  fi
+  exit 1
+fi
+
+# ---------------------------------------------------------------------------
+# Success: delete release branch and restore working tree
+# ---------------------------------------------------------------------------
+echo "==> Deleting release branch '${RELEASE_BRANCH}'..."
+trap - EXIT
+git checkout "$ORIGINAL_BRANCH" 2>/dev/null || git checkout main 2>/dev/null || true
+git branch -D "$RELEASE_BRANCH"
+git push origin --delete "$RELEASE_BRANCH"
+echo "    Branch '${RELEASE_BRANCH}' deleted locally and remotely."
+
+if [[ "$_STASH_APPLIED" == "true" ]]; then
+  echo "==> Restoring stashed changes..."
+  git stash pop \
+    || echo "  WARNING: Stash pop had conflicts — resolve manually with: git stash pop" >&2
+fi
+
+echo ""
+echo "Release ${VERSION} complete."
+
+_DEPLOY_TAG="${RELEASE_TAG:-<release-tag>}"
+echo ""
+echo "To deploy the released version run:"
+echo "  ${SCRIPT_DIR}/pipeline-fl-lcm.sh deploy --namespace <namespace> --kubeconfig <path_to_kubeconfig> --env <dev|prod> --release-tag ${_DEPLOY_TAG}"
+echo ""
