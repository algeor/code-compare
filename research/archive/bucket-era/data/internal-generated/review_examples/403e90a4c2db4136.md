+def _append_console_segment(
+    chunks: list[bytes],
+    remaining_bytes: int,
+    body: bytes,
+) -> tuple[int, bool]:
+    """Append as much of one segment as fits in the remaining budget."""
+    if not body:
+        return remaining_bytes, False
+    if remaining_bytes <= 0:
+        return remaining_bytes, True
+
+    normalized = body if body.endswith(b"\n") else body + b"\n"
+    separator = b"" if not chunks else b"\n"
+    separator_cost = len(separator)
+    if separator_cost >= remaining_bytes:
+        return remaining_bytes, True
+
+    available_for_body = remaining_bytes - separator_cost
+    if len(normalized) <= available_for_body:
+        if separator:
+            chunks.append(separator)
+        chunks.append(normalized)
+        return remaining_bytes - separator_cost - len(normalized), False
+
+    truncated = normalized[:available_for_body]
+    if separator:
+        chunks.append(separator)
+    chunks.append(truncated)
+    return 0, True
