+    metadata:
+      labels:

+        app.kubernetes.io/component: postgresql-readiness-job
+      annotations:
+        # On a fresh namespace the Istio proxy starts before the wait container
+        # (native sidecar ordering), but Istiod hasn't yet distributed mTLS certs
+        # to the pod — the proxy exists but can't establish mTLS to Postgres yet.
+        # holdApplicationUntilProxyStarts delays the wait container until the proxy
+        # is fully initialised (certs received, iptables rules in place), so
+        # pg_isready runs inside a working mesh rather than against an unmeshed port.
+        # Without this: "temporal-postgres-postgresql:5432 - no response" for the
+        # entire duration of the hook timeout, blocking every fresh install/upgrade.
+        proxy.istio.io/config: '{"holdApplicationUntilProxyStarts": true}'
+        # Postgres speaks plain TCP — exclude port 5432 from outbound interception
+        # so the sidecar doesn't try to mTLS-wrap the pg_isready connection.
+        traffic.sidecar.istio.io/excludeOutboundPorts: "5432"
+    spec:
+      restartPolicy: OnFailure
+      containers:
+        - name: wait
+          image: {{ .Values.postgresql.image.repository }}:{{ .Values.postgresql.image.tag }}
+          command:

+"""Logging configuration for FL Control Plane.
+
+Call configure_logging() once at process startup (app lifespan or worker entry
+point) before any other module emits a log line.
+
+Both structlog loggers and stdlib loggers (workflow.logger, uvicorn, sqlalchemy)
+are routed through the same JSON pipeline so every pod log line is a JSON
+object with a consistent shape.
+"""
+
+import logging
+import sys
+
+import structlog
+
+_VALID_LOG_LEVELS = ("DEBUG", "INFO", "WARNING", "ERROR")
+
+_SHARED_PROCESSORS: list = [
+    structlog.contextvars.merge_contextvars,
+    structlog.stdlib.add_logger_name,
+    structlog.processors.add_log_level,
+    structlog.stdlib.ExtraAdder(),
+    structlog.processors.TimeStamper(fmt="iso", utc=True),
+]
+
+
+def configure_logging(log_level: str) -> None:
+    """Wire structlog + stdlib logging to stdout at the requested level.
+
+    All log output — structlog loggers and stdlib loggers alike (workflow.logger,
+    uvicorn, sqlalchemy) — is rendered as JSON so Kyma/BTP Cloud Logging can
+    index every field.
+
+    Args:
+        log_level: One of DEBUG, INFO, WARNING, ERROR.
+
+    Raises:
+        ValueError: When log_level is not one of the valid values.
+    """
+    if log_level not in _VALID_LOG_LEVELS:
+        raise ValueError(
+            f"invalid log_level {log_level!r}; expected one of {_VALID_LOG_LEVELS}"
+        )
+
+    level = getattr(logging, log_level)
+
+    structlog.configure(
+        processors=_SHARED_PROCESSORS
+        + [
+            structlog.stdlib.ProcessorFormatter.wrap_for_formatter,
+        ],
+        wrapper_class=structlog.make_filtering_bound_logger(level),
+        logger_factory=structlog.stdlib.LoggerFactory(),
+        # False: modules that call get_logger() at import time would
+        # otherwise cache an unconfigured logger and bypass the JSON renderer.
+        cache_logger_on_first_use=False,
+    )
+
+    formatter = structlog.stdlib.ProcessorFormatter(
+        # foreign_pre_chain handles records from stdlib loggers (workflow.logger,
+        # uvicorn, etc.) before they hit the shared processors above.
+        foreign_pre_chain=_SHARED_PROCESSORS,
+        processors=[
+            structlog.stdlib.ProcessorFormatter.remove_processors_meta,
+            structlog.processors.JSONRenderer(),
+        ],
+    )
+
+    handler = logging.StreamHandler(sys.stdout)
+    handler.setFormatter(formatter)
+
+    root = logging.getLogger()
+    root.handlers.clear()
+    root.addHandler(handler)
+    root.setLevel(level)

+"""Document-ingestion Temporal workflow definition.
+
+Sandbox isolation
+-----------------
+The Temporal Python SDK re-executes the workflow module through a custom
+import hook (``workflow_sandbox/_importer.py``) for every workflow
+activation. This gives each workflow execution its own module namespace so
+that module-level global state (e.g. a cached singleton, a module-level
+counter) cannot leak between concurrent executions or corrupt deterministic
+replay.
+
+Practical consequences:
+
+* **Module-level globals are per-execution, not per-process.** A value
+  written to a module global inside workflow code is invisible to other
+  workflow executions running in the same worker process.
+
+* **Environment variables are not re-read per activation.** ``os.environ``
+  is part of the outer process state, not the sandboxed namespace, so
+  ``os.environ["FOO"]`` reads the same value across all executions. This is
+  generally what you want, but it means you cannot change env vars between
+  activations and expect workflow code to see the new value mid-run.
+
+* **``workflow.unsafe.imports_passed_through()``** opts specific imports out
+  of re-execution, passing the already-imported outer-process module directly
+  into the sandbox. Activity references and Temporal SDK internals are wrapped
+  in this block precisely because they must be the same object instances the
+  worker registered — a re-imported copy in a fresh namespace would not match.
+
+* **No I/O, no ``datetime.now()``, no ``random()`` in workflow code.** These
+  are banned not by the sandbox itself but by the determinism requirement:
+  Temporal replays workflow history on retry, and any non-deterministic call
+  would produce a different execution path than the original, causing a
+  non-determinism error.
+
+Activities are not sandboxed — they run as plain coroutines on the shared
+event loop with full access to process state, env vars, and external I/O.
+"""
+
+from datetime import timedelta
+
+from temporalio import workflow
+from temporalio.common import RetryPolicy
+
+with workflow.unsafe.imports_passed_through():
+    from temporalio.workflow import execute_activity
+
+    from fl_control_plane.data_extractor.activity import extract_data_activity
+    from fl_control_plane.metadata_extractor.activity import extract_metadata_activity
+    from fl_control_plane.temporal.models import (
+        DataExtractionResult,
+        IngestionRequest,
+        MetadataExtractionResult,
+        WorkflowResult,
+    )
+
+
+
+@workflow.defn
+class PipelineInspectionWorkflow:
+    """Temporal workflow that drives the pipeline inspection stage.
+
+    Orchestrates metadata extraction followed by data extraction, mapping
+    results to a WorkflowResult consumed by the caller.
+
+    Activity failures raise ``ApplicationError`` and propagate through
+    Temporal's retry policy. ``ConfigError`` and ``InvalidInput`` types are
+    non-retryable and fail the workflow immediately; ``TransientFailure`` is
+    retried up to ``maximum_attempts`` times with exponential backoff.
+
+    Determinism contract: this class contains no I/O, no datetime.now(), no
+    random calls, and no asyncio.sleep. All side effects are delegated to
+    extract_metadata_activity and extract_data_activity.
+    """
+
+    @workflow.run
+    async def run(self, request: IngestionRequest) -> WorkflowResult:
+        """Execute pipeline inspection for a single inspection request.
+
+        Args:
+            request: Ingestion parameters including tenant, source system, and
+                inspection identifiers.
+
+        Returns:
+            WorkflowResult carrying both activity outcomes and the name of the
+            next pipeline stage.
+        """
+        _activity_retry_policy = RetryPolicy(
+            maximum_attempts=3,
+            initial_interval=timedelta(seconds=5),
+            backoff_coefficient=2.0,
+            maximum_interval=timedelta(seconds=60),
+            non_retryable_error_types=["ConfigError", "InvalidInput"],
+        )
+
+        metadata_result: MetadataExtractionResult = await execute_activity(
+            extract_metadata_activity,
+            request,
+            start_to_close_timeout=timedelta(minutes=5),
+            retry_policy=_activity_retry_policy,
+        )
+
+        data_result: DataExtractionResult = await execute_activity(
+            extract_data_activity,
+            request,
+            start_to_close_timeout=timedelta(minutes=30),
+            retry_policy=_activity_retry_policy,
+        )
+
+        return WorkflowResult(
+            inspection_id=request.inspection_id,
+            tenant_id=request.tenant_id,
+            metadata_extraction_status=metadata_result.status,
+            metadata_summary=metadata_result.metadata_summary,
+            metadata_reference=metadata_result.metadata_reference,
+            data_extraction_status=data_result.status,
+            manifest_reference=data_result.manifest_reference,
+        )
