+"""Applicability evaluation logic for the Handler Orchestrator.
+
+Implements the two handler-level evaluation stages:
+  Level 2 — Infrastructure Filter (CI system, repo patterns, trigger scope, target branch)
+  Level 3 — Failure Matching (stage_scoped stage/log matching; pipeline_scoped pass-through)
+
+Also provides weight computation and version selection for handlers that share a fault_id.
+"""
+
+import fnmatch
+import json
+
+from fl_control_plane.database import FaultHandler
+from fl_control_plane.pipeline_url_validators import VALIDATORS
+
+from .models import FailedStageContext, PipelineContext, StageMatchCriteria
+
+
+def detect_ci_system(pipeline_url: str) -> str | None:
+    """Return the CI system name for this URL, or None if unrecognised.
+
+    Iterates VALIDATORS in declaration order; the first matching validator's
+    name is returned.
+    """
+    for validator in VALIDATORS:
+        if validator.matches(pipeline_url):
+            return validator.name
+    return None
+
+
+def _parse_string_list(stored: str | None) -> list[str]:
+    """Parse a VARCHAR list field stored as JSON array or comma-separated string."""
+    if not stored:
+        return []
+    try:
+        parsed = json.loads(stored)
+        if isinstance(parsed, list):
+            return [str(item) for item in parsed]
+        if isinstance(parsed, str):
+            return [parsed]
+        return [stored]
+    except (json.JSONDecodeError, ValueError):
+        return [s.strip() for s in stored.split(",") if s.strip()]
+
+
+def filter_level2(handler: FaultHandler, ctx: PipelineContext) -> str | None:
+    """Return a rejection reason string if handler fails the infrastructure filter.
+
+    Returns None if the handler passes Level 2.
+
+    Checks in order:
+      1. ci_systems: non-empty list and ctx.ci_system not in it
+      2. repo_patterns: non-empty list and no glob matches ctx.github_repo_url
+      3. merge_target_branch: set and ctx.pr_target_branch does not match
+      4. trigger_scope: pr_only/non_pr_only when ctx.is_pr is known
+    """
+    ci_systems = _parse_string_list(handler.ci_systems)
+    if ci_systems and ctx.ci_system not in ci_systems:
+        return f"ci_system '{ctx.ci_system}' not in handler ci_systems {ci_systems}"
+
+    repo_patterns = _parse_string_list(handler.repo_patterns)
+    if repo_patterns and ctx.github_repo_url is not None:
+        bare_url = ctx.github_repo_url.removeprefix("https://").removeprefix("http://")
+        if not any(fnmatch.fnmatch(bare_url, pat) for pat in repo_patterns):
+            return (
+                f"repo '{ctx.github_repo_url}' matches none of repo_patterns {repo_patterns}"
+            )
+
+    if handler.merge_target_branch and ctx.pr_target_branch is not None:
+        if ctx.pr_target_branch != handler.merge_target_branch:
+            return (
+                f"pr_target_branch '{ctx.pr_target_branch}' != "
+                f"merge_target_branch '{handler.merge_target_branch}'"
+            )
+
+    if ctx.is_pr is not None:
+        if handler.trigger_scope == "pr_only" and not ctx.is_pr:
+            return "trigger_scope is pr_only but pipeline is not a PR"
+        if handler.trigger_scope == "non_pr_only" and ctx.is_pr:
+            return "trigger_scope is non_pr_only but pipeline is a PR"
+
+    return None
+
+
+def match_stages(
+    criteria: StageMatchCriteria | None, stages: list[FailedStageContext]
+) -> list[str]:
+    """Return IDs of stages matching all supplied criteria.
+
+    A stage matches when:
+      - stage_name pattern is absent OR fnmatch matches the stage's name
+      - log_contains is absent OR the substring appears in the stage's error_message
+
+    Returns an empty list when no stages match.
+    """
+    matched: list[str] = []
+    for stage in stages:
+        if criteria is not None:
+            if criteria.stage_name is not None:
+                if not fnmatch.fnmatch(stage.stage_name, criteria.stage_name):
+                    continue
+            if criteria.log_contains is not None:
+                message = stage.error_message or ""
+                if criteria.log_contains not in message:
+                    continue
+            # TODO: evaluate 'relevance' score from architecture once defined
+        matched.append(stage.stage_id)
+    return matched
+
+
+def compute_weight(handler: FaultHandler) -> float:
+    """Return the applicability weight used for version selection.
+
+    More specific handlers score higher. The handler with the highest weight
+    wins when multiple CRs share the same fault_id. Each distinct applicability
+    constraint contributes 1 point.
+    """
+    weight = 0
+    if _parse_string_list(handler.repo_patterns):
+        weight += 1
+    if _parse_string_list(handler.ci_systems):
+        weight += 1
+    if handler.merge_target_branch:
+        weight += 1
+    if handler.trigger_scope != "all":
+        weight += 1
+    if handler.scope == "stage_scoped":
+        weight += 0.5
+    return weight
+
+
+def select_versions(
+    candidates: list[tuple[FaultHandler, list[str]]]
+) -> tuple[list[tuple[FaultHandler, list[str]]], list[tuple[FaultHandler, str]]]:
+    """Select one handler per fault_id from the Level 2+3 candidates.
+
+    Args:
+        candidates: List of (handler, matched_stage_ids) tuples that passed
+            both Level 2 and Level 3.
+
+    Returns:
+        A tuple of:
+          - selected: handlers that will be executed, one per fault_id
+          - discarded: (handler, reason) pairs for selected_out decisions
+    """
+    by_fault_id: dict[str, list[tuple[FaultHandler, list[str]]]] = {}
+    for handler, stage_ids in candidates:
+        by_fault_id.setdefault(handler.fault_id, []).append((handler, stage_ids))
+
+    selected: list[tuple[FaultHandler, list[str]]] = []
+    discarded: list[tuple[FaultHandler, str]] = []
+
+    for fault_id, candidates in by_fault_id.items():
+        if len(candidates) == 1:
+            selected.append(candidates[0])
+            continue
+
+        # Sort by weight descending, then by registered_at ascending as tiebreaker
+        candidates_sorted = sorted(
+            candidates,
+            key=lambda pair: (-compute_weight(pair[0]), pair[0].registered_at),
+        )
+        winner, winner_stage_ids = candidates_sorted[0]
+        selected.append((winner, winner_stage_ids))
+        winner_weight = compute_weight(winner)
+
+        for loser, _ in candidates_sorted[1:]:
+            reason = (
+                f"fault_id '{fault_id}' has higher-weight handler "
+                f"'{winner.cr_name}' (weight={winner_weight} vs {compute_weight(loser)})"
+            )
+            discarded.append((loser, reason))
+
+    return selected, discarded

+"""Handler Orchestrator — main entry point.
+
+Evaluates all active FaultHandlers against the inspection's pipeline context
+and returns an OrchestrationResult for the Execution Engine. Read-only: does
+not write any database rows.
+"""
+
+import json
+import logging
+
+from sqlalchemy import select
+from sqlalchemy.ext.asyncio import AsyncSession
+
+from fl_control_plane.database import FailedStage, FaultHandler, Inspection
+from fl_control_plane.execution_contracts import (
+    FaultHandlerCR,
+    FaultHandlerExecution,
+    FaultHandlerMetadata,
+    FaultHandlerSpec,
+)
+
+from .applicability import (
+    detect_ci_system,
+    filter_level2,
+    match_stages,
+    select_versions,
+)
+from .models import (
+    ApplicabilityDecision,
+    FailedStageContext,
+    OrchestrationResult,
+    ParsedFailureMatch,
+    ParsedHandlerSpec,
+    PipelineContext,
+)
+
+_Candidate = tuple[FaultHandler, ParsedHandlerSpec, list[str]]
+
+logger = logging.getLogger(__name__)
+
+
+async def _load_pipeline_context(inspection_id: str, db: AsyncSession) -> PipelineContext:
+    """Load Inspection and FailedStage rows and assemble a PipelineContext."""
+    inspection_result = await db.execute(
+        select(Inspection).where(Inspection.id == inspection_id)
+    )
+    inspection = inspection_result.scalar_one_or_none()
+    if inspection is None:
+        raise KeyError(f"Inspection '{inspection_id}' not found")
+
+    stages_result = await db.execute(
+        select(FailedStage).where(FailedStage.inspection_id == inspection_id)
+    )
+    failed_stages = [
+        FailedStageContext(
+            stage_id=stage.id,
+            stage_name=stage.stage_name,
+            error_message=stage.error_message,
+        )
+        for stage in stages_result.scalars().all()
+    ]
+
+    return PipelineContext(
+        inspection_id=inspection_id,
+        pipeline_url=inspection.pipeline_url,
+        ci_system=detect_ci_system(inspection.pipeline_url),
+        github_repo_url=inspection.github_repo_url,
+        commit_id=inspection.commit_id,
+        is_pr=None,
+        pr_target_branch=None,
+        failed_stages=failed_stages,
+    )
+
+
+async def _load_active_handlers(db: AsyncSession) -> list[FaultHandler]:
+    """Load all active FaultHandler rows from the DB."""
+    result = await db.execute(
+        select(FaultHandler).where(FaultHandler.is_active.is_(True))
+    )
+    return list(result.scalars().all())
+
+
+def _parse_spec(handler: FaultHandler) -> ParsedHandlerSpec | None:
+    """Parse spec_json from a FaultHandler row. Returns None if missing or malformed."""
+    if not handler.spec_json:
+        logger.warning(
+            "handler_spec_missing",
+            extra={"handler": handler.cr_name, "fault_id": handler.fault_id},
+        )
+        return None
+    try:
+        raw = json.loads(handler.spec_json)
+        spec = ParsedHandlerSpec.model_validate(raw)
+        # DB scope column is authoritative; keep failure_match strategy in sync
+        if spec.failure_match.strategy != handler.scope:
+            spec = spec.model_copy(
+                update={
+                    "failure_match": ParsedFailureMatch(
+                        strategy=handler.scope,  # type: ignore[arg-type]
+                        match=spec.failure_match.match,
+                    )
+                }
+            )
+        return spec
+    except (json.JSONDecodeError, ValueError) as exc:
+        logger.warning(
+            "handler_spec_parse_error",
+            extra={"handler": handler.cr_name, "fault_id": handler.fault_id, "error": str(exc)},
+        )
+        return None
+
+
+def _evaluate_handler(
+    handler: FaultHandler,
+    spec: ParsedHandlerSpec,
+    ctx: PipelineContext,
+) -> tuple[list[str], ApplicabilityDecision | None]:
+    """Apply Level 2 and Level 3 filters to a single handler.
+
+    Returns:
+        (matched_stage_ids, rejection_decision) — if rejected, matched_stage_ids
+        is empty and rejection_decision is set. If accepted, rejection_decision is None.
+    """
+    level2_reason = filter_level2(handler, ctx)
+    if level2_reason is not None:
+        return [], ApplicabilityDecision(
+            handler_db_id=handler.id,
+            fault_id=handler.fault_id,
+            handler_name=handler.cr_name,
+            result="filtered_level2",
+            reason=level2_reason,
+        )
+
+    if spec.failure_match.strategy == "pipeline_scoped":
+        return [], None
+
+    matched_stage_ids = match_stages(spec.failure_match.match, ctx.failed_stages)
+    if not matched_stage_ids:
+        return [], ApplicabilityDecision(
+            handler_db_id=handler.id,
+            fault_id=handler.fault_id,
+            handler_name=handler.cr_name,
+            result="filtered_level3",
+            reason="no failed stages matched handler's match criteria",
+        )
+
+    return matched_stage_ids, None
+
+
+def _build_handler_cr(handler: FaultHandler, spec: ParsedHandlerSpec) -> FaultHandlerCR:
+    """Build a FaultHandlerCR for dispatch_all from a DB row and its parsed spec."""
+    return FaultHandlerCR(
+        metadata=FaultHandlerMetadata(name=handler.cr_name),
+        spec=FaultHandlerSpec(
+            fault_id=handler.fault_id,
+            execution=FaultHandlerExecution(
+                type=handler.execution_type,
+                task=spec.execution.task or "",
+                skill_paths=spec.execution.skills,
+                mcp_config_paths=spec.execution.mcps,
+                image=spec.execution.image,
+                entrypoint=spec.execution.entrypoint,
+            ),
+        ),
+    )
+
+
+async def evaluate(inspection_id: str, db: AsyncSession) -> OrchestrationResult:
+    """Evaluate all active FaultHandlers and return an OrchestrationResult.
+
+    Read-only — does not write any database rows.
+
+    Usage:
+        result = await evaluate(inspection_id, db)
+        dispatch_all(inspection_id, result.handlers)
+
+    Args:
+        inspection_id: UUID of the inspection to evaluate.
+        db: Active async database session.
+
+    Raises:
+        KeyError: No inspection row found for inspection_id.
+    """
+    ctx = await _load_pipeline_context(inspection_id, db)
+    handlers = await _load_active_handlers(db)
+
+    decisions: list[ApplicabilityDecision] = []
+    candidates: list[_Candidate] = []
+
+    for handler in handlers:
+        spec = _parse_spec(handler)
+        if spec is None:
+            decisions.append(ApplicabilityDecision(
+                handler_db_id=handler.id,
+                fault_id=handler.fault_id,
+                handler_name=handler.cr_name,
+                result="filtered_level2",
+                reason="spec_json missing or unparseable",
+            ))
+            continue
+
+        matched_stage_ids, rejection = _evaluate_handler(handler, spec, ctx)
+        if rejection is not None:
+            decisions.append(rejection)
+            continue
+
+        candidates.append((handler, spec, matched_stage_ids))
+
+    selected_pairs, discarded = select_versions([(h, ids) for h, _, ids in candidates])
+
+    for loser, reason in discarded:
+        decisions.append(ApplicabilityDecision(
+            handler_db_id=loser.id,
+            fault_id=loser.fault_id,
+            handler_name=loser.cr_name,
+            result="selected_out",
+            reason=reason,
+        ))
+
+    spec_by_handler_id = {h.id: spec for h, spec, _ in candidates}
+
+    selected_handlers: list[FaultHandlerCR] = []
+    for handler, matched_stage_ids in selected_pairs:
+        spec = spec_by_handler_id[handler.id]
+        selected_handlers.append(_build_handler_cr(handler, spec))
+        decisions.append(ApplicabilityDecision(
+            handler_db_id=handler.id,
+            fault_id=handler.fault_id,
+            handler_name=handler.cr_name,
+            result="matched",
+            reason="passed all applicability levels",
+            matched_stage_ids=matched_stage_ids,
+        ))
+        logger.info(
+            "handler_matched",
+            extra={"inspection_id": inspection_id, "fault_id": handler.fault_id, "handler": handler.cr_name},
+        )
+
+    logger.info(
+        "orchestration_complete",
+        extra={"inspection_id": inspection_id, "handlers_evaluated": len(handlers), "handlers_matched": len(selected_handlers)},
+    )
+
+    return OrchestrationResult(handlers=selected_handlers, decisions=decisions)
