+## HANA Instance Mapping
+
+Before deploying the application to a new cluster, the HANA Cloud instance must be mapped to the cluster's environment context. Without this, the BTP service broker rejects `ServiceInstance` creation with:
+
+> `Can not create service instance '...': There is no database with database_id '...' in namespace '...' within cluster '...'`
+
+### Steps
+
+1. Find the Kyma cluster ID:
+
+   ```sh
+   kubectl get configmap sap-btp-operator-config -n kyma-system -o jsonpath='{.data.CLUSTER_ID}'
+   ```
+
+2. In the BTP cockpit, navigate to **SAP HANA Cloud** → select the HANA instance → **Instance Mapping** → **Add Mapping**.
+
+3. Set **Environment Type** to `Kubernetes` and paste the cluster ID from step 1.
+
+4. Copy the **Instance ID** (the `database_id`) from the HANA instance overview and update `chart/values.yaml`:
+
+   ```yaml
+   database:
+     hanaInstanceId: <instance-id>
+   ```
+
+5. If the `ServiceInstance` already exists in a `CreateFailed` state, nudge it to reconcile:
+
+   ```sh
+   kubectl label serviceinstance pipeline-fl-control-plane-db-schema \
+     -n <namespace> reconcile=1 --overwrite
+   ```
+
+   Then nudge the binding once the instance reaches `Created`:
+
+   ```sh
+   kubectl label servicebinding pipeline-fl-control-plane-db-binding \
+     -n <namespace> reconcile=1 --overwrite
+   ```

+"""Background heartbeating helper for Temporal activities."""
+
+import asyncio
+from contextlib import asynccontextmanager
+from typing import AsyncIterator
+
+from temporalio import activity
+
+_HEARTBEAT_INTERVAL_SECONDS = 10
+
+
+@asynccontextmanager
+async def run_with_heartbeat(
+    interval: float = _HEARTBEAT_INTERVAL_SECONDS,
+) -> AsyncIterator[None]:
+    """Async context manager that heartbeats every *interval* seconds.
+
+    Spawns a background task that calls ``activity.heartbeat()`` on the
+    configured interval. The task is always cancelled when the context
+    exits, whether by normal return or exception.
+
+    Args:
+        interval: Seconds between heartbeats. Should be less than half
+            the ``heartbeat_timeout`` configured on the workflow side.
+            Default matches the project-standard 10 s (heartbeat_timeout=30 s).
+    """
+
+    async def _heartbeat_loop() -> None:
+        while True:
+            await asyncio.sleep(interval)
+            activity.heartbeat()
+
+    heartbeat_task = asyncio.create_task(_heartbeat_loop())
+    try:
+        yield
+    finally:
+        heartbeat_task.cancel()
+        try:
+            await heartbeat_task
+        except asyncio.CancelledError:
+            pass

+def _literal_module_value(module: ast.Module, variable_name: str) -> object:
+    for statement in module.body:
+        if isinstance(statement, ast.AnnAssign) and isinstance(statement.target, ast.Name):
+            if statement.target.id == variable_name:
+                if statement.value is None:
+                    raise AssertionError(f"{variable_name} has no value")
+                return ast.literal_eval(statement.value)
+        if isinstance(statement, ast.Assign):
+            for target in statement.targets:
+                if isinstance(target, ast.Name) and target.id == variable_name:
+                    return ast.literal_eval(statement.value)
+    raise AssertionError(f"Missing {variable_name}")
+

+def _normalize_down_revision(value: object) -> tuple[str, ...]:
+    if value is None:
+        return ()
+    if isinstance(value, str):
+        return (value,)
+    if isinstance(value, (list, tuple)) and all(isinstance(item, str) for item in value):
+        return tuple(value)
+    raise AssertionError(f"Unsupported down_revision value: {value!r}")
+

+def _migration_metadata() -> list[tuple[str, tuple[str, ...], Path]]:
+    metadata: list[tuple[str, tuple[str, ...], Path]] = []
+    for path in sorted(VERSIONS_DIR.glob("*.py")):
+        module = ast.parse(path.read_text(), filename=str(path))
+        revision = _literal_module_value(module, "revision")
+        if not isinstance(revision, str):
+            raise AssertionError(f"{path.relative_to(PROJECT_ROOT)} has non-string revision {revision!r}")
+        down_revision = _normalize_down_revision(_literal_module_value(module, "down_revision"))
+        metadata.append((revision, down_revision, path))
+    return metadata

+def _format_revision_paths(revision_to_paths: dict[str, list[Path]]) -> str:
+    parts = []
+    for revision, paths in sorted(revision_to_paths.items()):
+        relative_paths = ", ".join(str(path.relative_to(PROJECT_ROOT)) for path in sorted(paths))
+        parts.append(f"{revision}: {relative_paths}")
+    return "; ".join(parts)

+def test_migrations_have_unique_revision_ids() -> None:
+    """Every Alembic revision id is globally unique."""
+    revision_to_paths: dict[str, list[Path]] = {}
+    for revision, _down_revision, path in _migration_metadata():
+        revision_to_paths.setdefault(revision, []).append(path)
+
+    duplicates = {
+        revision: paths
+        for revision, paths in revision_to_paths.items()
+        if len(paths) > 1
+    }
+    assert not duplicates, f"Duplicate Alembic revision ids: {_format_revision_paths(duplicates)}"
+
+
+def test_migrations_have_single_head() -> None:
+    """Alembic has one upgrade head unless a merge revision is added."""
+    metadata = _migration_metadata()
+    revisions = {revision for revision, _down_revision, _path in metadata}
+    referenced_revisions = {
+        parent_revision
+        for _revision, down_revision, _path in metadata
+        for parent_revision in down_revision
+    }
+
+    missing_revisions = sorted(referenced_revisions - revisions)
+    assert not missing_revisions, f"Unknown Alembic down_revision ids: {missing_revisions}"
+
+    heads = sorted(revisions - referenced_revisions)
+    assert len(heads) == 1, (
+        f"Expected one Alembic head, got {heads}. "
+        "If parallel branches are intentional, add an Alembic merge revision."
