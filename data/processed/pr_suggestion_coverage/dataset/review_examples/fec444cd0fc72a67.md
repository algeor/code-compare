+#!/usr/bin/env bash
+# Port-forwards the Temporal UI to localhost:8080 and opens the browser.
+
+set -euo pipefail
+
+SCRIPT_DIR="$(cd "$(dirname "${BASH_SOURCE[0]}")" && pwd)"
+REPO_DIR="$(cd "${SCRIPT_DIR}/.." && pwd)"
+KUBECONFIG_FILE="${REPO_DIR}/kube-config-dev.yaml"
+
+NAMESPACE="test-depl-name"
+RELEASE="pipeline-fl-control-plane"
+LOCAL_PORT="8080"
+SERVICE_PORT="8080"  # Temporal Web container port
+
+while [[ $# -gt 0 ]]; do
+  case "$1" in
+    --namespace|-n)    NAMESPACE="$2"; shift 2 ;;
+    --port|-p)         LOCAL_PORT="$2"; shift 2 ;;
+    --service-port|-s) SERVICE_PORT="$2"; shift 2 ;;
+    --kubeconfig|-k)   KUBECONFIG_FILE="$2"; shift 2 ;;
+    --help|-h)
+      echo "Usage: $(basename "$0") [OPTIONS]"
+      echo ""
+      echo "Port-forwards the Temporal UI to localhost and opens the browser."
+      echo ""
+      echo "Options:"
+      echo "  -n, --namespace     <ns>    Kubernetes namespace (default: ${NAMESPACE})"
+      echo "  -p, --port          <port>  Local port to forward to (default: ${LOCAL_PORT})"
+      echo "  -s, --service-port  <port>  Temporal Web container port (default: ${SERVICE_PORT})"
+      echo "  -k, --kubeconfig    <path>  Path to kubeconfig file (default: <repo>/kube-config-dev.yaml)"
+      echo "  -h, --help                  Show this help message"
+      exit 0
+      ;;
+    *) echo "ERROR: Unknown argument: $1" >&2; exit 1 ;;
+  esac
+done
+
+if [[ ! -f "$KUBECONFIG_FILE" ]]; then
+  echo "ERROR: kubeconfig not found at ${KUBECONFIG_FILE}" >&2
+  echo "       Run 'gardenlogin get-client-certificate' to refresh it." >&2
+  exit 1
+fi
+
+SERVICE="${RELEASE}-temporal-web"
+
+echo "==> Port-forwarding Temporal UI: svc/${SERVICE} → http://localhost:${LOCAL_PORT}"
+echo "    Namespace : ${NAMESPACE}"
+echo "    Press Ctrl-C to stop."
+echo ""
+
+# Kill any stale process on the local port before forwarding.
+if command -v lsof &>/dev/null; then
+  lsof -ti:"${LOCAL_PORT}" | xargs kill -9 2>/dev/null || true
+fi
+
+_BROWSER_PID=""
+_cleanup() {
+  [[ -n "$_BROWSER_PID" ]] && kill "$_BROWSER_PID" 2>/dev/null || true
+}
+trap _cleanup EXIT
+
+# Open the browser once the port-forward is ready (background).
+(sleep 2 && { command -v open &>/dev/null && open "http://localhost:${LOCAL_PORT}" || command -v xdg-open &>/dev/null && xdg-open "http://localhost:${LOCAL_PORT}" || true; }) &
+_BROWSER_PID=$!
+
+kubectl --kubeconfig="${KUBECONFIG_FILE}" \
+  port-forward \
+  --address 127.0.0.1 \
+  -n "${NAMESPACE}" \
+  "svc/${SERVICE}" \
+  "${LOCAL_PORT}:${SERVICE_PORT}"
