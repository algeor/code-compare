+# Scripts
+
+## Old/New FL Comparison
+
+Use this flow when you want old Photon Lithium Jenkins results to review, then compare them with the new FL inspection result.
+
+Comparison scripts live under:
+
+```text
+scripts/fl_comparison/
+```
+
+### 1. Export the review dataset
+
+This step queries HANA once.
+
+It exports only Jenkins URLs that have:
+
+- `MISSION`
+- `FAILED_STAGE`
+- `FIX_PROPOSAL`
+
+Run:
+
+```bash
+uv run python scripts/fl_comparison/export_review_dataset.py
+```
+
+Default outputs:
+
+- `.logs/old-new-comparison/review_targets.csv`
+- `.logs/old-new-comparison/old_results_export.csv`
+
+Optional smaller export:
+
+```bash
+uv run python scripts/fl_comparison/export_review_dataset.py --limit 10
+```
+
+Optional single Jenkins export:
+
+```bash
+uv run python scripts/fl_comparison/export_review_dataset.py \
+  --jenkins-url 'https://example.jenkins/job/project/job/PR-123/1/'
+```
+
+Required HANA environment variables, unless passed as flags:
+
+- `OLD_FL_HANA_HOST`
+- `OLD_FL_HANA_USER`
+- `OLD_FL_HANA_PASSWORD`
+- `OLD_FL_HANA_PORT`, optional, defaults to `443`
+- `OLD_FL_HANA_SCHEMA`, optional, defaults to `PHOTON_LITHIUM`
+
+### 2. Review the targets
+
+Open:
+
+```text
+.logs/old-new-comparison/review_targets.csv
+```
+
+Keep or remove rows depending on what you want to compare.
+
+Important columns:
+
+- `JENKINS_URL`: normalized URL used for the new inspection
+- `MISSION_COUNT`: old missions found for that URL
+- `FAILED_STAGE_COUNT`: old failed stages found for that URL
+- `FIX_PROPOSAL_COUNT`: old fix proposals found for that URL
+- `LATEST_FIX_PROPOSAL_UPDATED_AT`: newest old fix proposal timestamp
+
+Do not edit `.logs/old-new-comparison/old_results_export.csv` unless you intentionally want to change the old comparison input.
+
+### 3. Run comparisons
+
+This step does not query HANA.
+
+It reads:
+
+- `.logs/old-new-comparison/review_targets.csv`
+- `.logs/old-new-comparison/old_results_export.csv`
+
+Run:
+
+```bash
+uv run python scripts/fl_comparison/run_comparison_batch.py \
+  --resume \
+  --no-temporal-port-forward
+```
+
+Run inspections in parallel:
+
+```bash
+uv run python scripts/fl_comparison/run_comparison_batch.py \
+  --parallel-inspections 3 \
+  --resume \
+  --no-api-port-forward \
+  --no-temporal-port-forward
+```
+
+Parallel mode still writes separate files per Jenkins output folder. Use `--no-api-port-forward` and `--no-temporal-port-forward` for parallel live inspections, because multiple workers cannot safely bind the same local forwarded ports.
+
+The runner prints:
+
+- input files it is using
+- selected Jenkins count
+- per-Jenkins output folder
+- old rows found in `old_results_export.csv`
+- whether it is resuming or starting a new inspection
+- stdout/stderr files for the live inspection
+- periodic `inspection: still running` progress lines
+- comparison output paths
+
+Adjust progress frequency:
+
+```bash
+uv run python scripts/fl_comparison/run_comparison_batch.py \
+  --progress-interval-seconds 10 \
+  --resume \
+  --no-temporal-port-forward
+```
+
+Disable periodic progress messages:
+
+```bash
+uv run python scripts/fl_comparison/run_comparison_batch.py \
+  --progress-interval-seconds 0 \
+  --resume \
+  --no-temporal-port-forward
+```
+
+Optional dry validation without running inspections:
+
+```bash
+uv run python scripts/fl_comparison/run_comparison_batch.py --skip-inspection
+```
+
+Optional limited run:
+
+```bash
+uv run python scripts/fl_comparison/run_comparison_batch.py \
+  --limit 3 \
+  --resume \
+  --no-temporal-port-forward
+```
+
+Optional single Jenkins run:
+
+```bash
+uv run python scripts/fl_comparison/run_comparison_batch.py \
+  --jenkins-url 'https://example.jenkins/job/project/job/PR-123/1/' \
+  --resume \
+  --no-temporal-port-forward
+```
+
+Per Jenkins output folder:
+
+```text
+.logs/old-new-comparison/<hash>-<jenkins-readable-name>/
+```
+
+Useful files there:
+
+- `run-inspection.stdout.txt`
+- `run-inspection.stderr.txt`
+- `new-log-folder.txt`
+- `comparison.md`
+- `comparison.json`

+#!/usr/bin/env python3
+"""Build an LLM-ready comparison bundle from old FL results and new inspection logs."""
+
+from __future__ import annotations
+
+import argparse
+import csv
+import json
+import re
+import sys
+from pathlib import Path
+from typing import Any, Literal
+
+
+OutputFormat = Literal["json", "markdown"]
+_INSPECTION_ID_PATTERN = re.compile(r"\binspection(?:_id)?[:=]\s*'?([0-9a-f]{8}-[0-9a-f-]{27,})'?")
+_GATE_STOP_PATTERN = re.compile(r"\bgate\s+([A-Za-z0-9_-]+).*\bSTOP\b")
+_STOPPED_BY_PATTERN = re.compile(r"\bstopped_by=([A-Za-z0-9_-]+)")
+_FINAL_STATUS_PATTERN = re.compile(r"\bstatus=([A-Z_]+)")
+
+
+def normalize_jenkins_url(value: str) -> str:
+    """Normalize Jenkins URLs enough to correlate old missions and new inspections."""
+    normalized = value.strip()
+    normalized = normalized.removesuffix("/display/redirect")
+    normalized = normalized.replace("/view/change-requests/", "/")
+    if normalized and not normalized.endswith("/"):
+        normalized = f"{normalized}/"
+    return normalized
+
+
+def _cell(row: list[str], index: int) -> str:
+    return row[index].strip() if index < len(row) else ""
+
+
+def _is_header(row: list[str]) -> bool:
+    return bool(row) and row[0].strip().upper() in {"JENKINS_URL", "OLD_JENKINS_URL"}
+
+
+def _header_index(header: list[str], *names: str) -> int | None:
+    normalized = {name.strip().upper(): index for index, name in enumerate(header)}
+    for name in names:
+        index = normalized.get(name)
+        if index is not None:
+            return index
+    return None
+
+
+def _row_from_header(row: list[str], header: list[str]) -> dict[str, str]:
+    def _col(*names: str) -> str:
+        idx = _header_index(header, *names)
+        return _cell(row, idx) if idx is not None else ""
+
+    return {
+        "jenkins_url": _col("JENKINS_URL", "OLD_JENKINS_URL"),
+        "mission_id": _col("MISSION_ID"),
+        "failed_stage_id": _col("FAILED_STAGE_ID"),
+        "stage_name": _col("STAGE_NAME"),
+        "fix_proposal_id": _col("FIX_PROPOSAL_ID"),
+        "problem": _col("CRITICAL_MESSAGE", "PROBLEM"),
+        "fault_description": _col("FAULT_DESCRIPTION"),
+        "fix_proposal_description": _col("FIX_PROPOSAL_DESCRIPTION"),
+    }
+
+
+def _row_from_positions(row: list[str]) -> dict[str, str]:
+    if len(row) >= 10:
+        return {
+            "jenkins_url": _cell(row, 0),
+            "mission_id": _cell(row, 1),
+            "failed_stage_id": _cell(row, 4),
+            "stage_name": _cell(row, 5),
+            "fix_proposal_id": _cell(row, 6),
+            "problem": _cell(row, 7),
+            "fault_description": _cell(row, 8),
+            "fix_proposal_description": _cell(row, 9),
+        }
+    return {
+        "jenkins_url": _cell(row, 0),
+        "mission_id": _cell(row, 1),
+        "failed_stage_id": _cell(row, 2),
+        "stage_name": _cell(row, 3),
+        "fix_proposal_id": _cell(row, 4),
+        "problem": _cell(row, 5),
+        "fault_description": _cell(row, 6),
+        "fix_proposal_description": _cell(row, 7),
+    }
+
+
+def read_old_export(path: Path, requested_url: str | None) -> dict[str, Any]:
+    """Read old FL rows exported from DBeaver as tab-separated text."""
+    text = path.read_text(errors="replace")
+    first_line = text.splitlines()[0] if text.splitlines() else ""
+    delimiter = "\t" if "\t" in first_line else ","
+    rows = list(csv.reader(text.splitlines(), delimiter=delimiter))
+    if not rows:
+        raise ValueError(f"Old export is empty: {path}")
+
+    header = rows[0] if _is_header(rows[0]) else None
+    data_rows = rows[1:] if header else rows
+    requested_normalized_url = normalize_jenkins_url(requested_url) if requested_url else None
+
+    suggestions: list[dict[str, str]] = []
+    for raw_row in data_rows:
+        if not raw_row or not any(cell.strip() for cell in raw_row):
+            continue
+        parsed = _row_from_header(raw_row, header) if header else _row_from_positions(raw_row)
+        parsed["jenkins_url"] = normalize_jenkins_url(parsed["jenkins_url"])
+        if requested_normalized_url and parsed["jenkins_url"] != requested_normalized_url:
+            continue
+        suggestions.append(parsed)
+
+    if not suggestions:
+        if requested_normalized_url:
+            raise ValueError("No old suggestions matched the requested Jenkins URL")
+        raise ValueError(f"Old export contains no data rows: {path}")
+
+    urls = sorted({item["jenkins_url"] for item in suggestions})
+    mission_ids = sorted({item["mission_id"] for item in suggestions})
+    if len(urls) > 1 and requested_normalized_url is None:
+        raise ValueError("Old export contains multiple Jenkins URLs. Pass --jenkins-url to select one.")
+    if len(mission_ids) > 1 and requested_normalized_url is None:
+        raise ValueError("Old export contains multiple missions. Pass --jenkins-url to select one.")
+
+    return {
+        "jenkins_url": urls[0],
+        "mission_id": mission_ids[0],
+        "suggestion_count": len(suggestions),
+        "suggestions": suggestions,
+        "final_text": _format_old_final_text(urls[0], mission_ids[0], suggestions),
+    }
+
+
+def _format_old_final_text(jenkins_url: str, mission_id: str, suggestions: list[dict[str, str]]) -> str:
+    lines = [
+        "# Old Fault Localization Result",
+        "",
+        f"Jenkins URL: {jenkins_url}",
+        f"Mission ID: {mission_id}",
+        f"Suggestions: {len(suggestions)}",
+        "",
+    ]
+    for index, suggestion in enumerate(suggestions, start=1):
+        lines.extend(
+            [
+                f"## Suggestion {index}: {suggestion['problem']}",
+                "",
+                f"Stage: {suggestion['stage_name']}",
+                f"Failed stage ID: {suggestion['failed_stage_id']}",
+                f"Fix proposal ID: {suggestion['fix_proposal_id']}",
+                "",
+                "### Fault Description",
+                suggestion["fault_description"],
+                "",
+                "### Fix Proposal",
+                suggestion["fix_proposal_description"],
+                "",
+            ]
+        )
+    return "\n".join(lines).strip()
+
+
+def _resolve_live_log(path: Path) -> Path:
+    if path.is_dir():
+        path = path / "live.txt"
+    if not path.exists():
+        raise FileNotFoundError(f"New inspection log not found: {path}")
+    return path
+
+
+def _parse_json_suffix(line: str) -> dict[str, Any] | None:
+    start = line.find("{")
+    while start != -1:
+        try:
+            value = json.loads(line[start:])
+        except json.JSONDecodeError:
+            start = line.find("{", start + 1)
+            continue
+        return value if isinstance(value, dict) else None
+    return None
+
+
+def _clean_log_line(line: str) -> str:
+    return line.replace("\x00", "").strip()
+
+
+def read_new_completion(path: Path) -> dict[str, Any]:
+    """Read the new inspection outcome from a live log."""
+    live_log = _resolve_live_log(path)
+    completions: list[dict[str, Any]] = []
+    observed_jenkins_url = ""
+    inspection_id = ""
+    stopped_by = ""
+    final_status = ""
+    workflow_completed = False
+    gate_stop_evidence: list[str] = []
+
+    for raw_line in live_log.read_text(errors="replace").splitlines():
+        line = _clean_log_line(raw_line)
+        if "jenkins_url=" in line and not observed_jenkins_url:
+            marker = "jenkins_url='"
+            start = line.find(marker)
+            if start != -1:
+                start += len(marker)
+                end = line.find("'", start)
+                observed_jenkins_url = normalize_jenkins_url(line[start:end])
+        if "pipeline_url=" in line and not observed_jenkins_url:
+            marker = "pipeline_url='"
+            start = line.find(marker)
+            if start != -1:
+                start += len(marker)
+                end = line.find("'", start)
+                observed_jenkins_url = normalize_jenkins_url(line[start:end])
+
+        if not inspection_id:
+            inspection_match = _INSPECTION_ID_PATTERN.search(line)
+            if inspection_match:
+                inspection_id = inspection_match.group(1)
+
+        gate_match = _GATE_STOP_PATTERN.search(line)
+        if gate_match:
+            stopped_by = stopped_by or gate_match.group(1)
+            gate_stop_evidence.append(line)
+
+        stopped_by_match = _STOPPED_BY_PATTERN.search(line)
+        if stopped_by_match:
+            stopped_by = stopped_by_match.group(1)
+            gate_stop_evidence.append(line)
+
+        if "no failed stage source files found" in line:
+            gate_stop_evidence.append(line)
+
+        if "event=inspection_finalized" in line:
+            final_status_match = _FINAL_STATUS_PATTERN.search(line)
+            if final_status_match:
+                final_status = final_status_match.group(1)
+            gate_stop_evidence.append(line)
+
+        if "event=workflow_completed" in line:
+            workflow_completed = True
+            gate_stop_evidence.append(line)
+
+        payload = _parse_json_suffix(line)
+        if payload is None or payload.get("event") != "Completion result":
+            continue
+        body = payload.get("body")
+        result_json = body.get("result_json") if isinstance(body, dict) else None
+        if not isinstance(result_json, dict):
+            continue
+        completion_text = result_json.get("completion_text")
+        completions.append(
+            {
+                "inspection_id": payload.get("inspection_id"),
+                "outcome": "completed",
+                "completion_event_found": True,
+                "task_id": result_json.get("task_id"),
+                "completion_text": completion_text,
+                "tokens_in": result_json.get("tokens_in"),
+                "tokens_out": result_json.get("tokens_out"),
+                "mlflow_run_id": result_json.get("mlflow_run_id"),
+                "code_changes": body.get("code_changes") if isinstance(body, dict) else None,
+            }
+        )
+
+    if not completions:
+        if stopped_by:
+            return {
+                "inspection_id": inspection_id,
+                "outcome": "gate_stopped",
+                "completion_event_found": False,
+                "stopped_by": stopped_by,
+                "final_status": final_status,
+                "workflow_completed": workflow_completed,
+                "evidence": gate_stop_evidence,
+                "jenkins_url": observed_jenkins_url,
+                "final_text": _format_gate_stopped_final_text(
+                    stopped_by=stopped_by,
+                    final_status=final_status,
+                    evidence=gate_stop_evidence,
+                ),
+            }
+        raise ValueError(f"No Completion result event found in {live_log}")
+
+    latest = completions[-1]
+    latest["jenkins_url"] = observed_jenkins_url
+    latest["final_text"] = _format_new_final_text(latest.get("completion_text"))
+    return latest
+
+
+def _format_gate_stopped_final_text(stopped_by: str, final_status: str, evidence: list[str]) -> str:
+    lines = [
+        "# New Fault Localization Result",
+        "",
+        "The inspection stopped before handler or finalizer execution.",
+        f"Stopped by gate: {stopped_by}",
+    ]
+    if final_status:
+        lines.append(f"Final status: {final_status}")
+    lines.extend(
+        [
+            "Completion result event found: false",
+            "",
+            "## Evidence",
+        ]
+    )
+    lines.extend(f"- {line}" for line in evidence)
+    return "\n".join(lines).strip()
+
+
+def _format_new_final_text(value: object) -> str:
+    """Return finalizer completion text as a stable string for LLM comparison."""
+    if isinstance(value, str):
+        return value
+    if isinstance(value, (dict, list)):
+        return json.dumps(value, indent=2, ensure_ascii=False)
+    if value is None:
+        return ""
+    return str(value)
+
+
+def build_bundle(old_results: Path, new_log: Path, jenkins_url: str | None) -> dict[str, Any]:
+    """Build the old/new comparison payload."""
+    old = read_old_export(old_results, jenkins_url)
+    new = read_new_completion(new_log)
+    old_url = normalize_jenkins_url(old["jenkins_url"])
+    new_url = normalize_jenkins_url(new.get("jenkins_url") or jenkins_url or old_url)
+
+    return {
+        "correlation": {
+            "jenkins_url": old_url,
+            "new_log_jenkins_url": new_url,
+            "urls_match": old_url == new_url,
+            "old_mission_id": old["mission_id"],
+            "new_inspection_id": new["inspection_id"],
+        },
+        "old": old,
+        "new": new,
+        "llm_compare_prompt": _build_compare_prompt(old["final_text"], new["final_text"]),
+    }
+
+
+def _build_compare_prompt(old_text: str, new_text: str) -> str:
+    return "\n\n".join(
+        [
+            "Compare these two fault-localization outputs for the same Jenkins URL.",
+            "Focus on: root cause agreement, fix agreement, missing evidence, false positives, and actionability.",
+            "Return a concise comparison with a verdict: equivalent, partially equivalent, or different.",
+            "## Old output",
+            old_text,
+            "## New output",
+            new_text,
+        ]
+    )
+
+
+def _write_markdown(bundle: dict[str, Any]) -> None:
+    print("# FL Comparison Input")
+    print()
+    correlation = bundle["correlation"]
+    print(f"Jenkins URL: {correlation['jenkins_url']}")
+    print(f"URLs match: {correlation['urls_match']}")
+    print(f"Old mission ID: {correlation['old_mission_id']}")
+    print(f"New inspection ID: {correlation['new_inspection_id']}")
+    print()
+    print("## Prompt for LLM")
+    print()
+    print(bundle["llm_compare_prompt"])
+
+
+def main() -> int:
+    parser = argparse.ArgumentParser(description=__doc__)
+    parser.add_argument("--old-results", type=Path, help="CSV export from the old Photon Lithium query")
+    parser.add_argument("--old-export", dest="old_results", type=Path, help=argparse.SUPPRESS)
+    parser.add_argument("--new-log", required=True, type=Path, help="New .logs/<inspection-id> folder or live.txt")
+    parser.add_argument("--jenkins-url", help="Jenkins URL to select when the old results export has multiple URLs")
+    parser.add_argument("--format", choices=("json", "markdown"), default="json")
+    args = parser.parse_args()
+
+    if args.old_results is None:
+        parser.error("--old-results is required")
+
+    try:
+        bundle = build_bundle(args.old_results, args.new_log, args.jenkins_url)
+    except (FileNotFoundError, ValueError) as exc:
+        print(f"ERROR: {exc}", file=sys.stderr)
+        return 1
+
+    output_format: OutputFormat = args.format
+    if output_format == "json":
+        print(json.dumps(bundle, indent=2, ensure_ascii=False))
+    else:
+        _write_markdown(bundle)
+    return 0
+
+
+if __name__ == "__main__":
+    raise SystemExit(main())

+SELECT
+  TRIM(SUBSTRING(m.PROMPT, LOCATE(m.PROMPT, 'http'))) AS OLD_JENKINS_URL,
+  m.MISSION_ID,
+  fs.ID AS FAILED_STAGE_ID,
+  fs.STAGE_NAME,
+  fp.ID AS FIX_PROPOSAL_ID,
+  fp.CRITICAL_MESSAGE,

+def run_new_inspection(
+    repo_dir: Path,
+    jenkins_url: str,
+    output_dir: Path,
+    extra_args: list[str],
+    progress_interval_seconds: int,
+    log_prefix: str,
+) -> Path:
+    """Run live inspection and return the generated new-app log folder."""
+    run_file_stem = f"run-inspection-{_run_file_suffix()}"

+def _run_bundle_builder(
+    repo_dir: Path,
+    command: list[str],
+    output_file: Path,
+    error_file: Path,
+    log_prefix: str,
+) -> None:
+    with output_file.open("w") as stream:
+        completed = subprocess.run(
+            command,
+            cwd=repo_dir,
+            stdout=stream,
+            stderr=subprocess.PIPE,
+            text=True,
+            check=False,
+        )
+    if completed.returncode == 0:
+        return
+
+    error_file.write_text(
+        "\n".join(
+            [
+                f"Command: {shlex.join(command)}",
+                f"Exit code: {completed.returncode}",
+                "",
+                "stderr:",
+                completed.stderr.strip(),
+                "",
+            ]
+        )
+    )
+    _print_kv("comparison", "failed", prefix=log_prefix, exit_code=completed.returncode, error_file=error_file)
+    raise RuntimeError(f"Comparison bundle failed; see {error_file}")

+def _terminate_process(process: subprocess.Popen[str]) -> None:
+    if process.poll() is not None:
+        return
+    try:
+        os.killpg(process.pid, signal.SIGTERM)
+    except ProcessLookupError:
+        return
+    try:
+        process.wait(timeout=5)
+    except subprocess.TimeoutExpired:
+        try:
+            os.killpg(process.pid, signal.SIGKILL)
+        except ProcessLookupError:
+            pass
+        process.wait()

+def _run_comparisons(
+    work_items: list[ComparisonWorkItem],
+    repo_dir: Path,
+    old_results: Path,
+    resume: bool,
+    inspection_extra_args: list[str],
+    progress_interval_seconds: int,
+    parallel_inspections: int,
+    no_api_port_forward: bool,
+    no_temporal_port_forward: bool,
+) -> None:
+    if not work_items:
+        _print_kv("work", "no inspections to run")
+        return
+
+    _validate_parallel_settings(
+        work_items=work_items,
+        resume=resume,
+        parallel_inspections=parallel_inspections,
+        inspection_extra_args=inspection_extra_args,
+        no_api_port_forward=no_api_port_forward,
+        no_temporal_port_forward=no_temporal_port_forward,
+    )
+    worker_count = min(parallel_inspections, len(work_items))
+    _print_header("Running comparisons")
+    _print_kv("queued targets", len(work_items))
+    _print_kv("parallel inspections", worker_count)
+
+    failures: list[str] = []
+    if worker_count == 1:
+        for work_item in work_items:
+            try:
+                _run_comparison_work_item(
+                    work_item=work_item,
+                    repo_dir=repo_dir,
+                    old_results=old_results,
+                    resume=resume,
+                    inspection_extra_args=inspection_extra_args,
+                    progress_interval_seconds=progress_interval_seconds,
+                )
+            except (RuntimeError, OSError, subprocess.CalledProcessError) as exc:
+                failures.append(_format_failure(work_item, exc))
+                break
+    else:
+        with ThreadPoolExecutor(max_workers=worker_count) as executor:
+            futures = {
+                executor.submit(
+                    _run_comparison_work_item,
+                    work_item=work_item,
+                    repo_dir=repo_dir,
+                    old_results=old_results,
+                    resume=resume,
+                    inspection_extra_args=inspection_extra_args,
+                    progress_interval_seconds=progress_interval_seconds,
+                ): work_item
+                for work_item in work_items
+            }
+            for future in as_completed(futures):
+                work_item = futures[future]
+                try:
+                    future.result()
+                except (RuntimeError, OSError, subprocess.CalledProcessError) as exc:
+                    failures.append(_format_failure(work_item, exc))
+
+    if failures:
+        raise RuntimeError(f"{len(failures)} comparison(s) failed: " + "; ".join(failures))

+def _validate_parallel_settings(
+    work_items: list[ComparisonWorkItem],
+    resume: bool,
+    parallel_inspections: int,
+    inspection_extra_args: list[str],
+    no_api_port_forward: bool,
+    no_temporal_port_forward: bool,
+) -> None:
+    if parallel_inspections < 1:
+        raise ValueError("--parallel-inspections must be at least 1")
+    effective_parallel_inspections = min(parallel_inspections, len(work_items))
+    if effective_parallel_inspections == 1:

+    )
+    temporal_port_forward_disabled = no_temporal_port_forward or "--no-temporal-port-forward" in inspection_extra_args
+    if not api_port_forward_disabled:
+        raise ValueError(
+            "--parallel-inspections > 1 requires --no-api-port-forward or --inspection-extra-arg=--base-url"
+        )
+    if not temporal_port_forward_disabled:

+def _resolve_data_file(
+    repo_dir: Path,
+    output_root: Path,
+    explicit_path: Path | None,
+    default_name: str,
+    legacy_default_name: str,
+) -> Path:

+        return explicit_path

+def _select_targets(
+    all_targets: list[ComparisonTarget], requested_urls: list[str], limit: int | None
+) -> list[ComparisonTarget]:
+    requested = [normalize_jenkins_url(url) for url in requested_urls]
+    if requested:
+        available = {target.jenkins_url: target for target in all_targets}
+        missing = [url for url in requested if url not in available]
+        if missing:
+            raise ValueError("Requested URL(s) not found in review targets file: " + ", ".join(missing))
+        return [available[url] for url in requested]
+    return all_targets[:limit] if limit is not None else all_targets

+def _print_header(message: str) -> None:
+    with PRINT_LOCK:

+def _print_kv(label: str, value: object, prefix: str = "", **fields: object) -> None:
+    parts = [f"{label}: {_format_log_value(value)}"]
+    for key, field_value in fields.items():
+        parts.append(f"{key}={_format_log_value(field_value)}")
+    prefix_text = f"{prefix} " if prefix else ""
+    with PRINT_LOCK:
+        print("  " + prefix_text + " | ".join(parts), flush=True)
+
+
+def _format_log_value(value: object) -> str:
+    if isinstance(value, Path):
+        return str(value)
+    return str(value)

+def _format_duration(seconds: float) -> str:
+    total_seconds = int(seconds)
+    hours, remainder = divmod(total_seconds, 3600)
+    minutes, remaining_seconds = divmod(remainder, 60)
+    if hours:
+        return f"{hours}h{minutes:02d}m{remaining_seconds:02d}s"
+    if minutes:
+        return f"{minutes}m{remaining_seconds:02d}s"
+    return f"{remaining_seconds}s"

+#!/usr/bin/env python3
+"""Extract final completion text from run_inspection logs."""
+
+from __future__ import annotations
+
+import argparse
+import json
+import sys
+from pathlib import Path
+from typing import Any, Literal
+
+
+OutputFormat = Literal["json", "text"]
+
+
+def _resolve_log_file(path: Path) -> Path:
+    """Return the live log file for a path that may be a file or log directory."""
+    if path.is_dir():
+        path = path / "live.txt"
+    if not path.exists():
+        raise FileNotFoundError(f"Log file not found: {path}")
+    if not path.is_file():
+        raise ValueError(f"Log path is not a file: {path}")
+    return path
+
+
+def _parse_json_suffix(line: str) -> dict[str, Any] | None:
+    """Parse the JSON payload embedded after the formatted log prefix."""
+    start = line.find("{")
+    while start != -1:
+        candidate = line[start:]
+        try:
+            parsed = json.loads(candidate)
+        except json.JSONDecodeError:
+            start = line.find("{", start + 1)
+            continue
+        if isinstance(parsed, dict):
+            return parsed
+        return None
+    return None
+
+
+def extract_completions(log_path: Path) -> list[dict[str, Any]]:
+    """Extract completion result records from a live inspection log."""
+    resolved_log_path = _resolve_log_file(log_path)
+    completions: list[dict[str, Any]] = []
+
+    for line in resolved_log_path.read_text(errors="replace").splitlines():
+        payload = _parse_json_suffix(line)
+        if payload is None or payload.get("event") != "Completion result":
+            continue
+
+        body = payload.get("body")
+        if not isinstance(body, dict):
+            continue
+        result_json = body.get("result_json")
+        if not isinstance(result_json, dict):
+            continue
+
+        completions.append(
+            {
+                "inspection_id": payload.get("inspection_id"),
+                "task_id": result_json.get("task_id"),
+                "completion_text": result_json.get("completion_text"),
+                "tokens_in": result_json.get("tokens_in"),
+                "tokens_out": result_json.get("tokens_out"),
+                "mlflow_run_id": result_json.get("mlflow_run_id"),
+                "code_changes": body.get("code_changes"),
+            }
+        )
+
+    return completions
+
+
+def _write_json(completions: list[dict[str, Any]]) -> None:
+    print(json.dumps(completions, indent=2, ensure_ascii=False))
+
+
+def _write_text(completions: list[dict[str, Any]]) -> None:
+    for index, completion in enumerate(completions, start=1):
+        if len(completions) > 1:
+            print(f"# Completion {index}")
+        print(f"inspection_id: {completion.get('inspection_id')}")
+        print(f"task_id: {completion.get('task_id')}")
+        print(f"tokens_in: {completion.get('tokens_in')}")
+        print(f"tokens_out: {completion.get('tokens_out')}")
+        print(f"mlflow_run_id: {completion.get('mlflow_run_id')}")
+        print()
+        completion_text = completion.get("completion_text")
+        if completion_text:
+            print(completion_text)
+        if index != len(completions):
+            print()
+
+
+def main() -> int:
+    """Run the command line parser."""
+    parser = argparse.ArgumentParser(description=__doc__)
+    parser.add_argument("log_path", type=Path, help="Path to a .logs/<inspection-id> directory or live.txt file")
+    parser.add_argument(
+        "--format",
+        choices=("json", "text"),
+        default="json",
+        help="Output format. Default: json",
+    )
+    args = parser.parse_args()
+
+    try:
+        completions = extract_completions(args.log_path)
+    except (FileNotFoundError, ValueError) as exc:
+        print(f"ERROR: {exc}", file=sys.stderr)
+        return 2
+
+    if not completions:
+        print("ERROR: no Completion result event found", file=sys.stderr)
+        return 1
+
+    output_format: OutputFormat = args.format
+    if output_format == "json":
+        _write_json(completions)
+    else:
+        _write_text(completions)
+    return 0
+
+
+if __name__ == "__main__":
+    raise SystemExit(main())

+def current_timestamp():
+    return datetime.now(timezone.utc).strftime("%Y-%m-%d %H:%M:%S")

+def short_timestamp(value):
+    if not isinstance(value, str) or not value:
+        return current_timestamp()
+    match = ISO_TIMESTAMP_PATTERN.match(value)
+    if match:
+        return f"{match.group('date')} {match.group('time')}"
+    return value[:19].replace("T", " ")

+def markdown_line(line):
+    parts = []
+    position = 0
+    for match in URL_PATTERN.finditer(line):
+        matched_url = match.group(0)
+        url = matched_url.rstrip(".,;:]")
+        suffix = matched_url[len(url):]
+        parts.append(html.escape(line[position:match.start()], quote=False))
+        parts.append(
+            f'<a href="{html.escape(url, quote=True)}">'
+            f'{html.escape(url, quote=False)}</a>{html.escape(suffix, quote=False)}'
+        )
+        position = match.end()
+    parts.append(html.escape(line[position:], quote=False))
+    return "".join(parts)

+def write_markdown(target, line):
+    print(markdown_line(line) + "<br>", file=target, flush=True)

+def write_line(line, is_error=False):
+    print(line, flush=True)
+    print(line, file=log_file, flush=True)
+    write_markdown(markdown_file, line)
+    if is_error and line not in seen_error_lines:
+        seen_error_lines.add(line)
+        print(line, file=error_file, flush=True)
+        write_markdown(error_markdown_file, line)

+def clean_plain_message(line):
+    return line.removeprefix("==> ").strip()

+def is_scalar(value):
+    return value is None or isinstance(value, (str, int, float, bool))
+
+
+def format_value(value):
+    if value is None:
+        return "null"
+    if isinstance(value, bool):
+        return "true" if value else "false"
+    if isinstance(value, (int, float)):
+        return str(value)
+    text = str(value)
+    if not text:
+        return '""'
+    if re.search(r"\s|[\"'=:{}\[\],]", text):
+        if "'" in text:
+            escaped = text.replace("\\", "\\\\").replace('"', '\\"')
+            return f'"{escaped}"'
+        return f"'{text}'"
+    return text
+

+def ordered_items(value):
+    hidden = {"level", "timestamp"}
+    preferred = [
+        "inspection_id",
+        "event",
+        "mentioned_inspection_id",
+        "failed_stage_count",
+        "failed_stages",
+        "message",
+        "activity_type",
+        "activity_id",
+        "task_id",
+        "job",
+        "fault_id",
+        "handler",
+        "logger",
+    ]
+    normalized = dict(value)
+    if "inspection_id" not in normalized:
+        normalized["inspection_id"] = None
+
+    result = []
+    for key in preferred:
+        if key in normalized and key not in hidden:
+            result.append((key, normalized[key]))
+    for key, item in normalized.items():
+        if key not in hidden and key not in preferred:
+            result.append((key, item))
+    return result
+

+def format_payload(value):
+    items = ordered_items(value)
+    if all(is_scalar(item) for _, item in items):
+        return " ".join(f"{key}={format_value(item)}" for key, item in items)
+    return json.dumps({key: item for key, item in items}, ensure_ascii=False, separators=(",", ":"))

+def normalize_control_plane_payload(value):
+    event = value.get("event")
+    logger = value.get("logger")
+    if logger != "fl_control_plane.metadata_extractor.db_writer" or not isinstance(event, str):
+        return value
+
+    failed_stages_match = CP_DB_WRITER_FAILED_STAGES_PATTERN.match(event)
+    if not failed_stages_match:
+        return value
+
+    normalized = dict(value)
+    normalized["event"] = "db_writer_failed_stages_inserted"
+    normalized["mentioned_inspection_id"] = failed_stages_match.group("mentioned_inspection_id")
+    normalized["failed_stage_count"] = int(failed_stages_match.group("failed_stage_count"))
+    raw_failed_stages = failed_stages_match.group("failed_stages")
+    try:
+        failed_stages = ast.literal_eval(raw_failed_stages)
+    except (SyntaxError, ValueError):
+        normalized["failed_stages"] = raw_failed_stages
+    else:
+        if isinstance(failed_stages, list) and all(isinstance(item, str) for item in failed_stages):
+            normalized["failed_stages"] = " | ".join(failed_stages)
+        else:
+            normalized["failed_stages"] = raw_failed_stages
+    return normalized

+def parse_doit_plain_message(message):
+    remaining = message
+    fields = {}
+    field_pattern = re.compile(r"^(level|trace_id|category|origin)=(\S+)\s*")
+
+    while True:
+        match = field_pattern.match(remaining)
+        if not match:
+            break
+        key, value = match.groups()
+        if key not in fields:
+            fields[key] = value
+        remaining = remaining[match.end():]
+
+    level = fields.pop("level", "INFO").upper()
+    payload_items = []
+    for key in ("category", "origin", "trace_id"):
+        if key in fields:
+            payload_items.append(f"{key}={format_value(fields[key])}")
+    if remaining.strip():
+        payload_items.extend(format_doit_message_fields(remaining.strip()))
+    return level, " ".join(payload_items) if payload_items else f"message={format_value(message)}"
+

+def format_doit_message_fields(message):
+    task_finished = DOIT_TASK_FINISHED_PATTERN.match(message)
+    if not task_finished:
+        return [f"message={format_value(message)}"]
+
+    payload_items = [
+        f"atom_id={format_value(task_finished.group('atom_id'))}",
+        f"task_state={format_value(task_finished.group('task_state'))}",
+    ]
+    task_message = task_finished.group("message") or ""
+    error_key = DOIT_ERROR_KEY_PATTERN.search(task_message)
+    error_source = DOIT_ERROR_SOURCE_PATTERN.search(task_message)
+    if error_key:
+        payload_items.append(f"error_key={format_value(error_key.group('error_key'))}")
+    if error_source:
+        payload_items.append(f"error_source={format_value(error_source.group('error_source'))}")
+    if task_message and not (error_key or error_source):
+        payload_items.append(f"message={format_value(task_message)}")
+    return payload_items
+

+def normalized_level(value=None):
+    if isinstance(value, dict):
+        level = value.get("level")
+        if isinstance(level, str) and level:
+            normalized = level.upper()
+            return LEVEL_ALIASES.get(normalized, normalized)
+    return "INFO"

+def display_level(level, payload):
+    if level.upper() in ERROR_LEVELS:
+        return level
+    if ERROR_TEXT_PATTERN.search(payload):
+        return "ERROR"
+    return level

+
+def is_error_line(level):
+    return level.upper() in ERROR_LEVELS
+
+
+def format_header(level, source, timestamp):
+    source_width = max(6, len(source) + 2)
+    return f"[{level}]".ljust(7) + " " + f"[{source}]".ljust(source_width) + " " + timestamp

+def resolve_source(prefix):
+    if ":" in prefix:
+        base, context = prefix.split(":", 1)
+        base_source = SOURCE_PREFIXES.get(base, base)
+        return f"{base_source}:{context}", base
+    return SOURCE_PREFIXES.get(prefix, prefix), prefix
+
+
+def format_record(prefix, line, parsed=None):
+    source, base_prefix = resolve_source(prefix)
+    if isinstance(parsed, dict):
+        if source == "CP":
+            parsed = normalize_control_plane_payload(parsed)
+        source_level = normalized_level(parsed)
+        timestamp = short_timestamp(parsed.get("timestamp"))
+        payload = format_payload(parsed)
+        level = display_level(source_level, payload)
+        return f"{format_header(level, source, timestamp)} {payload}", is_error_line(level)
+
+    timestamp_match = PLAIN_TIMESTAMP_PATTERN.match(line)
+    if base_prefix in PLAIN_NO_CONTEXT_PREFIXES and not timestamp_match:
+        payload = clean_plain_message(line)
+        level = display_level("INFO", payload)
+        return f"{format_header(level, source, current_timestamp())} {payload}", is_error_line(level)
+
+    timestamp = short_timestamp(timestamp_match.group("timestamp")) if timestamp_match else current_timestamp()
+    message = timestamp_match.group("rest") if timestamp_match else line
+    if base_prefix == "DOIT" and timestamp_match:
+        source_level, payload = parse_doit_plain_message(message)
+        level = display_level(source_level, payload)
+    else:
+        payload = f"message={format_value(message)}"
+        level = display_level("INFO", payload)
+    return f"{format_header(level, source, timestamp)} {payload}", is_error_line(level)
+

+def normalize_json_strings(value):
+    if isinstance(value, dict):
+        return {key: normalize_json_strings(item) for key, item in value.items()}
+    if isinstance(value, list):
+        return [normalize_json_strings(item) for item in value]
+    if isinstance(value, str):
+        stripped = value.strip()
+        if stripped.startswith(("{", "[")):
+            try:
+                return normalize_json_strings(json.loads(stripped))
+            except json.JSONDecodeError:
+                return value
+    return value

+choose_available_port() {
+  local name="$1"
+  local port="$2"
+  local candidate="$port"
+
+  while port_open "$candidate"; do
+    echo "==> ${name}: 127.0.0.1:${candidate} is already in use; trying $((candidate + 1))." >&2
+    candidate=$((candidate + 1))
+    if [[ $((candidate - port)) -gt 50 ]]; then
+      echo "ERROR: could not find a free local port for ${name} near ${port}." >&2
+      exit 1
+    fi
+  done
+
+  echo "$candidate"
+}

+wait_for_port() {
+  local name="$1"
+  local port="$2"
+  local log_file="$3"
+
+  for _ in $(seq 1 40); do
+    if port_open "$port"; then
+      return 0
+    fi
+    if [[ -s "$log_file" ]] && grep -qiE 'error|unable|address already in use' "$log_file"; then
+      echo "ERROR: $name port-forward failed:" >&2
+      cat "$log_file" >&2
+      exit 1
+    fi
+    sleep 0.5
+  done
+  echo "ERROR: $name port-forward did not open 127.0.0.1:${port}." >&2
+  cat "$log_file" >&2 || true
+  exit 1
+}

+        print(node.get("id") or node.get("atom_id") or node.get("node_id") or "")
+        break
+PYEOF
+}

+decode_and_filter_doit_mart_log() {
+  python3 - 3<&0 <<'PYEOF'
+import json
+import os
+import re
+import sys
+
+with os.fdopen(3) as input_stream:
+    raw = input_stream.read()
+try:
+    decoded = json.loads(raw)
+except json.JSONDecodeError:
+    text = raw
+else:
+    data = decoded.get("data", raw) if isinstance(decoded, dict) else raw
+    text = data if isinstance(data, str) else json.dumps(data, indent=2, sort_keys=True)
+
+sensitive_line = re.compile(
+    r"TOKEN|SECRET|PASSWORD|PRIVATE KEY|CERTIFICATE|AUTHORIZATION|COOKIE|"
+    r"CREDENTIAL|API[_-]?KEY|ACCESS[_-]?KEY|REFRESH[_-]?TOKEN|"
+    r"CLIENT[_ -]?ID|CLIENT[_-]?SECRET|SENTRY|\bDSN\b|ENVIRONMENT VARIABLES|"
+    r"['\"]environment['\"]\s*:",
+    re.IGNORECASE,
+)
+
+for line in text.splitlines():
+    if sensitive_line.search(line):
+        continue
+    print(line)
+PYEOF
+}

+emit_doit_mart_lines() {
+  local doit_token="$1"
+  local doit_env="$2"
+  local atom_id="$3"
+  local atom_response_file="$4"
+  local seen_file="$5"
+  local monitoring_host="daas-monitoring.${doit_env}.me.sap.corp"
+  local mart_host="daas-mart.${doit_env}.me.sap.corp"
+  local graph_json
+  local runner_atom_id
+  local payloads_json
+  local mart_resource_id
+  local log_response
+  local line
+
+  if [[ -z "$atom_id" ]]; then
+    return 0
+  fi
+
+  mart_resource_id="$(extract_mart_resource_id <"$atom_response_file")"
+
+  if [[ -z "$mart_resource_id" ]]; then
+    if ! graph_json="$(doit_get "$doit_token" "https://${monitoring_host}/api/v1/proxy/scheduler/atoms/${atom_id}/graph?attribute=name&attribute=label")"; then
+      emit_line "DOIT" "failed to fetch atom graph."
+      return 0
+    fi
+    runner_atom_id="$(find_legacy_runner_atom_id <<<"$graph_json")"
+
+    if [[ -z "$runner_atom_id" ]]; then
+      emit_line "DOIT" "mart log resource not available yet."
+      return 0
+    fi
+
+    if ! payloads_json="$(doit_get "$doit_token" "https://${monitoring_host}/api/v1/proxy/scheduler/atoms/${runner_atom_id}?attribute=name&attribute=outgoing_payloads")"; then
+      emit_line "DOIT" "failed to fetch runner payloads."
+      return 0
+    fi
+    mart_resource_id="$(extract_mart_resource_id <<<"$payloads_json")"
+  fi
+
+  if [[ -z "$mart_resource_id" ]]; then
+    emit_line "DOIT" "mart log resource not available yet."
+    return 0
+  fi
+
+  if ! log_response="$(doit_get "$doit_token" "https://${mart_host}/api/v2/resources/${mart_resource_id}/file")"; then
+    emit_line "DOIT" "failed to download mart log resource."
+    return 0
+  fi
+
+  while IFS= read -r line; do
+    if [[ -z "$line" ]]; then
+      continue
+    fi
+    if grep -qxF "$line" "$seen_file" 2>/dev/null; then
+      continue
+    fi
+    printf '%s\n' "$line" >>"$seen_file"
+    emit_line "DOIT" "$line"
+  done < <(decode_and_filter_doit_mart_log <<<"$log_response" | tail -n "$DOIT_LOG_TAIL_LINES")

+is_inspection_terminal() {
+  local inspection_id="$1"
+  local response
+  local status
+  local curl_args=(-sS --fail-with-body "${BASE_URL}/api/v1/inspection/${inspection_id}")
+  if [[ "$AUTH" == "true" ]]; then
+    curl_args+=(-H "Authorization: Bearer ${TOKEN}")
+  fi
+  response="$(curl "${curl_args[@]}" 2>/dev/null)" || return 1
+  status="$(python3 -c 'import json,sys; print(json.loads(sys.argv[1]).get("status",""))' "$response")"
+  case "$status" in
+    COMPLETED|FAILED) return 0 ;;
+    *) return 1 ;;
+  esac

+_watch_single_atom() {
+  local doit_token="$1"
+  local doit_env="$2"
+  local atom_id="$3"
+  local source_job="$4"
+  local monitoring_host="daas-monitoring.${doit_env}.me.sap.corp"
+  local seen_file="${TMP_DIR}/doit-loki-seen-${atom_id}.txt"
+  local mart_seen_file="${TMP_DIR}/doit-mart-seen-${atom_id}.txt"
+  local http_status_file="${TMP_DIR}/doit-http-status-${atom_id}.txt"
+  local atom_response_file="${TMP_DIR}/doit-atom-${atom_id}.json"
+  local datasource_response_file="${TMP_DIR}/grafana-datasources.json"
+  local loki_response_file="${TMP_DIR}/doit-loki-${atom_id}.json"
+  local datasource_uid=""
+  local datasources_json
+  local atom_json
+  local atom_status=""
+  local loki_context
+  local loki_query_url
+  local loki_response
+  local datasource_status=""
+  local loki_status=""
+  local scheduler_status=""
+  local last_scheduler_poll_status=""
+  local last_datasource_poll_status=""
+  local last_loki_poll_status=""
+  local last_atom_status=""
+  local loki_link_emitted="${TMP_DIR}/doit-loki-link-emitted-${atom_id}"
+
+  : >"$seen_file"
+  : >"$mart_seen_file"
+
+  while true; do
+    if [[ -z "$datasource_uid" ]]; then
+      if doit_get_file "$doit_token" "https://grafana.me.sap.corp/api/datasources" "$datasource_response_file" "$http_status_file"; then
+        datasources_json="$(<"$datasource_response_file")"
+        datasource_uid="$(extract_grafana_loki_datasource_uid <<<"$datasources_json")"
+        if [[ -z "$datasource_uid" && "$last_datasource_poll_status" != "missing" ]]; then
+          emit_line "DOIT" "[${source_job}] grafana loki datasource not found; retrying."
+          last_datasource_poll_status="missing"
+        fi
+      else
+        datasource_status="$(<"$http_status_file")"
+        if [[ "$datasource_status" != "$last_datasource_poll_status" ]]; then
+          emit_line "DOIT" "[${source_job}] grafana datasource poll returned HTTP ${datasource_status}; retrying."
+          last_datasource_poll_status="$datasource_status"
+        fi
+      fi
+    fi
+
+    if doit_get_file "$doit_token" "https://${monitoring_host}/api/v1/proxy/scheduler/atoms/${atom_id}?attribute=name&attribute=label&attribute=status&attribute=state&attribute=status_transitions&attribute=outgoing_payloads" "$atom_response_file" "$http_status_file"; then
+      atom_json="$(<"$atom_response_file")"
+      atom_status="$(extract_atom_runtime_status <<<"$atom_json")"
+      if [[ -n "$atom_status" && "$atom_status" != "$last_atom_status" ]]; then
+        emit_line "DOIT" "[${source_job}] status: ${atom_status}"
+        last_atom_status="$atom_status"
+      fi
+
+      loki_context="$(build_inframon_loki_context "$doit_env" <<<"$atom_json")"
+      if [[ -n "$loki_context" && ! -f "$loki_link_emitted" ]]; then
+        emit_line "DOIT" "[${source_job}] inframon loki: $(build_inframon_loki_link <<<"$loki_context")"
+        : >"$loki_link_emitted"
+      fi
+
+      if should_query_loki_for_status "$atom_status"; then
+        if [[ -n "$datasource_uid" && -n "$loki_context" ]]; then
+          loki_query_url="$(build_loki_query_url "$datasource_uid" "$DOIT_LOG_TAIL_LINES" <<<"$loki_context")"
+          if doit_get_file "$doit_token" "$loki_query_url" "$loki_response_file" "$http_status_file"; then
+            loki_response="$(<"$loki_response_file")"
+            emit_loki_lines "$loki_response" "$seen_file"
+          else
+            loki_status="$(<"$http_status_file")"
+            if [[ "$loki_status" != "$last_loki_poll_status" ]]; then
+              emit_line "DOIT" "[${source_job}] inframon loki poll returned HTTP ${loki_status}; retrying."
+              last_loki_poll_status="$loki_status"
+            fi
+          fi
+        fi
+        emit_doit_mart_lines "$doit_token" "$doit_env" "$atom_id" "$atom_response_file" "$mart_seen_file"
+      fi
+
+      if is_terminal_doit_status "$atom_status"; then
+        emit_line "DOIT" "[${source_job}] atom ${atom_id} reached terminal: ${atom_status}; draining..."
+        sleep "$DOIT_POLL_SECONDS"
+        if [[ -n "$datasource_uid" && -n "$loki_context" ]]; then
+          loki_query_url="$(build_loki_query_url "$datasource_uid" "$DOIT_LOG_TAIL_LINES" <<<"$loki_context")"
+          if doit_get_file "$doit_token" "$loki_query_url" "$loki_response_file" "$http_status_file"; then
+            loki_response="$(<"$loki_response_file")"
+            emit_loki_lines "$loki_response" "$seen_file"
+          fi
+        fi
+        emit_doit_mart_lines "$doit_token" "$doit_env" "$atom_id" "$atom_response_file" "$mart_seen_file"
+        emit_line "DOIT" "[${source_job}] atom ${atom_id} polling finished: ${atom_status}"
+        return 0
+      fi
+    else
+      scheduler_status="$(<"$http_status_file")"
+      if [[ "$scheduler_status" != "$last_scheduler_poll_status" ]]; then
+        emit_line "DOIT" "[${source_job}] scheduler poll returned HTTP ${scheduler_status}; retrying."
+        last_scheduler_poll_status="$scheduler_status"
+      fi
+    fi
+    sleep "$DOIT_POLL_SECONDS"
+  done
+}

+start_doit_watch() {
+  local inspection_id="$1"
+
+  (
+    local doit_token
+    local deadline
+    local tracked_atoms=""
+    local atom_pids=()
+    local job_name job_logs metadata
+    local doit_status doit_url doit_env atom_id
+
+    doit_token="$(get_doit_token)"
+    if [[ -z "$doit_token" ]]; then
+      emit_line "DOIT" "logs skipped: set DOIT_TOKEN, DOIT_AUTH_TOKEN, or DOIT_API_TOKEN; root .env is checked first."
+      return 0
+    fi
+
+    deadline=$(( $(date +%s) + WAIT_SECONDS ))
+
+    while true; do
+      while IFS= read -r job_name; do
+        [[ -z "$job_name" ]] && continue
+        job_logs="$("${KUBECTL[@]}" logs "job/${job_name}" --all-containers=true --tail="$DOIT_METADATA_TAIL_LINES" 2>/dev/null || true)"
+        metadata="$(extract_doit_metadata <<<"$job_logs")"
+        [[ -z "$metadata" ]] && continue
+
+        IFS=$'\t' read -r doit_status doit_url doit_env atom_id <<<"$metadata"
+        [[ -z "$atom_id" ]] && continue
+
+        if printf '%s\n' "$tracked_atoms" | grep -qxF "$atom_id" 2>/dev/null; then
+          continue
+        fi
+
+        emit_line "DOIT" "[${job_name}] ${doit_status} ${doit_url}"
+        tracked_atoms="${tracked_atoms}${atom_id}"$'\n'
+
+        _watch_single_atom "$doit_token" "$doit_env" "$atom_id" "$job_name" &
+        atom_pids+=("$!")
+      done < <(find_jobs_for_inspection "$inspection_id")
+
+      if [[ "${#atom_pids[@]}" -gt 0 ]]; then
+        local any_alive=false
+        for pid in "${atom_pids[@]}"; do
+          if kill -0 "$pid" 2>/dev/null; then
+            any_alive=true
+            break
+          fi
+        done
+        if ! $any_alive; then
+          if is_inspection_terminal "$inspection_id"; then
+            emit_line "DOIT" "all atom watchers finished; inspection is terminal."
+            break
+          fi
+          emit_line "DOIT" "all atom watchers finished; waiting for new atoms (inspection still running)."
+          deadline=$(( $(date +%s) + WAIT_SECONDS ))
+          atom_pids=()
+        fi
+      fi
+
+      # Early exit: inspection already terminal — no jobs/atoms will ever appear
+      if [[ "${#atom_pids[@]}" -eq 0 && -z "$tracked_atoms" ]] && is_inspection_terminal "$inspection_id"; then
+        emit_line "DOIT" "inspection ${inspection_id} already terminal — no atoms expected."
+        return 0
+      fi
+
+      if [[ "$(date +%s)" -ge "$deadline" ]]; then
+        if [[ "${#atom_pids[@]}" -eq 0 && -z "$tracked_atoms" ]]; then
+          emit_line "DOIT" "no atom link found in inspection job logs within ${WAIT_SECONDS}s."
+        else
+          emit_line "DOIT" "no new atoms appeared within ${WAIT_SECONDS}s; giving up."
+        fi
+        return 0
+      fi
+
+      sleep "$DOIT_POLL_SECONDS"
+    done
+
+    for pid in "${atom_pids[@]}"; do
+      wait "$pid" 2>/dev/null || true
+    done
+  ) &
+  DOIT_WATCH_PID="$!"
+  PIDS+=("$DOIT_WATCH_PID")
+}

+start_k8s_watch() {
+  local inspection_id="$1"
+  local run_id="$2"
+
+  (
+    local deadline=$(( $(date +%s) + WAIT_SECONDS ))
+    local watched_jobs=""
+    local job_pids=()
+    local job_name handler_label
+
+    emit_line "K8S" "waiting for execution jobs..."
+
+    while true; do
+      while IFS= read -r job_name; do
+        [[ -z "$job_name" ]] && continue
+        if printf '%s\n' "$watched_jobs" | grep -qxF "$job_name" 2>/dev/null; then
+          continue
+        fi
+
+        handler_label="$(get_job_handler_label "$job_name")"
+        emit_line "K8S" "job: ${job_name} (${handler_label})"
+        "${KUBECTL[@]}" get job "$job_name" 2>&1 | pipe_with_prefix "K8S:${handler_label}"
+        print_pending_pod_events "$job_name"
+
+        (
+          wait_for_job_logs "$job_name" || exit 1
+          "${KUBECTL[@]}" logs -f "job/${job_name}" --all-containers=true --tail="$TAIL_LINES" 2>&1 \
+            | pipe_with_prefix "K8S:${handler_label}"
+        ) &
+        job_pids+=("$!")
+        watched_jobs="${watched_jobs}${job_name}"$'\n'
+      done < <(find_jobs_for_inspection "$inspection_id")
+
+      if [[ "${#job_pids[@]}" -gt 0 ]]; then
+        break
+      fi
+
+      if is_inspection_terminal "$inspection_id"; then
+        emit_line "K8S" "inspection ${inspection_id} already terminal — no jobs expected."
+        return 0
+      fi
+
+      if [[ "$(date +%s)" -ge "$deadline" ]]; then
+        emit_line "K8S" "no execution job found for inspection ${inspection_id} within ${WAIT_SECONDS}s."
+        print_platform_snapshot "temporal-worker" "TEMPORAL" "deploy/pipeline-fl-control-plane-temporal-worker" "$(matching_log_pattern "$inspection_id" "$run_id")"
+        print_platform_snapshot "control-plane" "CONTROL" "deploy/pipeline-fl-control-plane" "$(matching_log_pattern "$inspection_id" "$run_id")"
+        return 1
+      fi
+      sleep 2
+    done
+
+    while true; do
+      while IFS= read -r job_name; do
+        [[ -z "$job_name" ]] && continue
+        if printf '%s\n' "$watched_jobs" | grep -qxF "$job_name" 2>/dev/null; then
+          continue
+        fi
+
+        handler_label="$(get_job_handler_label "$job_name")"
+        emit_line "K8S" "new job: ${job_name} (${handler_label})"
+
+        (
+          wait_for_job_logs "$job_name" || exit 1
+          "${KUBECTL[@]}" logs -f "job/${job_name}" --all-containers=true --tail="$TAIL_LINES" 2>&1 \
+            | pipe_with_prefix "K8S:${handler_label}"
+        ) &
+        job_pids+=("$!")
+        watched_jobs="${watched_jobs}${job_name}"$'\n'
+      done < <(find_jobs_for_inspection "$inspection_id")
+
+      local any_alive=false
+      for pid in "${job_pids[@]}"; do
+        if kill -0 "$pid" 2>/dev/null; then
+          any_alive=true
+          break
+        fi
+      done
+      if ! $any_alive; then
+        break
+      fi
+
+      sleep "$JOB_DISCOVERY_SECONDS"
+    done
+
+    emit_line "K8S" "all job log streams finished."
+    for pid in "${job_pids[@]}"; do
+      wait "$pid" 2>/dev/null || true
+    done
+  ) &
+  K8S_WATCH_PID="$!"
+  PIDS+=("$K8S_WATCH_PID")
+}

+extract_doit_metadata() {
+  python3 - 3<&0 <<'PYEOF'
+import json
+import os
+import re
+import sys
+from urllib.parse import parse_qs, urlparse
+
+with os.fdopen(3) as input_stream:
+    lines = input_stream.read().splitlines()
+
+latest_status = "scheduled"
+latest_url = ""
+
+for line in lines:
+    if "atom_inspector_url" in line:
+        try:
+            payload = json.loads(line)
+        except json.JSONDecodeError:
+            payload = {}
+        latest_url = payload.get("atom_inspector_url") or latest_url
+        if payload.get("event") == "Atom scheduled":
+            latest_status = "scheduled"
+
+    match = re.search(r"\[(?P<status>[^\]]+)\].*(?P<url>https://(?:atom|doit)-inspector\.me\.sap\.corp\S+)", line)
+    if match:
+        latest_status = match.group("status")
+        latest_url = match.group("url")
+        continue
+
+    match = re.search(r"Scheduled\s+\S+\s+(?P<url>https://(?:atom|doit)-inspector\.me\.sap\.corp\S+)", line)
+    if match:
+        latest_status = "scheduled"
+        latest_url = match.group("url")
+
+if latest_url:
+    query = parse_qs(urlparse(latest_url).query)
+    doit_env = (query.get("env") or ["lithium"])[0]
+    atom_id = (query.get("atom_id") or [""])[0]
+    print("\t".join([latest_status, latest_url, doit_env, atom_id]))
+PYEOF

+emit_doit_mart_lines() {
+  local doit_token="$1"
+  local doit_env="$2"
+  local atom_id="$3"
+  local atom_response_file="$4"
+  local seen_file="$5"
+  local monitoring_host="daas-monitoring.${doit_env}.me.sap.corp"
+  local mart_host="daas-mart.${doit_env}.me.sap.corp"
+  local graph_json
+  local runner_atom_id
+  local payloads_json
+  local mart_resource_id
+  local log_response
+  local line
+
+  if [[ -z "$atom_id" ]]; then
+    return 0
+  fi
+
+  mart_resource_id="$(extract_mart_resource_id <"$atom_response_file")"
+
+  if [[ -z "$mart_resource_id" ]]; then
+    if ! graph_json="$(doit_get "$doit_token" "https://${monitoring_host}/api/v1/proxy/scheduler/atoms/${atom_id}/graph?attribute=name&attribute=label")"; then
+      emit_line "DOIT" "failed to fetch atom graph."
+      return 0
+    fi
+    runner_atom_id="$(find_legacy_runner_atom_id <<<"$graph_json")"
+
+    if [[ -z "$runner_atom_id" ]]; then
+      emit_line "DOIT" "mart log resource not available yet."
+      return 0
+    fi
+
+    if ! payloads_json="$(doit_get "$doit_token" "https://${monitoring_host}/api/v1/proxy/scheduler/atoms/${runner_atom_id}?attribute=name&attribute=outgoing_payloads")"; then
+      emit_line "DOIT" "failed to fetch runner payloads."
+      return 0
+    fi
+    mart_resource_id="$(extract_mart_resource_id <<<"$payloads_json")"
+  fi
+
+  if [[ -z "$mart_resource_id" ]]; then
+    emit_line "DOIT" "mart log resource not available yet."
+    return 0
+  fi
+
+  if ! log_response="$(doit_get "$doit_token" "https://${mart_host}/api/v2/resources/${mart_resource_id}/file")"; then
+    emit_line "DOIT" "failed to download mart log resource."
+    return 0
+  fi
+
+  while IFS= read -r line; do
+    if [[ -z "$line" ]]; then
+      continue
+    fi
+    if grep -qxF "$line" "$seen_file" 2>/dev/null; then
+      continue
+    fi
+    printf '%s\n' "$line" >>"$seen_file"
+    emit_line "DOIT" "$line"
+  done < <(decode_and_filter_doit_mart_log <<<"$log_response" | tail -n "$DOIT_LOG_TAIL_LINES")

+_watch_single_atom() {
+  local doit_token="$1"
+  local doit_env="$2"
+  local atom_id="$3"
+  local source_job="$4"
+  local monitoring_host="daas-monitoring.${doit_env}.me.sap.corp"
+  local seen_file="${TMP_DIR}/doit-loki-seen-${atom_id}.txt"
+  local mart_seen_file="${TMP_DIR}/doit-mart-seen-${atom_id}.txt"
+  local http_status_file="${TMP_DIR}/doit-http-status-${atom_id}.txt"
+  local atom_response_file="${TMP_DIR}/doit-atom-${atom_id}.json"
+  local datasource_response_file="${TMP_DIR}/grafana-datasources.json"
+  local loki_response_file="${TMP_DIR}/doit-loki-${atom_id}.json"
+  local datasource_uid=""
+  local datasources_json
+  local atom_json
+  local atom_status=""
+  local loki_context
+  local loki_query_url
+  local loki_response
+  local datasource_status=""
+  local loki_status=""
+  local scheduler_status=""
+  local last_scheduler_poll_status=""
+  local last_datasource_poll_status=""
+  local last_loki_poll_status=""
+  local last_atom_status=""
+  local loki_link_emitted="${TMP_DIR}/doit-loki-link-emitted-${atom_id}"
+
+  : >"$seen_file"
+  : >"$mart_seen_file"
+
+  while true; do
+    if [[ -z "$datasource_uid" ]]; then
+      if doit_get_file "$doit_token" "https://grafana.me.sap.corp/api/datasources" "$datasource_response_file" "$http_status_file"; then
+        datasources_json="$(<"$datasource_response_file")"
+        datasource_uid="$(extract_grafana_loki_datasource_uid <<<"$datasources_json")"
+        if [[ -z "$datasource_uid" && "$last_datasource_poll_status" != "missing" ]]; then
+          emit_line "DOIT" "[${source_job}] grafana loki datasource not found; retrying."
+          last_datasource_poll_status="missing"
+        fi
+      else
+        datasource_status="$(<"$http_status_file")"
+        if [[ "$datasource_status" != "$last_datasource_poll_status" ]]; then
+          emit_line "DOIT" "[${source_job}] grafana datasource poll returned HTTP ${datasource_status}; retrying."
+          last_datasource_poll_status="$datasource_status"
+        fi
+      fi
+    fi
+
+    if doit_get_file "$doit_token" "https://${monitoring_host}/api/v1/proxy/scheduler/atoms/${atom_id}?attribute=name&attribute=label&attribute=status&attribute=state&attribute=status_transitions&attribute=outgoing_payloads" "$atom_response_file" "$http_status_file"; then
+      atom_json="$(<"$atom_response_file")"
+      atom_status="$(extract_atom_runtime_status <<<"$atom_json")"
+      if [[ -n "$atom_status" && "$atom_status" != "$last_atom_status" ]]; then
+        emit_line "DOIT" "[${source_job}] status: ${atom_status}"
+        last_atom_status="$atom_status"
+      fi
+
+      loki_context="$(build_inframon_loki_context "$doit_env" <<<"$atom_json")"
+      if [[ -n "$loki_context" && ! -f "$loki_link_emitted" ]]; then
+        emit_line "DOIT" "[${source_job}] inframon loki: $(build_inframon_loki_link <<<"$loki_context")"
+        : >"$loki_link_emitted"
+      fi
+
+      if should_query_loki_for_status "$atom_status"; then
+        if [[ -n "$datasource_uid" && -n "$loki_context" ]]; then
+          loki_query_url="$(build_loki_query_url "$datasource_uid" "$DOIT_LOG_TAIL_LINES" <<<"$loki_context")"
+          if doit_get_file "$doit_token" "$loki_query_url" "$loki_response_file" "$http_status_file"; then
+            loki_response="$(<"$loki_response_file")"
+            emit_loki_lines "$loki_response" "$seen_file"
+          else
+            loki_status="$(<"$http_status_file")"
+            if [[ "$loki_status" != "$last_loki_poll_status" ]]; then
+              emit_line "DOIT" "[${source_job}] inframon loki poll returned HTTP ${loki_status}; retrying."
+              last_loki_poll_status="$loki_status"
+            fi
+          fi
+        fi
+        emit_doit_mart_lines "$doit_token" "$doit_env" "$atom_id" "$atom_response_file" "$mart_seen_file"
+      fi
+
+      if is_terminal_doit_status "$atom_status"; then
+        emit_line "DOIT" "[${source_job}] atom ${atom_id} reached terminal: ${atom_status}; draining..."
+        sleep "$DOIT_POLL_SECONDS"
+        if [[ -n "$datasource_uid" && -n "$loki_context" ]]; then
+          loki_query_url="$(build_loki_query_url "$datasource_uid" "$DOIT_LOG_TAIL_LINES" <<<"$loki_context")"
+          if doit_get_file "$doit_token" "$loki_query_url" "$loki_response_file" "$http_status_file"; then
+            loki_response="$(<"$loki_response_file")"
+            emit_loki_lines "$loki_response" "$seen_file"
+          fi
+        fi
+        emit_doit_mart_lines "$doit_token" "$doit_env" "$atom_id" "$atom_response_file" "$mart_seen_file"
+        emit_line "DOIT" "[${source_job}] atom ${atom_id} polling finished: ${atom_status}"
+        return 0
+      fi
+    else
+      scheduler_status="$(<"$http_status_file")"
+      if [[ "$scheduler_status" != "$last_scheduler_poll_status" ]]; then
+        emit_line "DOIT" "[${source_job}] scheduler poll returned HTTP ${scheduler_status}; retrying."
+        last_scheduler_poll_status="$scheduler_status"
+      fi
+    fi
+    sleep "$DOIT_POLL_SECONDS"
+  done
+}

+start_doit_watch() {
+  local inspection_id="$1"
+
+  (
+    local doit_token
+    local deadline
+    local tracked_atoms=""
+    local atom_pids=()
+    local job_name job_logs metadata
+    local doit_status doit_url doit_env atom_id
+
+    doit_token="$(get_doit_token)"
+    if [[ -z "$doit_token" ]]; then
+      emit_line "DOIT" "logs skipped: set DOIT_TOKEN, DOIT_AUTH_TOKEN, or DOIT_API_TOKEN; root .env is checked first."
+      return 0
+    fi
+
+    deadline=$(( $(date +%s) + WAIT_SECONDS ))
+
+    # Discovery loop: find atoms from job logs, spawn a watcher per atom
+    while true; do
+      while IFS= read -r job_name; do
+        [[ -z "$job_name" ]] && continue
+        job_logs="$("${KUBECTL[@]}" logs "job/${job_name}" --all-containers=true --tail="$DOIT_METADATA_TAIL_LINES" 2>/dev/null || true)"
+        metadata="$(extract_doit_metadata <<<"$job_logs")"
+        [[ -z "$metadata" ]] && continue
+
+        IFS=$'\t' read -r doit_status doit_url doit_env atom_id <<<"$metadata"
+        [[ -z "$atom_id" ]] && continue
+
+        # Already tracking?
+        if printf '%s\n' "$tracked_atoms" | grep -qxF "$atom_id" 2>/dev/null; then
+          continue
+        fi
+
+        emit_line "DOIT" "[${job_name}] ${doit_status} ${doit_url}"
+        tracked_atoms="${tracked_atoms}${atom_id}"$'\n'
+
+        _watch_single_atom "$doit_token" "$doit_env" "$atom_id" "$job_name" &
+        atom_pids+=("$!")
+      done < <(find_jobs_for_inspection "$inspection_id")
+
+      # Check if all atom watchers have exited
+      if [[ "${#atom_pids[@]}" -gt 0 ]]; then
+        local any_alive=false
+        for pid in "${atom_pids[@]}"; do
+          if kill -0 "$pid" 2>/dev/null; then
+            any_alive=true
+            break
+          fi
+        done
+        if ! $any_alive; then
+          # All current atoms finished — but new jobs (finalizer) may still spawn atoms.
+          # Check if the inspection has reached terminal status before giving up.
+          if is_inspection_terminal "$inspection_id"; then
+            emit_line "DOIT" "all atom watchers finished; inspection is terminal."
+            break
+          fi
+          emit_line "DOIT" "all atom watchers finished; waiting for new atoms (inspection still running)."
+          # Reset deadline to allow time for the next Doit atom to appear.
+          deadline=$(( $(date +%s) + WAIT_SECONDS ))
+          atom_pids=()
+        fi
+      fi
+
+      # Early exit: inspection already terminal — no jobs/atoms will ever appear
+      if [[ "${#atom_pids[@]}" -eq 0 && -z "$tracked_atoms" ]] && is_inspection_terminal "$inspection_id"; then
+        emit_line "DOIT" "inspection ${inspection_id} already terminal — no atoms expected."
+        return 0
+      fi
+
+      # Timeout: no atoms discovered within deadline
+      if [[ "$(date +%s)" -ge "$deadline" ]]; then
+        if [[ "${#atom_pids[@]}" -eq 0 && -z "$tracked_atoms" ]]; then
+          emit_line "DOIT" "no atom link found in inspection job logs within ${WAIT_SECONDS}s."
+        else
+          emit_line "DOIT" "no new atoms appeared within ${WAIT_SECONDS}s; giving up."
+        fi
+        return 0
+      fi
+
+      sleep "$DOIT_POLL_SECONDS"
+    done
+
+    for pid in "${atom_pids[@]}"; do
+      wait "$pid" 2>/dev/null || true
+    done
+  ) &
+  DOIT_WATCH_PID="$!"
+  PIDS+=("$DOIT_WATCH_PID")
+}

+start_k8s_watch() {
+  local inspection_id="$1"
+  local run_id="$2"
+
+  (
+    local deadline=$(( $(date +%s) + WAIT_SECONDS ))
+    local watched_jobs=""
+    local job_pids=()
+    local job_name handler_label
+
+    emit_line "K8S" "waiting for execution jobs..."
+
+    # Phase 1: wait for at least one job to appear
+    while true; do
+      while IFS= read -r job_name; do
+        [[ -z "$job_name" ]] && continue
+        # Already watching?
+        if printf '%s\n' "$watched_jobs" | grep -qxF "$job_name" 2>/dev/null; then
+          continue
+        fi
+
+        handler_label="$(get_job_handler_label "$job_name")"
+        emit_line "K8S" "job: ${job_name} (${handler_label})"
+        "${KUBECTL[@]}" get job "$job_name" 2>&1 | pipe_with_prefix "K8S:${handler_label}"
+        print_pending_pod_events "$job_name"
+
+        (
+          wait_for_job_logs "$job_name" || exit 1
+          "${KUBECTL[@]}" logs -f "job/${job_name}" --all-containers=true --tail="$TAIL_LINES" 2>&1 \
+            | pipe_with_prefix "K8S:${handler_label}"
+        ) &
+        job_pids+=("$!")
+        watched_jobs="${watched_jobs}${job_name}"$'\n'
+      done < <(find_jobs_for_inspection "$inspection_id")
+
+      if [[ "${#job_pids[@]}" -gt 0 ]]; then
+        break
+      fi
+
+      if is_inspection_terminal "$inspection_id"; then
+        emit_line "K8S" "inspection ${inspection_id} already terminal — no jobs expected."
+        return 0
+      fi
+
+      if [[ "$(date +%s)" -ge "$deadline" ]]; then
+        emit_line "K8S" "no execution job found for inspection ${inspection_id} within ${WAIT_SECONDS}s."
+        print_platform_snapshot "temporal-worker" "TEMPORAL" "deploy/pipeline-fl-control-plane-temporal-worker" "$(matching_log_pattern "$inspection_id" "$run_id")"
+        print_platform_snapshot "control-plane" "CONTROL" "deploy/pipeline-fl-control-plane" "$(matching_log_pattern "$inspection_id" "$run_id")"
+        return 1
+      fi
+      sleep 2
+    done
+
+    # Phase 2: keep discovering new jobs (finalizer appears later) until all streams close
+    while true; do
+      while IFS= read -r job_name; do
+        [[ -z "$job_name" ]] && continue
+        if printf '%s\n' "$watched_jobs" | grep -qxF "$job_name" 2>/dev/null; then
+          continue
+        fi
+
+        handler_label="$(get_job_handler_label "$job_name")"
+        emit_line "K8S" "new job: ${job_name} (${handler_label})"
+
+        (
+          wait_for_job_logs "$job_name" || exit 1
+          "${KUBECTL[@]}" logs -f "job/${job_name}" --all-containers=true --tail="$TAIL_LINES" 2>&1 \
+            | pipe_with_prefix "K8S:${handler_label}"
+        ) &
+        job_pids+=("$!")
+        watched_jobs="${watched_jobs}${job_name}"$'\n'
+      done < <(find_jobs_for_inspection "$inspection_id")
+
+      local any_alive=false
+      for pid in "${job_pids[@]}"; do
+        if kill -0 "$pid" 2>/dev/null; then
+          any_alive=true
+          break
+        fi
+      done
+      if ! $any_alive; then
+        break
+      fi
+
+      sleep "$JOB_DISCOVERY_SECONDS"
+    done
+
+    emit_line "K8S" "all job log streams finished."
+    for pid in "${job_pids[@]}"; do
+      wait "$pid" 2>/dev/null || true
+    done
+  ) &
+  K8S_WATCH_PID="$!"
+  PIDS+=("$K8S_WATCH_PID")
+}
