+"""Finalizer agent task runner — synchronous; called directly by scripts/run_agent_task.py.
+
+All pipeline3 and scheduling_doit calls are sequenced here. Tests can patch
+'app.agent_task_runner.pipeline3' and 'app.agent_task_runner.scheduling_doit'
+via unittest.mock.patch.
+"""
+from __future__ import annotations
+
+import logging
+
+from app.models import FinalizerTaskRequest, FinalizerTaskResult, RawTrajectory
+
+log = logging.getLogger(__name__)
+
+from hana_program_synthesis.cline_central_api import pipeline3, scheduling_doit
+
+
+def _adjust_resource_requirements(task: dict) -> None:
+    # Override default HANA-sized resource requirements — the defaults (4 CPUs,
+    # 16 GiB RAM) cause long scheduling queues. The finalizer is a lightweight
+    # synthesis task that completes comfortably on small allocations.
+    task["requirement_cpus_n"] = 1
+    task["requirement_mem_gib"] = 4
+    task["requirement_disk_gib"] = 8
+
+
+def run(request: FinalizerTaskRequest) -> FinalizerTaskResult:
+    """Execute the finalizer agent task synchronously via Cline Central.
+
+    Must be called from a thread (via run_in_executor) — never from the async event loop.
+
+    SECURITY: Never logs environment_variables values.
+
+    Args:
+        request: Fully assembled finalizer task request.
+
+    Returns:
+        FinalizerTaskResult with completion_text and token counts from the trajectory.
+
+    Raises:
+        RuntimeError: when the task_id is absent from the returned trajectories.
+        KeyError: when completion_text is absent from the trajectory (hard failure —
+            the agent produced no output).
+    """
+    log.info(
+        "Finalizer agent task runner started: task_id=%s repo_url=%s",
+        request.task_id,
+        request.repo_url,
+    )
+
+    swe_task = pipeline3.SoftwareEngineeringTask(
+        task_text=request.task_text,
+        repo_url=request.repo_url,
+        repo_commitish=request.repo_commitish,
+        config_repo_url=request.config_repo_url,
+        config_repo_commitish=request.config_repo_commitish,
+        config_mcp_paths=request.config_mcp_paths,
+        config_skill_paths=request.config_skill_paths,
+    )
+
+    tasks_map = pipeline3.trigger_cline_central({request.task_id: swe_task})
+
+    for task in tasks_map.values():
+        _adjust_resource_requirements(task)
+
+    scheduling_doit.configure_doit()
+
+    scheduled_tasks = scheduling_doit.schedule_tasks(tasks_map, atom_graph_priority=100)
+
+    atom = scheduled_tasks.get(request.task_id) or {}
+    log.info(
+        "Atom scheduled: task_id=%s atom_id=%s atom_inspector_url=%s",
+        request.task_id,
+        atom.get("atom_id"),
+        atom.get("atom_inspector_url"),
+    )
+
+    trajs = scheduling_doit.collect_trajectories(scheduled_tasks)
+
+    raw_traj_dict = trajs.get(request.task_id)
+    if raw_traj_dict is None:
+        raise RuntimeError(
+            f"task_id {request.task_id!r} not found in trajectories returned by SDK"
+        )
+    traj = RawTrajectory.model_validate(raw_traj_dict)
+
+    # KeyError here is intentional — completion_text is the primary task output.
+    completion_text: str = pipeline3.get_completion_text(trajs)[request.task_id]
+
+    result = FinalizerTaskResult(
+        task_id=request.task_id,
+        completion_text=completion_text,
+        mlflow_run_id=traj.mlflow_run_id,
+    )
+
+    try:
+        cline_state = traj.parse_cline_state()
+        item = cline_state.current_task_item
+        if item is not None:
+            result.tokens_in = item.tokens_in
+            result.tokens_out = item.tokens_out
+    except Exception:
+        log.warning("Failed to extract token counts for task %s", request.task_id, exc_info=True)
+
+    log.info(
+        "Finalizer agent task runner completed: task_id=%s mlflow_run_id=%s",
+        request.task_id,
+        result.mlflow_run_id,
+    )
+    return result

+"""Finalizer selection — routing and selection hierarchy."""
+from __future__ import annotations
+
+import structlog
+from sqlalchemy import select
+from sqlalchemy.ext.asyncio import AsyncSession
+
+from fl_control_plane.database import FinalizerRegistry, PipelineAnalyzerResult

+async def should_dispatch(inspection_id: str, session: AsyncSession) -> tuple[bool, str | None]:
+    """Decide whether to dispatch the finalizer based on pipeline analyzer gate verdicts.
+
+    Any STOP verdict with require_finalizer=False takes precedence and suppresses
+    finalization entirely — a kill switch active means stop everything, regardless
+    of what other gates say.
+
+    Returns:
+        (True, None) when all gates passed, or all STOP verdicts require finalization.
+        (False, skip_reason) when any STOP verdict has require_finalizer=False.
+    """
+    result = await session.execute(
+        select(PipelineAnalyzerResult).where(
+            PipelineAnalyzerResult.inspection_id == inspection_id
+        )
+    )
+    verdicts = result.scalars().all()
+
+    stop_verdicts = [v for v in verdicts if v.analysis_decision == "STOP"]
+
+    if not stop_verdicts:
+        return True, None
+
+    # Any gate that says "stop and don't notify" takes precedence over all others.
+    suppress_gates = [v.gate for v in stop_verdicts if not v.require_finalizer]
+    if suppress_gates:
+        skip_reason = f"analysis stopped by gate(s) {suppress_gates!r} with no finalizer required"
+        log.info("finalizer_skipped", inspection_id=inspection_id, reason=skip_reason)
+        return False, skip_reason

+## Why
+
+After all fault handlers have run and their findings are stored in HDLF, there is no component to act on those findings and close the loop with the developer. The Finalizer Dispatcher fills this gap: it is the bridge between handler execution completing and the developer receiving a result (a PR comment, a notification, or nothing when appropriate).
+
+## What Changes
+
+- **NEW** `finalizer_dispatcher` Temporal activity (`dispatch_finalizer_activity`) — the final step in `PipelineInspectionWorkflow`, called after all handler k8s Jobs reach a terminal state
+- **NEW** `finalizer_registry` self-bootstrap — the default finalizer entry is upserted on first use; no manual DB seeding required
+- **NEW** Gate-based routing — reads `pipeline_analyzer_results` to decide whether to dispatch the finalizer or skip it entirely; a gate that set `require_finalizer = false` suppresses dispatch
+- **NEW** Finalizer selection hierarchy — repo-level config → org-level config → default. Initial implementation hardcodes the default; repo and org hooks are empty stubs ready for future implementation
+- **NEW** `FinalizationResult` Temporal model — returned by the activity carrying dispatch outcome, `finalizer_id`, `finalizer_execution_id`, and skip/error reason
+- **MODIFIED** `PipelineInspectionWorkflow` — `dispatch_finalizer_activity` added as the final step with `start_to_close_timeout = timedelta(minutes=50)`
+- **MODIFIED** `inspection.status` — set to `COMPLETED` or `FAILED` by the Finalizer Dispatcher as its last act, making it the component that closes the inspection lifecycle
+
+## Capabilities
+
+### New Capabilities
+
+- `finalizer-dispatcher`: Selects and dispatches the appropriate finalizer for a completed inspection; records the outcome in `finalizer_execution`; marks the inspection terminal

+one MCP to read inspection findings, and either the GitHub MCP or curl to post the result.
