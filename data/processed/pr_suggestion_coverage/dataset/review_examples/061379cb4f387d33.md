+Production uses Alembic migrations against HANA. For local kind deploys with SQLite,

+## 5. Circuit Breaker
+
+### 5.1 Trigger open state
+
+1. Stop Postgres: `docker compose -f docker-compose.dev.yml stop`
+2. Observe controller logs: 3 consecutive failures → "circuit breaker OPEN"
+3. Backoff increases exponentially (10s → 20s → 40s...)
+
+### 5.2 Recovery
+
+1. Restart Postgres: `docker compose -f docker-compose.dev.yml start`
+2. Next successful reconcile → "circuit breaker CLOSED"
+3. Normal relist interval resumes
+

+"""Circuit breaker for controller reconciliation failures."""
+
+import logging
+import time
+
+_logger = logging.getLogger("fl_controller")
+
+_DEFAULT_FAILURE_THRESHOLD = 3
+_DEFAULT_BASE_BACKOFF_SECONDS = 10.0
+_DEFAULT_MAX_BACKOFF_SECONDS = 300.0
+
+
+class CircuitBreaker:
+    """Exponential backoff circuit breaker for the reconcile loop.
+    After `failure_threshold` consecutive failures the circuit opens and imposes
+    exponential backoff before allowing the next attempt. A single success resets
+    the breaker to closed.
+    """
+
+    def __init__(
+        self,
+        failure_threshold: int = _DEFAULT_FAILURE_THRESHOLD,
+        base_backoff_seconds: float = _DEFAULT_BASE_BACKOFF_SECONDS,
+        max_backoff_seconds: float = _DEFAULT_MAX_BACKOFF_SECONDS,
+    ) -> None:
+        self._failure_threshold = failure_threshold
+        self._base_backoff_seconds = base_backoff_seconds
+        self._max_backoff_seconds = max_backoff_seconds
+        self._consecutive_failures = 0
+        self._last_failure_time: float | None = None
+
+    @property
+    def is_open(self) -> bool:
+        """Whether the circuit is open (failures exceeded threshold)."""
+        return self._consecutive_failures >= self._failure_threshold
+
+    @property
+    def consecutive_failures(self) -> int:
+        """Number of consecutive failures since the last success."""
+        return self._consecutive_failures
+
+    def record_success(self) -> None:
+        """Reset the breaker on a successful operation."""
+        if self._consecutive_failures > 0:
+            _logger.info(
+                "Circuit breaker reset after %d consecutive failures",
+                self._consecutive_failures,
+            )
+        self._consecutive_failures = 0
+        self._last_failure_time = None
+
+    def record_failure(self) -> None:
+        """Record a failure. Opens the circuit after threshold is reached."""
+        self._consecutive_failures += 1
+        self._last_failure_time = time.monotonic()
+        if self._consecutive_failures == self._failure_threshold:
+            _logger.critical(
+                "Circuit breaker opened after %d consecutive failures",
+                self._failure_threshold,
+            )
+
+    def backoff_seconds(self) -> float:
+        """Compute the current backoff duration based on failure count.
+        Returns 0 if the circuit is closed (below threshold).
+        """
+        if not self.is_open:
+            return 0.0
+        exponent = self._consecutive_failures - self._failure_threshold
+        backoff = self._base_backoff_seconds * (2**exponent)
+        return min(backoff, self._max_backoff_seconds)
+
+    def should_allow_request(self) -> bool:
+        """Check whether enough time has elapsed since last failure to allow a retry."""
+        if not self.is_open:
+            return True
+        if self._last_failure_time is None:
+            return True
+        elapsed = time.monotonic() - self._last_failure_time
+        return elapsed >= self.backoff_seconds()
+
+    def reset(self) -> None:
+        """Force-reset the breaker to closed state (used in tests)."""
+        self._consecutive_failures = 0
+        self._last_failure_time = None

+async def _relist_loop() -> None:
+    """Global background task that runs _full_list_reconcile on a single interval.
+
+    Unlike @kopf.daemon (which spawns one instance per CR), this runs exactly once
+    regardless of how many CRs exist. Uses the circuit breaker for exponential backoff
+    on persistent failures.
+    """
+    assert _relist_stop_event is not None
+
+    try:
+        await asyncio.wait_for(_relist_stop_event.wait(), timeout=app_settings.controller_relist_interval_seconds)
+        return
+    except asyncio.TimeoutError:
+        pass
+
+    while not _relist_stop_event.is_set():
+        if not _circuit_breaker.should_allow_request():
+            backoff = _circuit_breaker.backoff_seconds()
+            _logger.warning("Circuit breaker open, backing off %.0fs", backoff)
+            try:
+                await asyncio.wait_for(_relist_stop_event.wait(), timeout=backoff)
+                return
+            except asyncio.TimeoutError:
+                continue
+
+        try:
+            async with session_scope() as session:
+                start = time.monotonic()
+                await _full_list_reconcile(session)
+                duration_ms = int((time.monotonic() - start) * 1000)
+                await _update_controller_status(session, duration_ms)
+            _circuit_breaker.record_success()
+        except CrdNotFoundError:
+            _logger.debug("CRD still not installed — will retry next relist cycle")
+        except Exception:  # pylint: disable=broad-exception-caught
+            _logger.exception("Re-LIST cycle failed")
+            _circuit_breaker.record_failure()
+
+        try:
+            await asyncio.wait_for(_relist_stop_event.wait(), timeout=app_settings.controller_relist_interval_seconds)
+            return
+        except asyncio.TimeoutError:
+            pass
+
+
+async def _full_list_reconcile(session: AsyncSession) -> None:
+    """Perform a full LIST and heal the DB in both directions.
+
+    Deactivates DB entries not in the live set (deleted CRs whose events were missed),
+    and reconciles CRs that exist in K8s but are missing or stale in the DB (missed
+    ADDED/MODIFIED events).
+
+    Raises:
+        CrdNotFoundError: When the FaultHandler CRD is not installed (404 from K8s API).
+    """
+    api = get_async_custom_objects_api()
+
+    all_items: list[RawCRObject] = []
+    continue_token: str | None = None
+
+    while True:
+        list_kwargs: dict[str, Any] = {
+            "group": CRD_GROUP,
+            "version": CRD_VERSION,
+            "namespace": app_settings.controller_watch_namespace,
+            "plural": CRD_PLURAL,
+        }
+        if continue_token:
+            list_kwargs["_continue"] = continue_token
+
+        try:
+            response = await asyncio.wait_for(
+                api.list_namespaced_custom_object(**list_kwargs),
+                timeout=_K8S_API_TIMEOUT_SECONDS,
+            )
+        except ApiException as exc:
+            if exc.status == 404:
+                raise CrdNotFoundError(
+                    "FaultHandler CRD not installed — skipping relist cycle"
+                ) from exc
+            raise
+        items = response["items"]
+        all_items.extend(RawCRObject.model_validate(item) for item in items)
+
+        metadata = response.get("metadata", {})
+        continue_token = metadata.get("continue")
+
+        if not continue_token:
+            break
+
+    live_set: set[tuple[str, str]] = set()
+    for cr in all_items:
+        live_set.add((cr.metadata.name, cr.metadata.namespace))
+
+    deactivated = await deactivate_handlers_not_in(session, live_set)
+    if deactivated > 0:
+        await session.commit()
+        _logger.info("Deactivated %d handlers not in live set", deactivated)
+
+    for cr in all_items:
+        try:
+            async with session_scope() as cr_session:
+                await _reconcile_cr(cr, cr_session)
+        except kopf.TemporaryError:
+            _logger.debug("Re-LIST reconcile skipped %s/%s (transient error)", cr.metadata.name, cr.metadata.namespace)
+        except Exception:  # pylint: disable=broad-exception-caught
+            _logger.warning("Re-LIST reconcile failed for %s/%s", cr.metadata.name, cr.metadata.namespace)
+
+
+async def _check_crd_exists() -> bool:
+    """Check whether the FaultHandler CRD is installed by attempting a LIST with limit=1.
+
+    Returns True if the CRD exists, False if not found. Non-fatal — the controller
+    can start without the CRD and begin reconciling once it appears.
+    """
+    api = get_async_custom_objects_api()
+    try:
+        await asyncio.wait_for(
+            api.list_namespaced_custom_object(
+                group=CRD_GROUP,
+                version=CRD_VERSION,

+                plural=CRD_PLURAL,
+                limit=1,

+        )
+        return True
+    except asyncio.TimeoutError:
+        _logger.warning("Timed out checking FaultHandler CRD existence — will retry in relist loop")
+        return False
+    except ApiException as exc:
+        if exc.status == 404:
+            _logger.warning(
+                "FaultHandler CRD not found (group=%s, version=%s, plural=%s) "
+                "— controller will wait and retry each relist cycle",
+                CRD_GROUP, CRD_VERSION, CRD_PLURAL,
+            )
+            return False
+        raise

+async def _wait_for_db(*, max_retries: int = 60) -> None:
+    """Wait with exponential backoff until DB connection succeeds.
+
+    Raises:
+        DatabaseUnavailableError: If DB is unreachable after max_retries attempts.
+    """
+    interval = 1.0
+    max_interval = 30.0
+    for attempt in range(1, max_retries + 1):
+        try:
+            await verify_db_connection()
+            await ensure_tables()
+            _logger.info("Database connection established")
+            return
+        except Exception as exc:  # pylint: disable=broad-exception-caught
+            if attempt == max_retries:
+                raise DatabaseUnavailableError(f"Database unreachable after {max_retries} attempts (~10 min)") from exc
+            _logger.warning("Database not ready, retrying in %.0fs (%d/%d)", interval, attempt, max_retries)
+            await asyncio.sleep(interval)
+            interval = min(interval * 2, max_interval)
+

+        clusterwide=False,
+    )

+async def validate_fault_id_specificity(
+    spec: FaultHandlerSpec,
+    cr_name: str,
+    cr_namespace: str,
+    session: AsyncSession,
+) -> list[str]:
+    """Check that no duplicate default handler exists for the same fault_id.
+
+    Per design (§5.4): multiple CRs with the same fault_id may coexist when they
+    have different repo_patterns — the runtime dispatcher picks the most specific
+    match by weight. The only invalid case is two default handlers (both without
+    repo_patterns) for the same fault_id, because the weight algorithm cannot
+    choose between them.
+
+    Uses SELECT FOR UPDATE to prevent concurrent default handlers from both
+    passing the check simultaneously.
+    """
+    errors: list[str] = []
+
