+def _failed_stages_from_pipeline_tree(
+    raw: list[dict],
+) -> list[tuple[str, str | None]]:
+    """Derive (stage_name, None) pairs from pipeline_tree.json entries.
+
+    Mirrors the logic in _blue_ocean_stage_items: a stage is failed when its
+    result is not SUCCESS, NOT_EXECUTED, or None (indeterminate is treated as
+    failed for consistency with the capture plan builder).
+    """
+    failed: list[tuple[str, str | None]] = []
+    for item in raw:
+        node = RawBlueOceanNode.model_validate(item).to_blue_ocean_node()
+        if node.result in ("SUCCESS", "NOT_EXECUTED"):
+            continue
+        # Indeterminate — treat as failed (same as capture plan builder)
+        failed.append((node.display_name, None))
+
+    return failed

+def _failed_stages_from_wfapi(
+    raw: dict,
+) -> list[tuple[str, str | None]]:
+    """Derive (stage_name, error_message) pairs from wfapi_describe.json.

+    Mirrors the logic in _wfapi_stage_items: skips SUCCESS and NOT_EXECUTED.
+    """
+    failed: list[tuple[str, str | None]] = []

+        status = stage.get("status", "")
+        if status in ("SUCCESS", "NOT_EXECUTED"):
+            continue
+        stage_name = stage.get("name", "unknown")
+        error_message: str | None = stage.get("error", {}).get("message") if stage.get("error") else None
+        failed.append((stage_name, error_message))
+    return failed
