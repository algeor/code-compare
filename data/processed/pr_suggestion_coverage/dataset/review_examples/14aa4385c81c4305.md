 # pipeline-fl-control-plane
 FL Control Plane - ingests CI/CD pipeline failure events, orchestrates fault analysis across isolated Kubernetes jobs, and collects structured findings. Runs on BTP Kyma.
+
+## Database migrations
+
+Schema is managed with Alembic. The SQLAlchemy ORM in `fl_control_plane/database.py` is the source of truth; `db/schema.sql` is kept as a human-readable snapshot.
+
+### Adding a migration
+
+1. Change the ORM model.
+2. Generate the migration:
+   ```
+   alembic revision --autogenerate -m "describe the change"
+   ```
+3. **Review the generated file in `db/migrations/versions/`** — autogen handles most cases but may miss server-side defaults, CHECK constraints, or HANA-specific DDL. Edit the file if needed.
+4. Commit the model change AND the migration in the same PR.
+
+### Applying migrations
+
+Locally:
+```
+alembic upgrade head            # apply all pending migrations
+alembic downgrade -1            # roll back one migration
+alembic current                 # show the applied revision
+alembic history                 # list all revisions
+```
+
+In production: an init container in the Pod runs `alembic upgrade head` automatically before the API container starts (`chart/templates/deployment.yaml`). Migrations execute against the `DATABASE_URL` from the `pipeline-fl-control-plane-db` Secret.
+
+### Notes
+
+- The same `DATABASE_URL` env var drives both the app and migrations — `db/migrations/env.py` reads it from `Settings`.
+- Tests still load `db/schema.sql` directly (`tests/conftest.py`) and do not exercise Alembic. The two are kept in sync via the parity check in the migration's verification step.
+- Rollback in production is "deploy the previous image"; the older image's init container will see a higher applied revision and exit cleanly without running anything (it does not auto-downgrade).
+

+"""Alembic environment for FL Control Plane.
+
+Pulls the database URL from ``fl_control_plane.config.settings.database_url``
+so the same ``DATABASE_URL`` env var drives the app and the migrations. The
+app uses async SQLAlchemy; Alembic itself is sync, so we drive the async
+engine via ``connection.run_sync`` while the migration script runs in the
+sync callback.
+
+``render_as_batch`` is enabled only on SQLite — local dev / tests use SQLite
+which needs batch mode for ALTER TABLE on CHECK constraints, FKs, etc.
+HANA does not need batch mode.
+"""
+
+from __future__ import annotations
+
+import asyncio
+from logging.config import fileConfig
+
+from alembic import context
+from sqlalchemy.engine import Connection
+from sqlalchemy.ext.asyncio import async_engine_from_config
+
+from fl_control_plane.config import settings
+from fl_control_plane.database import Base
+
+config = context.config
+
+if config.config_file_name is not None:
+    fileConfig(config.config_file_name)
+
+# Programmatically set the URL from Settings. Must happen before
+# engine_from_config picks up the [alembic] section.
+config.set_main_option("sqlalchemy.url", settings.database_url)
+
+target_metadata = Base.metadata
+
+
+def _is_sqlite_url(url: str) -> bool:
+    return url.startswith("sqlite")
+
+
+def _do_run_migrations(connection: Connection) -> None:
+    context.configure(
+        connection=connection,
+        target_metadata=target_metadata,
+        # CHECK / column-type / default diffs all matter for the FL schema.
+        compare_type=True,
+        compare_server_default=True,
+        # Batch mode is required on SQLite to alter CHECK / FK constraints,
+        # but emits unnecessary table-rebuild ops on HANA — gate on dialect.
+        render_as_batch=_is_sqlite_url(settings.database_url),
+    )
+    with context.begin_transaction():
+        context.run_migrations()
+
+
+def run_migrations_offline() -> None:
+    """Run migrations in offline mode (no DB connection — emits SQL)."""
+    context.configure(
+        url=settings.database_url,
+        target_metadata=target_metadata,
+        literal_binds=True,
+        dialect_opts={"paramstyle": "named"},
+        compare_type=True,
+        compare_server_default=True,
+        render_as_batch=_is_sqlite_url(settings.database_url),
+    )
+    with context.begin_transaction():
+        context.run_migrations()
+
+
+async def _run_async_migrations() -> None:
+    connectable = async_engine_from_config(
+        config.get_section(config.config_ini_section, {}),
+        prefix="sqlalchemy.",
+    )
+    async with connectable.connect() as connection:
+        await connection.run_sync(_do_run_migrations)
+    await connectable.dispose()
+
+
+def run_migrations_online() -> None:
+    """Run migrations in online mode.
+
+    Tests inject an existing connection via ``config.attributes['connection']``
+    so the migration applies to the same in-memory SQLite the test will use.
+    Production has no such attribute and falls through to the async engine
+    path that resolves ``DATABASE_URL`` from Settings.
+    """
+    existing_connection = config.attributes.get("connection")
+    if existing_connection is not None:
+        _do_run_migrations(existing_connection)
+        return
+    asyncio.run(_run_async_migrations())
+
+
+if context.is_offline_mode():
+    run_migrations_offline()
+else:
+    run_migrations_online()

+
+Validates IAS-issued JWTs against the tenant JWKS, returning a `Principal`
+identifying the caller. Acts as defense-in-depth alongside the Kyma APIRule
+which performs front-line validation at the ingress.
+
+Local dev bypass: set `AUTH_DISABLED=true` to skip JWT checks. In that mode
+`verify_token` still returns a `Principal` — synthetic, with `sub="local-dev"`
+— so route handlers can rely on a non-None type and audit logs distinguish
+local-dev calls from real-but-anonymous tokens (CR-05).
+"""

+import logging
+
+from fastapi import Depends, HTTPException, status
+from fastapi.security import HTTPAuthorizationCredentials, HTTPBearer
+from jose import JWTError, jwt  # type: ignore
+from jose.exceptions import ExpiredSignatureError  # type: ignore
+
+from fl_control_plane.auth_models import (
+    AuthNotConfiguredError,
+    JwksUnavailableError,
+    Principal,
+    TokenExpiredError,
+    TokenInvalidError,
+)
+from fl_control_plane.config import settings
+from fl_control_plane.jwks_client import JwksClient
+
+logger = logging.getLogger(__name__)
+
+security = HTTPBearer(auto_error=False)
+
+# Synthetic identity returned when AUTH_DISABLED=true. Distinguishable from
+# any real IAS-issued Principal because IAS subjects are GUIDs, not the
+# literal string "local-dev". Audit logs that record `client_id="local-dev"`
+# make local-dev traffic obvious.
+_LOCAL_DEV_PRINCIPAL = Principal(
+    sub="local-dev",
+    client_id="local-dev",
+    scopes=[],
+)

+

+def _get_jwks_client() -> JwksClient:
+    global _jwks_client
+    if not settings.ias_jwks_uri:
+        raise AuthNotConfiguredError("IAS_JWKS_URI not set")
+    if _jwks_client is None or _jwks_client.jwks_uri != settings.ias_jwks_uri:
+        _jwks_client = JwksClient(settings.ias_jwks_uri)
+    return _jwks_client
+
+
+def reset_jwks_client() -> None:
+    """Clear the cached JWKS client. Intended for test teardown."""
+    global _jwks_client
+    _jwks_client = None
+

+async def verify_token(
+    credentials: HTTPAuthorizationCredentials | None = Depends(security),
+) -> Principal:
+    """Validate the bearer token and return the caller's Principal.
+
+    Always returns a `Principal` (non-Optional) so route handlers can rely on
+    `principal.client_id` without defensive None checks. When
+    AUTH_DISABLED=true returns a synthetic local-dev Principal; otherwise
+    validates the bearer token against IAS JWKS and constructs a Principal
+    from the verified claims.
+
+    Raises:
+        HTTPException: 401 on missing or invalid token; 503 when IAS settings
+            are missing or the JWKS endpoint is unreachable. Fail-closed so a
+            misconfigured deploy can't silently open the API.
+    """
+    if settings.auth_disabled:
+        return _LOCAL_DEV_PRINCIPAL
+
+    if not credentials:
+        raise HTTPException(
+            status_code=status.HTTP_401_UNAUTHORIZED,
+            detail="Missing authentication token",
+            headers={"WWW-Authenticate": "Bearer"},
+        )
+
+    try:
+        return await _verify_and_extract(credentials.credentials)
+    except (TokenInvalidError, TokenExpiredError) as e:
+        # Suppress chain (`from None`) so the response body never echoes
+        # JWT internals (e.g. ExpiredSignatureError messages include
+        # timestamps). The original is logged for diagnostics.
+        logger.info("token rejected: %s", e)
+        raise HTTPException(
+            status_code=status.HTTP_401_UNAUTHORIZED,
+            detail="Invalid or expired token",
+            headers={"WWW-Authenticate": "Bearer"},
+        ) from None
+    except (AuthNotConfiguredError, JwksUnavailableError) as e:
+        logger.error("auth backend unavailable: %s", e)
+        raise HTTPException(
+            status_code=status.HTTP_503_SERVICE_UNAVAILABLE,
+            detail="Authentication temporarily unavailable",
+        ) from None
+
+

+    if not token:
+        raise TokenInvalidError("empty token")
+    if not settings.ias_issuer_url or not settings.ias_audience:
+        raise AuthNotConfiguredError("IAS_ISSUER_URL or IAS_AUDIENCE not set")
+
+    # python-jose requires the unverified header to look up the signing key
+    # by `kid` before signature verification can run.
+    try:
+        header = jwt.get_unverified_header(token)
+    except JWTError as e:
+        raise TokenInvalidError(f"malformed token header: {e}") from e
+
+    kid = header.get("kid")
+    if not kid:
+        raise TokenInvalidError("token header missing 'kid'")
+
+    jwks_client = _get_jwks_client()
+    key = await jwks_client.get_key(kid)
+

+        claims = jwt.decode(
+            token,
+            key,
+            algorithms=["RS256"],
+            issuer=settings.ias_issuer_url,
+            audience=settings.ias_audience,
+        )
+    except ExpiredSignatureError as e:
+        raise TokenExpiredError(str(e)) from e
+    except JWTError as e:
+        raise TokenInvalidError(str(e)) from e
+
+    sub = claims.get("sub")
+    if not sub:
+        raise TokenInvalidError("token missing 'sub' claim")
+

+    # IdPs return a list — handle both shapes defensively.
+    scope_claim = claims.get("scope", "")
+    if isinstance(scope_claim, str):
+        scopes = scope_claim.split()
+    elif isinstance(scope_claim, list):
+        scopes = [str(s) for s in scope_claim]
+    else:
+        scopes = []
+
+    return Principal(
+        sub=sub,
+        client_id=claims.get("client_id"),
+        scopes=scopes,
+    )

+"""Validate and normalize CI pipeline URLs at the API boundary.
+
+Each supported CI system implements `PipelineUrlValidator`. The module-level
+`VALIDATORS` list is iterated in order; the first validator whose `matches()`

+system: implement the protocol and append an instance to VALIDATORS.
+"""

+from typing import Protocol
+
+from pydantic import HttpUrl
+
+
+class PipelineUrlValidator(Protocol):

+        either return a normalized HttpUrl or raise ValueError.

+

+    _PATTERN = re.compile(r"^https?://[^/]+(/job/[^/]+){2,}/\d+/$")

+    def matches(self, raw: str) -> bool:
+        return "/job/" in raw

+        # Jenkins multibranch UI inserts /view/change-requests/ between project
+        # and branch segments for PR builds — strip so the canonical
+        # /job/<project>/job/<branch>/<build>/ shape matches.
+        normalized = raw.replace("/view/change-requests/", "/")

+                "https://<host>/job/<project>/job/<branch>/<build_number>/. "

+    """Dispatch `url` to the first matching CI validator.
+

+            HttpUrl by Pydantic.
+

+        The same URL after CI-specific normalization (e.g. trailing slash,
+        Jenkins UI segment stripping).
+
+    Raises:
+        ValueError: No registered validator claims this URL shape.
+    """
+    raw = str(url)
+    for validator in VALIDATORS:
+        if validator.matches(raw):

+    supported = ", ".join(v.name for v in VALIDATORS)
+    raise ValueError(
+        f"pipeline_url does not match any supported CI system ({supported}). Got: {raw}"
+    )

+"""JWKS fetch + cache client for IAS-issued JWT verification.
+
+IAS publishes its signing keys at a JWKS URI (typically
+`${issuer}/oauth2/certs`). Keys rotate. Two refresh triggers:
+
+  1. TTL expiry — refresh on first request after the cache window.
+  2. Unknown kid — token references a key not in cache → refresh once and
+     retry. Prevents a key rotation from causing a window of 401s while the
+     cache is still warm with old keys.
+"""
+
+import time
+from typing import Any
+
+import httpx
+
+from fl_control_plane.auth_models import JwksUnavailableError
+
+_CACHE_TTL_SECONDS = 600  # 10 minutes
+_HTTP_TIMEOUT_SECONDS = 5.0
+
+
+class JwksClient:
+    """Fetches and caches JWKS from a single issuer.
+
+    Thread-safety: not safe for cross-thread use. FastAPI's async event loop
+    is single-threaded so this is fine. If we ever switch to a worker pool
+    that shares this client, add an asyncio.Lock around `_refresh`.
+    """
+
+    def __init__(self, jwks_uri: str, ttl_seconds: int = _CACHE_TTL_SECONDS) -> None:
+        self._jwks_uri = jwks_uri
+        self._ttl_seconds = ttl_seconds
+        self._keys_by_kid: dict[str, dict[str, Any]] = {}
+        self._fetched_at: float = 0.0
+
+    @property
+    def jwks_uri(self) -> str:
+        return self._jwks_uri
+
+    async def get_key(self, kid: str) -> dict[str, Any]:
+        """Return the JWK with matching `kid`. Refreshes on miss.
+
+        Raises:
+            JwksUnavailableError: JWKS endpoint unreachable, or `kid` still
+                not present after a forced refresh.
+        """
+        if self._is_stale() or kid not in self._keys_by_kid:
+            await self._refresh()
+        key = self._keys_by_kid.get(kid)
+        if key is None:
+            raise JwksUnavailableError(f"Unknown JWT signing key id: {kid}")
+        return key
+
+    def _is_stale(self) -> bool:
+        return (time.monotonic() - self._fetched_at) > self._ttl_seconds
+
+    async def _refresh(self) -> None:
+        try:
+            async with httpx.AsyncClient(timeout=_HTTP_TIMEOUT_SECONDS) as client:
+                response = await client.get(self._jwks_uri)
+                response.raise_for_status()
+                payload = response.json()
+        except httpx.HTTPError as e:
+            raise JwksUnavailableError(f"JWKS fetch failed: {e}") from e
+        except ValueError as e:
+            # response.json() raises json.JSONDecodeError (a ValueError subclass)
+            # on malformed JSON. Surface as JwksUnavailableError so the auth
+            # path maps it to 503 instead of leaking a 500.
+            raise JwksUnavailableError(f"JWKS response is not valid JSON: {e}") from e
+
+        keys = payload.get("keys")
+        if not isinstance(keys, list):
+            raise JwksUnavailableError("JWKS response missing 'keys' array")
+
+        self._keys_by_kid = {k["kid"]: k for k in keys if "kid" in k}
+        self._fetched_at = time.monotonic()
