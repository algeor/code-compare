+def _extract_json_from_llm_response(text: str) -> str:
+    """Extract JSON object from LLM response, handling markdown code blocks."""
+    start = text.find("{")
+    end = text.rfind("}")
+    if start == -1 or end == -1 or end < start:
+        return text
+    return text[start: end + 1]
+
+
+def _has_json_structure(text: str) -> bool:
+    """Return True if text contains matching braces that could be a JSON object."""
+    start = text.find("{")
+    end = text.rfind("}")
+    return start != -1 and end != -1 and end > start
+
+
+def _extract_and_parse_json(
+    fl_result: str,
+    result_filter: Callable[[str], str] | None = None,
+) -> dict[str, Any]:
+    """Extract and parse JSON from an LLM response string.
+
+    Raises:
+        ValueError: If no JSON structure is found.
+        json.JSONDecodeError: If extracted text is not valid JSON.
+    """
+    if result_filter:
+        fl_result = result_filter(fl_result)
+    else:
+        if not _has_json_structure(fl_result):
+            logger.warning("Response contains no JSON structure: %s", fl_result[:200])
+            raise ValueError(f"Response does not contain valid JSON: {fl_result[:200]}")
+        fl_result = _extract_json_from_llm_response(fl_result)
+
+    parsed: dict[str, Any] = json.loads(fl_result, strict=False)
+    return parsed
+

+def _get_plain_text_refusal_message(response_text: str) -> str | None:
+    """Return refusal message if the plain-text response contains refusal keywords."""
+    lower = response_text.lower()
+    if any(kw in lower for kw in REFUSAL_KEYWORDS) and not _has_json_structure(response_text):
+        return response_text[:MAX_REFUSAL_MESSAGE_LENGTH]
+    return None

+def _get_structured_refusal_message(result_dict: dict[str, Any]) -> str | None:
+    """Return refusal message if the parsed JSON response signals a refusal."""
+    if result_dict.get("refusal"):
+        return str(result_dict["refusal"])
+    finish_reason = result_dict.get("finish_reason", "")
+    if finish_reason in ("content_filter", "content_filtered"):
+        return f"Content filtered (finish_reason: {finish_reason})"
+    if result_dict.get("stopReason") == "content_filtered":
+        return "Content filtered by AWS Bedrock"
+    if "error" in result_dict:
+        error_msg = str(result_dict["error"])
+        if any(kw in error_msg.lower() for kw in REFUSAL_KEYWORDS):
+            return error_msg
+    return None

+def _send_chat_to_aws(
+    model_name: str,
+    messages: list[dict[str, str]],
+) -> dict[str, Any]:
+    """Send a chat request to AWS Bedrock with retries.
+
+    Returns:
+        Dict with keys "response_text" and "usage".
+
+    Raises:
+        FaultLocalizationLLMException: After all retries are exhausted.
+    """
+    system_prompt = []
+    conversations = []
+    for message in messages:
+        if message["role"] == "system":
+            system_prompt = [{"text": message["content"]}]
+        else:
+            conversations.append({"role": message["role"], "content": [{"text": message["content"]}]})
+
+    for attempt in range(RETRY_LLM_CHAT + 1):
+        response = None
+        try:
+            response = _get_bedrock_client(model_name).converse(
+                system=system_prompt,
+                messages=conversations,
+                inferenceConfig={"maxTokens": 5000, "temperature": 0},
+            )
+            return {
+                "response_text": str(response["output"]["message"]["content"][0]["text"]),
+                "usage": {
+                    "model_name": model_name,
+                    "prompt_tokens": response["usage"]["inputTokens"],
+                    "completion_tokens": response["usage"]["outputTokens"],
+                    "total_tokens": response["usage"]["totalTokens"],
+                },
+            }
+        except Exception as exc:  # noqa: BLE001
+            logger.exception(
+                "AWS Bedrock request failed (attempt %d). Response: %s. Error: %s",
+                attempt + 1, response, exc,
+            )
+            if attempt == RETRY_LLM_CHAT:
+                raise FaultLocalizationLLMException(
+                    "Failed to get response from SAP AI Hub after all retries"
+                ) from exc
+        logger.info("Retrying AWS Bedrock call (attempt %d)...", attempt + 2)
+        time.sleep(30)
+
+    raise AssertionError("Unreachable: _send_chat_to_aws loop invariant violated")

+    messages: list[dict[str, str]],

+    """Send a chat request to OpenAI with retries.

+    Returns:

+    Raises:

+def _inline_defs(schema: dict[str, Any]) -> None:
+    """Inline $defs references in-place to avoid dangling $ref pointers."""
+    defs = schema.get("$defs", {})
+    if not defs:
+        return
+
+    def replace_refs(obj: Any) -> Any:
+        if isinstance(obj, list):
+            return [replace_refs(item) for item in obj]
+        if not isinstance(obj, dict):
+            return obj
+        if "$ref" in obj:
+            ref_path = obj["$ref"]
+            if ref_path.startswith("#/$defs/"):
+                def_name = ref_path.replace("#/$defs/", "")
+                if def_name in defs:
+                    resolved = defs[def_name].copy()
+                    for key, value in obj.items():
+                        if key != "$ref" and key not in resolved:
+                            resolved[key] = value
+                    return resolved
+            return obj
+        return {key: replace_refs(value) for key, value in obj.items()}
+
+    for key in list(schema.keys()):
+        if key != "$defs":
+            schema[key] = replace_refs(schema[key])
+    schema.pop("$defs", None)
+
+
+def _inject_schema_into_messages(
+    messages: list[dict[str, str]],
+    response_model: type[BaseModel],
+) -> list[dict[str, str]]:
+    """Append JSON schema instruction to the system message."""
+    schema = response_model.model_json_schema()
+    schema.pop("title", None)
+    if "$defs" in schema:
+        _inline_defs(schema)
+
+    schema_instruction = (
+        "\n\n**CRITICAL: Output Format Requirements**\n"
+        "1. Respond with ONLY a valid JSON object\n"
+        "2. Do NOT include markdown code blocks, explanations, or other text\n"
+        "3. The JSON must match this exact schema:\n\n"
+        f"{json.dumps(schema, indent=2)}\n\n"
+        '4. Example format: {"field1": "value", "field2": 123}\n'
+        "5. Ensure all required fields are present and types match exactly"
+    )
+
+    messages_copy = [m.copy() for m in messages]
+    for msg in messages_copy:
+        if msg["role"] == "system":
+            msg["content"] += schema_instruction
+            return messages_copy
+
+    messages_copy.insert(0, {"role": "system", "content": schema_instruction.strip()})
+    return messages_copy

+def _format_validation_feedback(
+    raw_response: str,
+    validation_error: ValidationError,
+    response_model: type[BaseModel],
+) -> str:
+    """Format Pydantic validation errors into a correction prompt for the LLM."""
+    error_lines = [
+        f"  - Field '{' -> '.join(str(x) for x in e['loc'])}': {e['msg']}"
+        for e in validation_error.errors()
+    ]
+    truncated = raw_response[:MAX_VALIDATION_RESPONSE_LENGTH] + (
+        "..." if len(raw_response) > MAX_VALIDATION_RESPONSE_LENGTH else ""
+    )
+    return (
+        "Your previous response did not match the required JSON schema.\n\n"
+        f"**Your response:**\n```\n{truncated}\n```\n\n"
+        "**Validation errors:**\n" + "\n".join(error_lines) + "\n\n"
+        f"**Required fields:** {list(response_model.model_fields.keys())}\n\n"
+        "Please provide a corrected JSON response matching the schema exactly."
+    )

+def sap_ai_hub_chat(
+    messages: list[dict[str, str]],
+    result_filter: Callable[[str], str] | None = None,
+) -> dict[str, Any]:
+    """Send a chat request to SAP AI Hub and return parsed JSON plus usage info.
+
+    Routes to AWS Bedrock for Claude/Amazon models, OpenAI otherwise.
+    Handles refusal detection and JSON extraction.
+
+    Returns:
+        Parsed JSON dict with an added "usage" key.
+
+    Raises:
+        FaultLocalizationRefusalException: LLM refused to respond.
+        FaultLocalizationLLMException: API/network error after retries.
+        ValueError: Response contains no valid JSON.
+        json.JSONDecodeError: Response is not valid JSON.
+    """
+    model_name = settings.sap_ai_hub_model_name
+    logger.info("SAP AI Hub request using model: %s", model_name)
+
+    if "claude" in model_name or "amazon" in model_name:
+        raw = _send_chat_to_aws(model_name, messages)
+    else:
+        raw = _send_chat_to_openai(model_name, messages)
+
+    response_text = raw["response_text"]
+    usage = raw["usage"]
+
+    plain_refusal = _get_plain_text_refusal_message(response_text)
+    if plain_refusal:
+        logger.warning("LLM plain-text refusal: %s", plain_refusal)
+        raise FaultLocalizationRefusalException(
+            "LLM declined to generate response", plain_refusal, usage
+        )
+
+    try:
+        result = _extract_and_parse_json(response_text, result_filter)
+    except (ValueError, json.JSONDecodeError) as exc:
+        logger.error("Failed to parse JSON from response: %s. Raw: %s", exc, response_text[:200])
+        raise FaultLocalizationLLMException(
+            f"Failed to parse LLM response as JSON: {exc}", usage
+        ) from exc
+
+    structured_refusal = _get_structured_refusal_message(result)
+    if structured_refusal:
+        logger.warning("LLM structured refusal: %s", structured_refusal)
+        raise FaultLocalizationRefusalException(
+            "LLM declined to generate response", structured_refusal, usage
+        )
+
+    result["usage"] = usage
+    logger.info("SAP AI Hub request complete.")
+    return result
+

+def sap_ai_hub_chat_structured(
+    messages: list[dict[str, str]],
+    response_model: type[T],
+    *,
+    include_schema_in_prompt: bool = True,
+    validation_retries: int = VALIDATION_RETRIES,
+    result_filter: Callable[[str], str] | None = None,
+) -> tuple[T, dict[str, Any]]:
+    """Send a chat request and validate the response against a Pydantic model.
+
+    Retries with validation feedback when the response does not match the schema.
+
+    Args:
+        messages: Chat messages (system + user).
+        response_model: Pydantic model the response must conform to.
+        include_schema_in_prompt: Inject JSON schema into the system prompt.
+        validation_retries: Max retries on validation failure.
+        result_filter: Optional preprocessor for the raw response string.
+
+    Returns:
+        Tuple of (validated model instance, usage info dict).
+
+    Raises:
+        FaultLocalizationRefusalException: LLM refused to respond.
+        FaultLocalizationValidationException: Schema mismatch after all retries.
+        FaultLocalizationLLMException: API/network error.
+    """
+    working_messages = (
+        _inject_schema_into_messages(messages, response_model)
+        if include_schema_in_prompt
+        else [m.copy() for m in messages]
+    )
+
+    for attempt in range(validation_retries + 1):

+async def _request_with_retry(
+    client: httpx.AsyncClient,
+    url: str,
+    *,
+    headers: dict | None = None,
+    timeout: float = 30,
+) -> httpx.Response:
+    """GET url, retrying on 429/503 with exponential backoff or Retry-After delay."""
+    response: httpx.Response | None = None
+    for attempt in range(_MAX_RETRIES + 1):
+        response = await client.get(url, headers=headers, timeout=timeout)
+        if response.status_code not in _RETRYABLE_STATUS_CODES or attempt == _MAX_RETRIES:
+            return response
+        retry_after = response.headers.get("Retry-After", "")
+        delay = float(retry_after) if retry_after.isdigit() else _BACKOFF_BASE_SECONDS * (2**attempt)
+        logger.warning(
+            "Jenkins returned %s for %s — retrying in %.1fs (attempt %d/%d)",
+            response.status_code, url, delay, attempt + 1, _MAX_RETRIES,
+        )
+        await asyncio.sleep(delay)
+    return response  # type: ignore[return-value]  # loop always assigns before here

+class _CheckResult:
+    """Internal structured outcome of the Jenkins permission check."""
+
+    def __init__(self) -> None:

+async def _stage1_authenticated(pipeline_url: str, username: str, token: str) -> _CheckResult:
+    """GET pipeline_url with Basic auth and classify the response.
+
+    Returns a _CheckResult with stop=False on 200 OK, stop=True on all other
+    outcomes. On 404 the result is left open (auth_check='not_found', stop=False)
+    so the caller can proceed to the discover check.
+    """
+    result = _CheckResult()
+    headers = {"Authorization": _basic_auth_header(username, token)}
+
+    try:
+        async with httpx.AsyncClient(verify=settings.ssl_ca_bundle) as client:
+            resp = await _request_with_retry(client, pipeline_url, headers=headers)
+        result.reachable = True
+        result.http_status = resp.status_code
+
+        if resp.is_success:
+            result.auth_check = "ok"
+            result.guidance = (
+                "FL can reach Jenkins and the credentials are accepted. "
+                "This gate did not block the mission. No action needed."
+            )
+            return result
+
+        if resp.status_code == 401:
+            result.auth_check = "expired_credentials"
+            result.stop = True
+            result.require_finalizer = False
+            result.stop_message = _MSG_UNREACHABLE
+            result.guidance = _GUIDANCE_EXPIRED_CREDENTIALS
+            return result
+
+        if resp.status_code == 403:
+            result.auth_check = "forbidden"
+            result.stop = True
+            result.require_finalizer = True
+            result.stop_message = _MSG_NO_PERMISSION
+            result.guidance = _GUIDANCE_PERMISSION_TEMPLATE.format(
+                permissions="Overall Read, Job Read"
+            )
+            return result
+
+        if resp.status_code != 404:
+            result.auth_check = f"http_{resp.status_code}"
+            result.stop = True
+            result.require_finalizer = False
+            result.stop_message = _MSG_UNREACHABLE
+            result.guidance = _GUIDANCE_INFRASTRUCTURE
+            return result
+
+        # 404 — caller proceeds to discover check
+        result.auth_check = "not_found"
+        return result
+
+    except (httpx.ConnectError, httpx.TimeoutException) as exc:
+        result.reachable = False
+        result.auth_check = f"network_error: {exc}"
+        result.stop = True
+        result.require_finalizer = False
+        result.stop_message = _MSG_UNREACHABLE
+        result.guidance = _GUIDANCE_INFRASTRUCTURE
+        return result

+async def _stage2_discover(pipeline_url: str, headers: dict) -> _CheckResult:
+    """GET /api/json?tree=jobs[name] to distinguish missing Job:Discover from genuine 404.
+
+    A 200 OK with an empty jobs list most probably means Job:Discover is missing —
+    on any real Jenkins instance a user with Job:Discover would see at least one job.
+    A non-empty list means Overall:Read and Job:Discover are present and the pipeline
+    URL itself is the problem (build deleted, job moved, etc.).
+    """
+    result = _CheckResult()
+    result.discover_check_performed = True
+
+    domain = _extract_domain(pipeline_url)
+    if not domain:
+        result.discover_check_result = "domain_parse_failed"
+        result.stop = True
+        result.require_finalizer = False
+        result.stop_message = _MSG_INVALID_URL
+        result.guidance = _GUIDANCE_INFRASTRUCTURE
+        return result
+
+    try:
+        async with httpx.AsyncClient(verify=settings.ssl_ca_bundle) as client:
+            discover_resp = await _request_with_retry(
+                client,
+                f"{domain}/api/json?tree=jobs[name]",
+                headers=headers,
+            )
+        discover_resp.raise_for_status()
+        jobs = discover_resp.json().get("jobs", [])
+
+        if len(jobs) == 0:
+            result.discover_check_result = "empty_job_list"
+            result.stop = True
+            result.require_finalizer = True
+            result.stop_message = _MSG_NO_DISCOVER
+            result.guidance = _GUIDANCE_PERMISSION_TEMPLATE.format(
+                permissions="Overall Read, Job Read, Job Discover"
+            )
+        else:
+            result.discover_check_result = f"ok_{len(jobs)}_jobs_visible"
+            result.stop = True
+            result.require_finalizer = False
+            result.stop_message = _MSG_BUILD_NOT_FOUND
+            result.guidance = _GUIDANCE_INFRASTRUCTURE
+
+    except Exception as exc:  # noqa: BLE001
+        result.discover_check_result = f"check_failed:{exc}"
+        result.stop = True
+        result.require_finalizer = False
+        result.stop_message = _MSG_UNREACHABLE
+        result.guidance = _GUIDANCE_INFRASTRUCTURE
+
+    return result

+class JenkinsConnectivityGate:
+    """Verifies FL can reach and authenticate against the Jenkins instance.
+
+    Three outcome buckets:
+    - Permission denied (403 or 404+no-discover): STOP, require_finalizer=True.
+      The finalizer posts a PR comment explaining FL and what permissions are needed.
+    - Credential issue (401, or missing credential(s) with no anonymous access):
+      STOP, require_finalizer=False. No PR comment — the FL platform team resolves
+      these internally using the clinefl script.
+    - Infrastructure failure (unreachable, timeout, other errors): STOP,
+      require_finalizer=False. No PR comment — logged as an operational alert.
+
+    Special case: missing credential(s) but anonymous access succeeds → CONTINUE.
+    The HDLF findings document that FL is running without credentials.
+
+    Credentials are fetched from Vault via get_jenkins_credentials().
+    findings always records the specific auth_check value for debugging.
+    """
+
+    gate_id = _GATE_ID
+
+    schema = """\
+# jenkins_connectivity gate result schema
+
+## File
+`pipeline_analyzer/jenkins_connectivity.json`
+
+## Structure
+```json
+{
+  "gate": "jenkins_connectivity",
+  "analysis_decision": "<string>",
+  "require_finalizer": "<bool>",
+  "message": "<string | null>",
+  "findings": {
+    "url_checked": "<string | null>",
+    "reachable": "<bool | null>",
+    "http_status": "<int | null>",
+    "auth_check": "<string>",
+    "discover_check_performed": "<bool>",
+    "discover_check_result": "<string>",
+    "missing_vault_fields": ["<string>"],
+    "guidance": "<string>"
+  }
+}
+```
+
+## `analysis_decision`
+- `"STOP"` — FL cannot proceed with this inspection.
+- `"CONTINUE"` — Jenkins is reachable and credentials are accepted (or anonymous access succeeded).
+
+## `require_finalizer`
+- `true` — Jenkins returned 403 (permission denied) or 404 with empty job list (Job Discover missing).
+  The finalizer posts a PR comment asking the developer to grant the FL technical user access.
+- `false` — All other outcomes. No PR comment is posted.
+
+## `auth_check` values
+
+| Value | Meaning |
+|---|---|
+| `"ok"` | Credentials accepted, 200 OK |
+| `"anonymous"` | No credentials in Vault; anonymous access succeeded |
+| `"forbidden"` | 403 — Overall:Read or Job:Read missing |
+| `"not_found"` | 404 with jobs visible — build deleted or URL wrong |
+| `"expired_credentials"` | 401 — credentials most probably stale or rotated; update via clinefl |
+| `"missing_credentials_no_anonymous_access"` | No credential(s) in Vault and anonymous attempt failed |
+| `"empty_job_list"` | 404 + empty job list — Job:Discover most probably missing |
+| `"connection_error"` | Network unreachable |
+| `"timeout"` | Request timed out |
+| `"network_error: <detail>"` | ConnectionError or Timeout with exception detail |
+| `"http_<N>"` | Unexpected HTTP status N |
+| `"url_missing"` | No `pipeline_url` in mission context |
+| `"credential_fetch_failed"` | Vault lookup raised an exception |
+| `"domain_parse_failed"` | Could not extract domain from the pipeline URL |
+| `"check_failed:<detail>"` | Stage 2 discover request raised an exception |
+
+## `discover_check_performed`
+`true` only when the pipeline URL returned 404 and a secondary check of
+`GET {domain}/api/json?tree=jobs[name]` was performed to distinguish
+missing Job:Discover permission from a genuine missing build.
+
+## `discover_check_result`
+Only meaningful when `discover_check_performed` is `true`.
+
+| Value | Meaning |
+|---|---|
+| `"empty_job_list"` | `/api/json` returned 200 with zero jobs — Job:Discover most probably missing |
+| `"ok_<N>_jobs_visible"` | `/api/json` returned N jobs — genuine 404, build not found |
+| `"domain_parse_failed"` | Could not extract domain from the pipeline URL |
+| `"check_failed:<detail>"` | Discover request raised an exception |
+
+## `missing_vault_fields`
+List of credential fields absent from the Vault secret. Only present when
+`auth_check` is `"anonymous"` or `"missing_credentials_no_anonymous_access"`.
+Possible values: `"username"`, `"token"`.
+
+## `guidance`
+Free-text instructions for the LLM finalizer: what the finding means,
+whether to post a PR comment, and what action to take. Not intended for
+end users — contains internal routing information.
+"""
+
+    async def evaluate(
+        self,
+        context: GateEvaluationContext,
+        _config: GateConfig,
+    ) -> GateVerdict:
+        """Return STOP if Jenkins is unreachable or credentials are rejected."""
+        if not context.pipeline_url:
+            logger.warning(
+                "no pipeline_url for inspection %s",
+                context.inspection_id,
+            )
+            return GateVerdict(
+                gate_id=self.gate_id,
+                analysis_decision="STOP",
+                require_finalizer=False,
+                message=_MSG_URL_MISSING,
+                findings={
+                    "url_checked": None,
+                    "auth_check": "url_missing",
+                    "guidance": _GUIDANCE_INFRASTRUCTURE,
+                },
+            )
+
+        try:
+            username, token = _get_credentials(context.pipeline_url)
+        except Exception as exc:  # noqa: BLE001
+            logger.error(
+                "credential fetch failed for inspection %s: %s",
+                context.inspection_id, exc,
+            )
+            return GateVerdict(
+                gate_id=self.gate_id,
+                analysis_decision="STOP",
+                require_finalizer=False,
+                message=_MSG_UNREACHABLE,
+                findings={
+                    "url_checked": context.pipeline_url,
+                    "auth_check": "credential_fetch_failed",
+                    "credential_error": str(exc),
+                    "guidance": _GUIDANCE_INFRASTRUCTURE,
+                },
+            )
+
+        if not username or not token:
+            missing = []
+            if not username:
+                missing.append("username")
+            if not token:
+                missing.append("token")
+            logger.warning(
+                "Vault secret for inspection %s is missing: %s — trying anonymous access",
+                context.inspection_id,
+                ", ".join(missing),
+            )
+            check = await _try_anonymous(context.pipeline_url)
+            if check.auth_check == "anonymous":
+                logger.info(
+                    "anonymous access succeeded for inspection %s — proceeding without credentials",
+                    context.inspection_id,
+                )
+            else:
+                logger.warning(
+                    "anonymous access failed for inspection %s (%s) — "
+                    "no credentials and no anonymous access",
+                    context.inspection_id,
+                    check.auth_check,
+                )
+            findings = {
+                "url_checked": context.pipeline_url,
+                "reachable": check.reachable,
+                "http_status": check.http_status,
+                "auth_check": check.auth_check,
+                "missing_vault_fields": missing,
+                "guidance": check.guidance,
+            }
+            if check.stop:
+                return GateVerdict(
+                    gate_id=self.gate_id,
+                    analysis_decision="STOP",
+                    require_finalizer=False,
+                    message=check.stop_message,
+                    findings=findings,
+                )
+            return GateVerdict(
+                gate_id=self.gate_id,
+                analysis_decision="CONTINUE",
+                require_finalizer=False,
+                message=None,
+                findings=findings,
+            )
+
+        check = await _check_jenkins_permissions(context.pipeline_url, username, token)
+
+        if check.auth_check == "expired_credentials":
+            logger.warning(
+                "STOP for inspection %s — credentials most probably expired; "
+                "update using clinefl script",
+                context.inspection_id,
+            )
+        elif check.stop:
+            logger.warning(
+                "STOP for inspection %s — %s",
+                context.inspection_id, check.stop_message,

+class WipDetector:
+    """Detects if a pull request is Work-In-Progress.
+
+    Uses a priority-ordered rule set. For high-confidence rule results,
+    returns immediately. For medium/low confidence, optionally escalates
+    to an AI review.
+    """
+
+    def __init__(
+        self,
+        rules: list[WipRule] | None = None,
+        enable_ai_review: bool = True,
+    ) -> None:
+        self.rules = sorted(
+            rules if rules is not None else WIP_DETECTION_RULES,
+            key=lambda r: r.priority,
+        )
+        self.enable_ai_review = enable_ai_review
+
+    def _check_rules(self, pr_data: PRData) -> WipDetectionResult:
+        logger.info("Running rule-based WIP detection")
+        for rule in self.rules:
+            logger.debug("Checking rule: %s", rule.name)
+            try:
+                if rule.check_function(pr_data):
+                    logger.info("WIP detected by rule: %s", rule.name)
+                    return WipDetectionResult(
+                        is_wip=True,
+                        reason=rule.message,
+                        matched_rule=rule.name,
+                        confidence=rule.confidence,
+                        explanation=rule.explanation,
+                        fix_suggestion=rule.fix_suggestion,
+                        ai_enhanced=False,
+                    )
+            except Exception as err:  # noqa: BLE001
+                logger.warning("Rule %s failed: %s. Continuing.", rule.name, err)
+
+        logger.info("No WIP indicators found — PR appears ready for review")
+        return WipDetectionResult(
+            is_wip=False,
+            reason="No WIP indicators detected",
+            matched_rule=None,
+            confidence=ConfidenceLevel.HIGH,
+            explanation="No work-in-progress indicators were found in the PR metadata.",
+            fix_suggestion="",
+            ai_enhanced=False,
+        )
+
+    async def _ai_wip_review(
+        self,
+        pr_data: PRData,
+        rule_result: WipDetectionResult,
+    ) -> WipDetectionResult:
+        """Escalate ambiguous cases to AI review.
+
+        Raises:
+            WipDetectionException: If the AI call fails for any reason.
+                The caller falls back to the rule-based result.
+        """
+        logger.info("Starting AI review for medium/low confidence WIP case")
+        try:
+            user_prompt = build_user_prompt(pr_data, rule_result)
+            messages = [
+                {"role": "system", "content": SYSTEM_PROMPT},
+                {"role": "user", "content": user_prompt},
+            ]
+            response, _ = await asyncio.to_thread(
+                sap_ai_hub_chat_structured, messages, WipDetectionResponse
+            )
+            logger.info("AI review completed: %s", "WIP" if response.is_wip else "Ready")
+            return WipDetectionResult(
+                is_wip=response.is_wip,
+                reason=response.reason,
+                matched_rule=f"ai_review_{rule_result.matched_rule or 'unknown'}",
+                confidence=ConfidenceLevel.HIGH,
+                explanation=response.reason,
+                fix_suggestion=response.actionable_steps,
+                ai_enhanced=True,
+            )
+        except (
+            FaultLocalizationLLMException,
+            FaultLocalizationRefusalException,
+            FaultLocalizationValidationException,
+        ) as exc:
+            logger.error("AI WIP review failed: %s", exc, exc_info=True)
+            raise WipDetectionException(f"AI review failed: {exc}") from exc
+
+    async def scan_pr_data(self, pr_data: PRData) -> WipDetectionResult:
+        """Run rule-based detection, escalate to AI for medium/low confidence.
+
+        Args:
+            pr_data: Pull request metadata from GithubClient.collect_pr_metadata().
+
+        Returns:
+            WipDetectionResult with final decision.
+        """
+        logger.info("Running WIP detection on PR metadata")
+        rule_result = self._check_rules(pr_data)
+
+        if rule_result.confidence == ConfidenceLevel.HIGH:
+            logger.info(
+                "High confidence %s from rules — %s",
+                "WIP" if rule_result.is_wip else "ready",
+                rule_result.matched_rule or "no indicators",
+            )
+            return rule_result
+
+        if self.enable_ai_review:
+            logger.info(
+                "%s confidence from rules — escalating to AI review",
+                rule_result.confidence,
+            )
+            try:
+                return await self._ai_wip_review(pr_data, rule_result)
+            except WipDetectionException as err:
+                logger.warning("AI WIP review failed: %s. Falling back to rule-based result.", err)
+                return rule_result
+
+        logger.info(
+            "AI review disabled — using %s confidence rule-based result",
+            rule_result.confidence,
+        )
+        return rule_result

+def is_gate_active(config: GateConfig, context: GateEvaluationContext) -> bool:
+    """Return True if the gate should run for this mission.
+
+    Evaluation order:
+    1. If any exclude_for entry matches the context → False (always wins).
+    2. If any activate_for entry matches the context → True.
+    3. Otherwise → False (empty activate_for means gate never runs).
+
+    Args:
+        config: Gate configuration defined in analyzer.py.
+        context: Mission metadata for the current mission.
+    """
+    for condition in config.exclude_for:
+        if _condition_matches(condition, context):
+            logger.debug(
+                "resolver: gate %s excluded for inspection %s",
+                config.gate_id,
+                context.inspection_id,
+            )
+            return False
+
+    for entry in config.activate_for:
+        if entry == "all":
+            return True
+        if isinstance(entry, ActivationCondition) and _condition_matches(entry, context):
+            return True
+
+    return False
+
+
+def _condition_matches(condition: ActivationCondition, context: GateEvaluationContext) -> bool:
+    """Return True if all set fields in the condition match the context.
+
+    Fields that are None on the condition are skipped (match unconditionally).
+    All set fields must match — AND logic within one condition entry.
+    """
+    if condition.ci_system is not None:
+        if context.ci_system is None or context.ci_system.lower() != condition.ci_system.lower():
+            return False
+
+    if condition.org is not None:
+        if context.org is None or context.org.lower() != condition.org.lower():
+            return False
+
+    if condition.repo_pattern is not None:
+        repo = context.repo_name or ""
+        if not fnmatch.fnmatch(repo.lower(), condition.repo_pattern.lower()):
+            return False
+
+    if condition.trigger_scope is not None:
+        if condition.trigger_scope == "pr_only" and not context.is_pr:
+            return False
+        if condition.trigger_scope == "non_pr_only" and context.is_pr:
+            return False
+
+    if condition.target_branch_pattern is not None:
+        branch = context.pr_target_branch or ""
+        if not fnmatch.fnmatch(branch.lower(), condition.target_branch_pattern.lower()):
+            return False
+
+    return True

+## Why
+
+The FL mission lifecycle currently has no component that evaluates platform-level policy before committing to a full analysis. Without this, every mission — including WIP pull requests, missions on kill-switched repos, and missions where FL cannot reach the CI system — proceeds to the Data Extractor, paying the full cost of log extraction and handler scheduling before being rejected. The Pipeline Analyzer introduces that decision point cleanly, between the Metadata Extractor and the Data Extractor.
+
+## What Changes
+
+- **NEW** `pipeline_analyzer` module (`fl_control_plane/pipeline_analyzer/`) — evaluates all platform-level gates using mission metadata; each gate writes its own row to the database and its own findings file to HDLF
+- **NEW** `pipeline_analyzer_results` database table — one row per gate per mission, always written regardless of gate verdict
+- **NEW** `pipeline_analyzer/<gate_id>.json` in HDLF — one file per gate per mission, always written regardless of gate verdict
+- **MODIFIED** `inspections` table — adds `status_message` column for human-readable outcome context, set by the external routing component
+- **NEW** `WipDetectionGate` — detects WIP pull requests; verdict: STOP, require_finalizer=True
+- **NEW** `KillSwitchGate` — emergency stop for operator use; verdict: STOP, require_finalizer=False
+- **NEW** `JenkinsConnectivityGate` — verifies FL can reach and authenticate against Jenkins; verdict: STOP, require_finalizer=False (permission denied) or True (credentials rejected)
+
+## Position in the Mission Lifecycle
+
+```
+Ingestion API
+      │ mission_id
+      ▼
+Metadata Extractor ──► HDLF: metadata.json
+      │ mission_id
+      ▼
+Pipeline Analyzer      ◄── this change
+      │
+      │  writes pipeline_analyzer_results rows to DB (one per gate)
+      │  writes pipeline_analyzer/<gate>.json to HDLF (one per gate)
+      │
+      ▼
+External orchestrator reads pipeline_analyzer_results
+and routes the mission (out of scope for this change)
+```
+
+The Pipeline Analyzer receives only a `mission_id`. It reads all inputs from the database and HDLF. All output goes to the database and HDLF. Routing — whether to proceed to the Data Extractor, call the finalizer, or terminate — is handled by an external orchestrator that reads `pipeline_analyzer_results`. That orchestrator is out of scope for this change.
+
+## Decisions
+
+### D1: All active gates always run
+
+Every gate whose activation conditions match the mission runs and writes its verdict. There is no early exit. This gives a complete audit trail — the external orchestrator sees every gate's verdict for every mission.
+
+### D2: Gate verdict is recorded per gate, routing is external
+
+Each gate returns a `GateVerdict` carrying `analysis_decision` (STOP/CONTINUE), `require_finalizer`, `message`, and `findings`. The Pipeline Analyzer records this verdict to the database and HDLF on behalf of the gate. It does not act on the verdict itself. All routing decisions are made externally by reading `pipeline_analyzer_results`.
+
+Adding a new gate requires one new file in `gates/` and registering it with its config in `analyzer.py`. The resolver and the rest of the analyzer loop require no changes.
+
+### D3: Gate activation uses activate_for / exclude_for whitelist model
+
+Each gate's `GateConfig` has two lists, defined in code in `analyzer.py`.
+
+- `activate_for` — whitelist of conditions under which the gate runs. Multiple entries are OR'd. Fields within one entry are AND'd. The string `"all"` as an entry matches every mission unconditionally. Empty list means the gate never runs.
+- `exclude_for` — whitelist of exceptions. Always wins over `activate_for`.
+
+Available fields in a condition entry:
+- `ci_system: str`
+- `org: str`
+- `repo_pattern: str` (glob)
+- `trigger_scope: pr_only | non_pr_only | all`
+- `target_branch_pattern: str` (glob)
+
+### D4: Gate configurations are defined in code, not the database
+
+Gate activation configs (`activate_for`, `exclude_for`) are hardcoded in `analyzer.py` as `GateConfig` instances. There is no `gate_configs` database table. To change when a gate runs, edit the code. This keeps configuration explicit, reviewable via PR, and co-located with the gate registration.
+
+Gate-specific behaviour (e.g. WIP title patterns) is hardcoded as constants inside each gate file — they are implementation details, not configuration.
+
+### D5: Each gate always writes to both DB and HDLF when active
+
+Every active gate writes one row to `pipeline_analyzer_results` and one file to `pipeline_analyzer/<gate_id>.json` in HDLF, regardless of verdict. Gates that are not active (activation conditions do not match) write nothing.
+
+### D6: GateVerdict carries structured findings with LLM guidance
+
+Each gate populates a `findings` dict with structured evidence it collected (e.g. matched rule, HTTP status, auth check result) plus a `guidance` key — free-text instructions addressed to an LLM finalizer explaining what the findings mean, whether to post a PR comment, and how to proceed.
+
+### D7: Jenkins connectivity gate distinguishes permission failures from infrastructure failures
+
+- **Permission denied** (403, or 404 with empty job list): `require_finalizer=True` — the finalizer posts a PR comment asking the developer to grant FL the required Jenkins permissions.
+- **Infrastructure failure** (unreachable, timeout, other errors): `require_finalizer=False` — no PR comment, operational alert only.
+
+### D8: Jenkins credentials fetched from Vault
+
+The Jenkins connectivity gate fetches credentials via `get_jenkins_credentials(pipeline_url)` from `fl_control_plane/credentials.py`. No Jenkins credentials are stored in settings or `.env`.
+
+## What the Pipeline Analyzer Does
+
+1. Receives `mission_id`
+2. Reads `metadata.json` from HDLF for this mission
+3. Builds `GateEvaluationContext` from metadata
+4. For each gate in fixed order (kill switch → WIP detection → Jenkins connectivity):
+   - Checks `is_gate_active(config, context)` using the hardcoded `GateConfig`
+   - If not active: skips — writes nothing
+   - If active: calls `gate.evaluate(context, config)`, writes verdict to DB and HDLF
+5. Done — no routing, no status transitions, no downstream calls
+

+### Requirement: Jenkins connectivity gate
+The Jenkins connectivity gate SHALL verify that FL can reach and authenticate against the Jenkins instance for this mission.
+
+Three outcome buckets:
+
+- **Permission denied** (403, or 404 with empty job list): `require_finalizer = True` — the finalizer posts a PR comment asking the developer to grant FL the required Jenkins permissions.
+- **Credential issue** (401, or missing credential(s) with no anonymous access): `require_finalizer = False` — no PR comment. The FL platform team resolves these internally using the clinefl script. Details are written to HDLF findings and logged.
+- **Infrastructure failure** (unreachable, timeout, other HTTP errors): `require_finalizer = False` — no PR comment, operational alert only.
+
+Credentials are fetched from Vault via `get_jenkins_credentials(pipeline_url)`, which returns `(username, token)`. Either or both may be empty strings if the Vault secret is incomplete.
+
+Default activation (defined in `analyzer.py`):
+- `activate_for: [ActivationCondition(ci_system="jenkins")]`
+- `exclude_for: []`
+
+#### Findings contract — `auth_check` values
+
+Every gate verdict includes a findings dict with an `auth_check` field. Defined values:
+
+| `auth_check` value | Meaning |
+|---|---|
+| `"ok"` | Credentials accepted, 200 OK |
+| `"anonymous"` | No credentials present; anonymous access succeeded |
+| `"forbidden"` | 403 — Overall:Read or Job:Read missing |
+| `"not_found"` | 404 with jobs visible — build deleted or URL wrong |
+| `"expired_credentials"` | 401 — credentials most probably stale/rotated |
+| `"missing_credentials_no_anonymous_access"` | No credential(s) in Vault, anonymous attempt failed |
+| `"empty_job_list"` | 404 + empty job list — Job:Discover most probably missing |
+| `"connection_error"` | Network unreachable |
+| `"timeout"` | Request timed out |
+| `"http_<N>"` | Unexpected HTTP status N |
+| `"url_missing"` | No `pipeline_url` in context |
+| `"credential_fetch_failed"` | Vault lookup raised an exception |
+| `"domain_parse_failed"` | Could not extract domain from pipeline URL |
+| `"check_failed:<exc>"` | Stage 2 discover request raised an exception |
+
+#### Scenario: Jenkins reachable and credentials valid
+- **WHEN** FL can connect to Jenkins with valid credentials and the request returns 200 OK
+- **THEN** the gate writes `analysis_decision = CONTINUE`, `require_finalizer = False`, `message = null`, `auth_check = "ok"`
+
+#### Scenario: Jenkins returns 403
+- **WHEN** Jenkins returns 403 Forbidden
+- **THEN** the gate writes `analysis_decision = STOP`, `require_finalizer = True`, message asking developer to grant Overall Read and Job Read permissions, `auth_check = "forbidden"`
+
+#### Scenario: Jenkins returns 404 with empty job list
+- **WHEN** Jenkins returns 404 on the pipeline URL AND the discover check (`GET {domain}/api/json?tree=jobs[name]`) returns 200 OK with an empty jobs list
+- **THEN** the gate writes `analysis_decision = STOP`, `require_finalizer = True`, message asking developer to grant Job Discover permission, `auth_check = "empty_job_list"`
+- **NOTE** The 200 OK on `/api/json` confirms Overall:Read is present. An empty jobs list most probably means Job:Discover is missing — on any real Jenkins instance a user with Job:Discover would see at least one job.
+
+#### Scenario: Jenkins returns 404 with jobs visible
+- **WHEN** Jenkins returns 404 on the pipeline URL AND the discover check returns a non-empty jobs list
+- **THEN** the gate writes `analysis_decision = STOP`, `require_finalizer = False`, message indicating the build URL was not found (build likely deleted or URL incorrect), `auth_check = "not_found"`
+
+#### Scenario: Jenkins returns 401
+- **WHEN** Jenkins returns 401 Unauthorized
+- **THEN** the gate writes `analysis_decision = STOP`, `require_finalizer = False`, `message = null`, `auth_check = "expired_credentials"`
+- **NOTE** 401 most probably means the Vault credentials are stale or have been rotated. The log and HDLF findings SHALL include an instruction to update credentials using the clinefl script. No PR comment is posted — the FL platform team resolves this internally.
+
+#### Scenario: Missing credential(s) — anonymous access succeeds
+- **WHEN** `get_jenkins_credentials` returns an empty username, an empty token, or both AND a request to the pipeline URL without an Authorization header returns 200 OK
+- **THEN** the gate writes `analysis_decision = CONTINUE`, `require_finalizer = False`, `message = null`, `auth_check = "anonymous"`
+- **NOTE** The HDLF findings SHALL document that FL is running without credentials via anonymous access. No message is set on `GateVerdict` and no PR comment is posted.
+
+#### Scenario: Missing credential(s) — anonymous access fails
+- **WHEN** `get_jenkins_credentials` returns an empty username, an empty token, or both AND the anonymous request does not return 200 OK (including 403, 404, or network failure)
+- **THEN** the gate writes `analysis_decision = STOP`, `require_finalizer = False`, `message = null`, `auth_check = "missing_credentials_no_anonymous_access"`
+- **NOTE** It is not possible to distinguish between "anonymous access not granted", "permission denied", and "missing credentials" from the HTTP response alone when no credentials are present. The HDLF findings and log SHALL describe the situation as: no credential(s) found in Vault and anonymous access attempt did not succeed. No PR comment is posted — the FL platform team resolves this using the clinefl script.
+
+#### Scenario: Jenkins unreachable or infrastructure failure
+- **WHEN** the request fails with a network error, timeout, or unexpected HTTP status (not 200, 401, 403, 404)
+- **THEN** the gate writes `analysis_decision = STOP`, `require_finalizer = False`, `message = "Jenkins instance is unreachable."`, `auth_check = "connection_error"` / `"timeout"` / `"http_<N>"` as appropriate
+
+#### Scenario: Jenkins URL not available
+- **WHEN** `pipeline_url` is absent from the mission context
+- **THEN** the gate writes `analysis_decision = STOP`, `require_finalizer = False`, `message = "Jenkins URL could not be resolved for this mission."`, `auth_check = "url_missing"`
+
+#### Scenario: Non-Jenkins mission — gate not active
+- **WHEN** `ci_system` is not `jenkins`
+- **THEN** the gate is not active and is skipped
