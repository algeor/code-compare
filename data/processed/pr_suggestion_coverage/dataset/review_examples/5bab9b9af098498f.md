+def _json_escape(value: str) -> str:
+    """Escape a string for safe substitution into a JSON string literal.
+
+    The agent runtime (cline_run.py's interpolate_mcp_env_vars) substitutes MCP
+    env values into the MCP config JSON with an unescaped str.replace into a
+    "$VAR" placeholder that sits inside double quotes. A raw value containing
+    newlines or quotes — e.g. a multi-line PEM certificate or a serialized JSON
