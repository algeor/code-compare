+# Triggers the pipeline-fl-lcm GitHub Action to build and deploy (or undeploy) the
+# FL control plane to a test namespace on the dev Gardener shoot cluster.

+#   ./pipeline-fl-lcm.sh --namespace my-ns       # deploy with explicit namespace
+#   ./pipeline-fl-lcm.sh --branch feature/x      # deploy a different branch
+#   ./pipeline-fl-lcm.sh --kubeconfig path/to/kc # deploy with custom kubeconfig
+#   ./pipeline-fl-lcm.sh --push-local-changes    # stage+commit+push local changes without asking
+#   ./pipeline-fl-lcm.sh --ignore-local-changes  # deploy despite local changes (no push)
+#
+# Prerequisites:

+#   - gardenctl installed and configured

+# inside the workflow.  This check runs after any --push-local-changes push so
+# that a freshly-pushed new branch is accepted.
+if ! git -C "$REPO_DIR" ls-remote --exit-code --heads origin "$BRANCH" &>/dev/null; then
+  echo "ERROR: Branch '${BRANCH}' does not exist on the remote." >&2
+  echo "       Push it first:  git push origin ${BRANCH}" >&2
+  echo "       Or specify an existing remote branch:  --branch <name>" >&2
+  exit 1
+fi
+
+# ---------------------------------------------------------------------------
+# Derive namespace (must match the logic in the GitHub Action)
+# ---------------------------------------------------------------------------
+if [[ -z "$NAMESPACE" ]]; then
+  NAMESPACE=$(echo "$BRANCH" \
+    | tr '[:upper:]' '[:lower:]' \
+    | sed 's/[^a-z0-9]/-/g' \
+    | sed 's/-\+/-/g' \
+    | sed 's/^-//;s/-$//' \
+    | cut -c1-53)
+fi
+echo "==> Namespace: ${NAMESPACE}"
+
+# ---------------------------------------------------------------------------
+# Produce a static kubeconfig to pass to the GitHub Action.
+#
+# The dev kubeconfig uses gardenlogin as an exec credential plugin — it stores
+# only the plugin binary reference, not credentials.  kubectl config view
+# --raw copies the exec stanza verbatim, and kubectl only resolves credentials
+# in-memory during a live API call, never writing them back to disk.
+#
+# We invoke the exec plugin directly via Python, capturing the ExecCredential
+# JSON it returns (which contains clientCertificateData + clientKeyData), then
+# assemble a self-contained kubeconfig the runner can use without gardenlogin.
+# ---------------------------------------------------------------------------
+echo "==> Refreshing gardenlogin credentials..."
+
+_STATIC_KUBECONFIG="$(mktemp /tmp/fl-test-kubeconfig-XXXXXX.yaml)"
+_CRED_CAPTURE="$(mktemp /tmp/fl-test-gardenlogin-cred-XXXXXX.json)"
+_WRAPPER_SCRIPT="$(mktemp /tmp/fl-test-gardenlogin-wrapper-XXXXXX.sh)"
+_PATCHED_KC="$(mktemp /tmp/fl-test-patched-kc-XXXXXX.yaml)"
+trap 'rm -f "$_STATIC_KUBECONFIG" "$_CRED_CAPTURE" "$_WRAPPER_SCRIPT" "$_PATCHED_KC"' EXIT
+
+# We need a static kubeconfig with real credentials to pass to the GitHub
+# Action runner (which has no gardenlogin installed).  gardenlogin is an
+# exec credential plugin — kubectl sets KUBERNETES_EXEC_INFO correctly
+# (including the cluster extensions gardenlogin requires) but when we invoke
+# the binary directly we would have to reconstruct that payload exactly.
+#
+# Instead we let kubectl do the credential resolution by replacing the exec
+# command in a temp kubeconfig with a wrapper script that calls the real
+# gardenlogin and tees the ExecCredential JSON to a capture file while also
+# forwarding it to stdout so kubectl proceeds normally.
+_KC_JSON=$(kubectl --kubeconfig="$KUBECONFIG_FILE" config view --raw --minify -o json)
+
+cat > "$_WRAPPER_SCRIPT" << WRAPPER
+#!/usr/bin/env bash
+set -euo pipefail
+output=\$(kubectl-gardenlogin "\$@")
+echo "\$output" > "${_CRED_CAPTURE}"
+echo "\$output"
+WRAPPER
+chmod +x "$_WRAPPER_SCRIPT"
+
+# Build a patched kubeconfig that points the exec command at our wrapper.
+echo "$_KC_JSON" | python3 -c "
+import sys, json
+kc = json.load(sys.stdin)
+kc['users'][0]['user']['exec']['command'] = sys.argv[1]
+print(json.dumps(kc))
+" "$_WRAPPER_SCRIPT" > "$_PATCHED_KC"
+
+# Run a lightweight kubectl command — this triggers the exec plugin through
+# kubectl's normal credential resolution, which correctly populates
+# KUBERNETES_EXEC_INFO (including the cluster extensions gardenlogin needs).
+if ! kubectl --kubeconfig="$_PATCHED_KC" cluster-info &>/dev/null; then
+  echo "ERROR: kubectl cluster-info failed — gardenlogin may need a fresh session:" >&2
+  echo "         gardenctl target --garden sap-landscape-canary --project pipelinefl --shoot ppln-fl-dev" >&2
+  echo "         kubectl --kubeconfig=${KUBECONFIG_FILE} cluster-info" >&2
+  exit 1
+fi
+
+if [[ ! -s "$_CRED_CAPTURE" ]]; then
+  echo "ERROR: gardenlogin wrapper ran but produced no output." >&2
+  exit 1
+fi
+
+export _CRED_CAPTURE
+
+python3 - "$_PATCHED_KC" "$_CRED_CAPTURE" << PYEOF > "$_STATIC_KUBECONFIG"
+import base64, json, os, sys
+
+with open(sys.argv[1]) as f:
+    kc = json.load(f)
+cluster = kc["clusters"][0]["cluster"]
+server = cluster["server"]
+ca = cluster.get("certificate-authority-data", "")
+
+with open(sys.argv[2]) as f:
+    status = json.load(f).get("status", {})
+
+cert = status.get("clientCertificateData", "")
+key = status.get("clientKeyData", "")
+token = status.get("token", "")
+
+if not cert and not token:
+    sys.stderr.write("ERROR: gardenlogin returned neither a client certificate nor a token.\n")
+    sys.exit(1)
+
+# gardenlogin returns raw PEM; kubeconfig requires base64-encoded PEM.
+def _is_pem(s: str) -> bool:
+    return s.strip().startswith("-----BEGIN")
+
+def _b64(s: str) -> str:
+    return base64.b64encode(s.encode()).decode() if s and _is_pem(s) else s
+
+user_entry: dict = {}
+if cert:
+    user_entry["client-certificate-data"] = _b64(cert)
+    user_entry["client-key-data"] = _b64(key)
+else:
+    user_entry["token"] = token
+
+kc_out = {
+    "apiVersion": "v1",
+    "kind": "Config",
+    "clusters": [{"name": "fl-dev", "cluster": {"server": server, "certificate-authority-data": ca}}],
+    "users": [{"name": "fl-dev", "user": user_entry}],
+    "contexts": [{"name": "fl-dev", "context": {"cluster": "fl-dev", "user": "fl-dev"}}],
+    "current-context": "fl-dev",
+}
+print(json.dumps(kc_out))
+PYEOF
+
+_SERVER=$(python3 -c "import json,sys; kc=json.load(open(sys.argv[1])); print(kc['clusters'][0]['cluster']['server'])" "$_PATCHED_KC")
+echo "    Cluster reachable: $_SERVER"
+
+KUBECONFIG_B64=$(base64 < "$_STATIC_KUBECONFIG" | tr -d '\n')
+
+# ---------------------------------------------------------------------------
+# Trigger the workflow by piping inputs as JSON.
+# --field uses gh's @-syntax expansion which can corrupt base64 values
+# containing + or = characters; --json bypasses that entirely.
+# ---------------------------------------------------------------------------
+_INPUTS_JSON=$(python3 -c "
+import json, sys
+inputs = {
+    'action': sys.argv[1],
+    'branch': sys.argv[2],
+    'namespace': sys.argv[3],
+    'kubeconfig_b64': sys.argv[4],
+    'drop_data': sys.argv[6] == 'true',
+}
+if sys.argv[5]:
+    inputs['image_tag'] = sys.argv[5]
+print(json.dumps(inputs))
+" "$ACTION" "$BRANCH" "$NAMESPACE" "$KUBECONFIG_B64" "$IMAGE_TAG" "$DROP_DATA")
+
+echo "==> Triggering workflow '${WORKFLOW_FILE}' (action: ${ACTION})..."
+_TRIGGER_TIME=$(date -u +"%Y-%m-%dT%H:%M:%SZ")
+
+echo "$_INPUTS_JSON" | gh workflow run "$WORKFLOW_FILE" \
+  --repo "$GH_REPO" \
+  --ref "$BRANCH" \
+  --json
+
+echo "==> Waiting for workflow run to appear..."
+_RUN_ID=""
+for _i in $(seq 1 30); do
+  _RUN_ID=$(gh run list \
+    --repo "$GH_REPO" \
+    --workflow "$WORKFLOW_FILE" \
+    --branch "$BRANCH" \
+    --limit 5 \
+    --json databaseId,status,createdAt \
+    2>/dev/null \
+    | jq -r --arg since "$_TRIGGER_TIME" \
+        '[.[] | select(.createdAt >= $since)] | .[0].databaseId // empty' \
+    2>/dev/null || true)
+  if [[ -n "$_RUN_ID" && "$_RUN_ID" != "null" ]]; then
+    break
+  fi
+  sleep 3
+done
+
+if [[ -z "$_RUN_ID" || "$_RUN_ID" == "null" ]]; then
+  echo "WARNING: Could not obtain a run ID — the workflow may have completed very quickly." >&2
+  echo "         Check: gh run list --hostname ${GH_HOST} --repo ${GH_REPO} --workflow ${WORKFLOW_FILE}" >&2
+else
+  echo "==> Run ID: ${_RUN_ID}"
+  echo "    Streaming logs (Ctrl-C to stop following — the run continues in CI)..."
+  echo ""
+  gh run watch "$_RUN_ID" \
+    --repo "$GH_REPO" \
+    --exit-status || {
+    echo "" >&2
+    echo "ERROR: Workflow run failed. Full logs:" >&2
+    echo "  gh run view ${_RUN_ID} --hostname ${GH_HOST} --repo ${GH_REPO} --log" >&2
+    exit 1
+  }
+fi
+
+# ---------------------------------------------------------------------------
+# Final output
+# ---------------------------------------------------------------------------
+echo ""
+if [[ "$ACTION" == "deploy" ]]; then
+  echo "============================================================"
+  echo "  Deployed successfully"
+  echo "  Namespace : ${NAMESPACE}"
+  echo "  Branch    : ${BRANCH}"
+  echo ""
+  echo "  Useful commands:"
+  echo "    # App API"
+  echo "    kubectl --kubeconfig=${KUBECONFIG_FILE} port-forward -n ${NAMESPACE} svc/pipeline-fl-control-plane 8000:8000"
+  echo "    # Temporal UI"
+  echo "    kubectl --kubeconfig=${KUBECONFIG_FILE} port-forward -n ${NAMESPACE} svc/pipeline-fl-control-plane-temporal-web 8080:8080"
+  echo "============================================================"
+  echo ""
+  echo "${NAMESPACE}"
+else
+  echo "Namespace '${NAMESPACE}' deletion initiated — it may take a few minutes to fully terminate."
+fi
