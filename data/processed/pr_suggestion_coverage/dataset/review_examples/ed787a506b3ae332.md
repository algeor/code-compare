+"""Shared helpers for handler-managed Kubernetes resource names."""
+
+from __future__ import annotations
+
+import hashlib
+import re
+
+_K8S_RESOURCE_NAME_MAX_LENGTH = 63
+_INVALID_DNS_CHARS = re.compile(r"[^a-z0-9-]+")
+_MULTIPLE_DASHES = re.compile(r"-+")
+
+
+def _k8s_resource_name_part(value: str) -> str:
+    """Convert a CR/env-var fragment into a Kubernetes resource-name component.
+
+    ExternalSecret and Secret ``metadata.name`` values use DNS-label-like
+    Kubernetes naming rules. Env vars such as ``SONAR_TOKEN`` are valid inside
+    a Secret's data, but not as resource names because they contain uppercase
+    letters and underscores. Examples: ``SONAR_TOKEN`` becomes
+    ``sonar-token`` and ``common/path/AICORE_AUTH_URL`` becomes
+    ``common-path-aicore-auth-url``.
+    """
+    normalized = _INVALID_DNS_CHARS.sub("-", value.lower())
+    normalized = _MULTIPLE_DASHES.sub("-", normalized).strip("-")
+    return normalized or "secret"
+
+
+def handler_vault_resource_name(cr_namespace: str, cr_name: str, vault_config: str) -> str:
+    """Return the ExternalSecret/Secret name for one CR vault_config group.
+
+    A single ESO ExternalSecret can reference only one SecretStore, so each
+    FaultHandler CR gets one controller-managed ExternalSecret per vault_config.
+    """
+    digest = hashlib.sha256(f"{cr_namespace}:{cr_name}:{vault_config}".encode()).hexdigest()[:10]
+    base = (
+        f"fh-{_k8s_resource_name_part(cr_namespace)}-{_k8s_resource_name_part(cr_name)}-"
+        f"{_k8s_resource_name_part(vault_config)}"
+    )
+    max_base_length = _K8S_RESOURCE_NAME_MAX_LENGTH - len(digest) - 1
+    trimmed = base[:max_base_length].rstrip("-") or "fh-secret"
+    return f"{trimmed}-{digest}"
